+def _iso(dt: datetime | None) -> str | None:

+def _age_hours(when: datetime | None, *, now: datetime) -> float | None:
+    if when is None:
+        return None
+    w = when if when.tzinfo else when
+    n = now if now.tzinfo else now
+    # naive compare
+    if getattr(w, "tzinfo", None) and getattr(n, "tzinfo", None) is None:
+        n = n.replace(tzinfo=w.tzinfo)
+    try:
+        return max(0.0, (n - w).total_seconds() / 3600.0)
+    except TypeError:
+        return None
