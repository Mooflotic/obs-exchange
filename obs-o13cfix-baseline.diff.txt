+def _coverage_days_elapsed(started_at: datetime | None, now: datetime) -> float:
+    if started_at is None:
+        return 0.0
+    return max(0.0, (now - started_at).total_seconds() / 86400.0)

+def _complete_utc_days_with_data(db: Session, started_at: datetime | None, now: datetime) -> list[str]:
+    """UTC calendar days (YYYY-MM-DD) fully elapsed since start that have ≥1 non-meta row."""
+    if started_at is None:
+        return []
+    start_day = started_at.date()
+    today = now.date()
+    # Complete days = strictly before today.
+    days: list[str] = []
+    d = start_day
+    from datetime import timedelta
+
+    while d < today:
+        day_start = datetime(d.year, d.month, d.day)
+        day_end = day_start + timedelta(days=1)
+        n = db.scalar(
+            select(func.count())
+            .select_from(EgressObservation)
+            .where(
+                EgressObservation.scope.in_(["ext", "int"]),
+                EgressObservation.first_seen >= day_start,
+                EgressObservation.first_seen < day_end,
+            )
+        ) or 0
+        if int(n) > 0:
+            days.append(d.isoformat())
+        d = d + timedelta(days=1)
+    return days

+def evaluate_baseline_readiness(
+    db: Session, *, now: datetime | None = None
+) -> dict[str, Any]:
+    """B2 — measured criteria. Never uses deferred==0. Can and must say «non pronta»."""
+    now = now or utcnow()

+    ev = dict(row.evidence or {})
+    started = row.first_seen
+    coverage_days = _coverage_days_elapsed(started, now)
+    complete_days = _complete_utc_days_with_data(db, started, now)
+    novelty_by_day = dict(ev.get("novelty_by_day") or {})
+    # Stabilization series: complete UTC days that have a recorded novelty_new count.
+    series_days = sorted(k for k in novelty_by_day.keys() if k in set(complete_days) or k < now.date().isoformat())
+    # Prefer intersection with complete days that actually have store activity.
+    series_in_window = [d for d in complete_days if d in novelty_by_day]
+    coverage_ok = coverage_days >= float(MIN_BASELINE_COVERAGE_DAYS)
+    series_ok = len(series_in_window) >= MIN_NOVELTY_SERIES_DAYS
+    # Hard veto: window shorter than minimum coverage → never ready (even if series forged).
+    ready = bool(coverage_ok and series_ok)
+    missing = {
+        "coverage_days_needed": MIN_BASELINE_COVERAGE_DAYS,
+        "coverage_days_have": round(coverage_days, 4),
+        "coverage_days_remaining": max(0.0, MIN_BASELINE_COVERAGE_DAYS - coverage_days),
+        "novelty_series_needed": MIN_NOVELTY_SERIES_DAYS,
+        "novelty_series_have": len(series_in_window),
+        "novelty_series_remaining": max(0, MIN_NOVELTY_SERIES_DAYS - len(series_in_window)),
+        "complete_utc_days_with_rows": complete_days,
+        "novelty_by_day": {k: novelty_by_day[k] for k in series_in_window},
+    }
+    return {
+        "ready": ready,
+        "coverage_ok": coverage_ok,
+        "series_ok": series_ok,
+        "coverage_days": round(coverage_days, 4),
+        "min_coverage_days": MIN_BASELINE_COVERAGE_DAYS,
+        "min_coverage_source": "O13B M1: 3 giorni misurati; asintoto non stimabile",
+        "min_series_days": MIN_NOVELTY_SERIES_DAYS,

+def _record_cycle_novelty(db: Session, created: int, *, now: datetime) -> None:
+    """Accumulate novelty_new per UTC day (creates in this cycle count as new tuples)."""
+    row = ensure_baseline_meta(db)
+    ev = dict(row.evidence or {})
+    day = now.date().isoformat()
+    by_day = dict(ev.get("novelty_by_day") or {})
+    entry = dict(by_day.get(day) or {"novelty_new": 0, "cycles": 0})
+    entry["novelty_new"] = int(entry.get("novelty_new") or 0) + int(created or 0)
+    entry["cycles"] = int(entry.get("cycles") or 0) + 1
+    by_day[day] = entry
+    ev["novelty_by_day"] = by_day
+    ev["cycle_novelty_new"] = int(created or 0)
+    row.evidence = ev
+    db.flush()

+def mark_baseline_ready(db: Session, *, user_id: int | None = None, reason: str = "") -> dict[str, Any]:
+    """Explicit ready ONLY if B2 criteria already satisfied — else refuse (I2)."""
+    readiness = evaluate_baseline_readiness(db)

+        bl = baseline_status(db)
+        bl["refused"] = True
+        bl["refuse_reason"] = (
+            "O13C-FIX: mark_baseline_ready rifiutato — criteri B2 non soddisfatti"

+        return bl
+    row = ensure_baseline_meta(db)
+    ev = dict(row.evidence or {})
+    now = utcnow()
+    ev["baseline_ready"] = True
+    ev["baseline_status"] = "ready"
+    ev["baseline_ready_at"] = now.isoformat() + "Z"
+    ev["marked_ready_by"] = user_id
+    ev["marked_ready_reason"] = reason or "azione esplicita con criteri B2 soddisfatti"
+    ev["o13cfix_marker"] = O13CFIX_MARKER
+    row.evidence = ev
+    row.last_seen = now
+    db.flush()
+    return baseline_status(db, now=now)

+def build_egress_signals(
+    db: Session,
+    *,
+    now: datetime | None = None,
+    settings: Any | None = None,
+) -> dict[str, Any]:
+    now = now or utcnow()
+    ensure_baseline_meta(db)
+    bl = baseline_status(db, now=now)
+    novelty_on = False
+    if settings is not None:
+        novelty_on = bool(getattr(settings, "egress_novelty_signals_enabled", False))
+    acked = {
+        s.uid
+        for s in db.scalars(
+            select(Suggestion).where(
+                Suggestion.source == "egress",
+                Suggestion.status.in_(["approved", "rejected"]),
+            )
+        ).all()
+    }
+    cards = [
+        c
+        for c in novelty_egress_cards(
+            db, now=now, novelty_signals_enabled=novelty_on
+        )
+        if c["uid"] not in acked
+    ]
+    invalid = list_invalid_baseline_emissions(db)
+    open_, reason = evaluate_egress_breaker(db)
+    note_parts: list[str] = []
+    if not novelty_on:
+        note_parts.append(
+            "Segnali N5 sospesi (EGRESS_NOVELTY_SIGNALS_ENABLED=false); "
+            "ingest egress indipendente e attivo se EGRESS_INGEST_ENABLED."
+        )
+    if not bl["baseline_ready"]:
+        rd = bl.get("readiness") or {}
+        miss = rd.get("missing") or {}
+        note_parts.append(
+            f"Baseline egress in costruzione dal {bl.get('baseline_started_at')}: "
+            f"copertura {rd.get('coverage_days')}d / {rd.get('min_coverage_days')}d "
+            f"(mancano {miss.get('coverage_days_remaining')}d); "
+            f"serie novità {miss.get('novelty_series_have')}/{rd.get('min_series_days')} "
+            f"(fonte: {rd.get('min_coverage_source')}). "
+            "Nessun segnale N5 finché non pronta."
+        )
+        if bl.get("cycle_seen") is not None:
+            note_parts.append(
+                f"Ultimo ciclo: seen={bl.get('cycle_seen')} "
+                f"created={bl.get('cycle_created')} deferred={bl.get('cycle_deferred')} "
+                f"novelty_new={bl.get('cycle_novelty_new')}."
+            )
+    if invalid:
+        note_parts.append(
+            f"{len(invalid)} emissioni su baseline non valida: marcate, "
+            "ispezionabili e ripescabili (non cancellate)."
+        )
+    return {
+        "cards": cards,
+        "invalid_baseline_emissions": invalid,
+        "invalid_baseline_emissions_count": len(invalid),
+        "baseline": bl,
+        "novelty_signals_enabled": novelty_on,
+        "breaker_open": open_,
+        "breaker_reason": reason,
+        "table": table_stats(db),
+        "note": " ".join(note_parts) if note_parts else None,
+        "o13c_marker": O13C_MARKER,
+        "o13cfix_marker": O13CFIX_MARKER,
+        "defense_priority_id": DEFENSE_PRIORITY_ID,
+    }
