+/**
+ * O15 — vincolo responsive: CSS matrice senza carosello/tab/accordion.
+ */
+import assert from "node:assert/strict";
+import { readFileSync } from "node:fs";
+import { dirname, join } from "node:path";
+import { fileURLToPath } from "node:url";
+import { describe, it } from "node:test";
+
+const here = dirname(fileURLToPath(import.meta.url));
+const vue = readFileSync(
+  join(here, "components/OggiDecisionMatrix.vue"),
+  "utf8",
+);
+const style = vue.includes("<style")
+  ? vue.slice(vue.indexOf("<style"))
+  : vue;
+
+describe("O15 responsive matrix CSS", () => {
+  it("traspone a ≤768 con tre celle in linea", () => {
+    assert.match(style, /@media \(max-width: 768px\)/);
+    assert.match(style, /grid-template-areas:/);
+    assert.match(style, /"c1 c2 c3"/);
+    assert.match(style, /\.odm-actions/);
+    assert.match(style, /grid-template-columns: 1fr 1fr 1fr/);
+    // vietati pattern strutturali che nascondono alternative (non commenti)
+    for (const bad of ["carousel", "swiper", "tab-panel", "v-tabs", "role=\"tab\""]) {
+      assert.equal(style.toLowerCase().includes(bad), false, bad);
+    }
+  });
+
+  it("pulsanti senza classe primary nel markup azioni", () => {
+    const tmpl = vue.slice(0, vue.indexOf("<style"));
+    const actions = tmpl.slice(tmpl.indexOf('data-o15="matrix-actions"'));
+    assert.match(actions, /data-o15-col="approfondisci"/);
+    assert.match(actions, /data-o15-col="applica"/);
+    assert.match(actions, /data-o15-col="non_applicare"/);
+    assert.equal(/class="[^"]*primary/.test(actions), false);
+  });
+});
