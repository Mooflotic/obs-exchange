+#!/usr/bin/env python3
+"""One-shot calibration reset for OBS-DB-SLIM 3b-iii. Does NOT bump VERSION.
+
+Touches:
+  - settings key ``calibration_started_at`` = {"at": <now ISO Z>}  (day-clock epoch)
+  - DELETE FROM metric_snapshots  (serie metriche azzerata)
+
+Does NOT touch:
+  - sensor_runs, observations, observations_raw, assets, scoring_calibrated env
+  - retention / deploy / backup scripts
+
+Day-clock after reset: floor is day=1/14 (code never returns 0). Assert «0/14»
+nel prompt = inizio finestra pulita → day 1/14 con started_at = now.
+"""
+from __future__ import annotations
+
+import argparse
+from datetime import datetime, timezone
+
+from sqlalchemy import delete, func, select
+
+from app.db import _session_factory
+from app.models import MetricSnapshot, Setting
+from app.services.reliability_metrics import build_calibration_status
+from app.config import get_settings
+
+
+def main() -> None:
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("--apply", action="store_true", help="Esegue le scritture")
+    args = parser.parse_args()
+    now = datetime.now(timezone.utc).replace(tzinfo=None)
+    SessionLocal = _session_factory()
+    with SessionLocal() as db:
+        n_before = db.scalar(select(func.count()).select_from(MetricSnapshot)) or 0
+        cal_before = build_calibration_status(db, get_settings(), now=now)
+        print("BEFORE snapshots", n_before, "calibration", cal_before)
+        if not args.apply:
+            print("DRY-RUN only. Pass --apply to execute.")
+            return
+        row = db.get(Setting, "calibration_started_at")
+        payload = {"at": now.isoformat() + "Z"}
+        if row is None:
+            db.add(Setting(key="calibration_started_at", value=payload))
+        else:
+            row.value = payload
+        deleted = db.execute(delete(MetricSnapshot)).rowcount
+        db.commit()
+        cal_after = build_calibration_status(db, get_settings(), now=now)
+        n_after = db.scalar(select(func.count()).select_from(MetricSnapshot)) or 0
+        print("DELETED metric_snapshots", deleted)
+        print("AFTER snapshots", n_after, "calibration", cal_after)
+
+
+if __name__ == "__main__":
+    main()
