+    n = 0
+    for r in rows:

+    marked: list[dict[str, Any]] = []

+        r.value_json = vj
+        marked.append({"id": r.id, "port_id": int(r.subject_id), "mac": r.value_norm})
+    db.flush()
+    return marked

"""O4/O5 — read helpers for port.fdb_mac (currency path lives in facts/).

Consumers outside facts/ must use these helpers — never FactAssertion directly.
"""

from __future__ import annotations

from typing import Any

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from app.models import FactAssertion

FACT_PORT_FDB_MAC = "port.fdb_mac"
# O5 F0: spurious mac_move from O4 concurrent-presence defect (never deleted).
O4_DEFECT_MOVE_REASON = "o4_defect_concurrent_presence_move"


def _row_view(row: FactAssertion) -> dict[str, Any]:
    return {
        "id": int(row.id),
        "subject_type": row.subject_type,
        "subject_id": int(row.subject_id),
        "fact_key": row.fact_key,
        "state": row.state,
        "value_norm": row.value_norm or "",
        "value_json": dict(row.value_json or {}),
        "source": row.source,
        "authority": row.authority,
        "excl_key": row.excl_key,
        "reason": row.reason or "",
        "observed_at": row.observed_at,
        "created_at": row.created_at,
        "last_seen_at": row.last_seen_at,
        "superseded_by_id": row.superseded_by_id,
    }


def list_current_port_fdb_mac(db: Session) -> list[dict[str, Any]]:
    rows = db.scalars(
        select(FactAssertion).where(
            FactAssertion.fact_key == FACT_PORT_FDB_MAC,
            FactAssertion.state == "current",
        )
    ).all()
    return [_row_view(r) for r in rows]


def count_o4_defect_moves(db: Session) -> int:
    """How many spurious concurrent-presence moves were marked (not deleted)."""
    return int(
        db.scalar(
            select(func.count())
            .select_from(FactAssertion)
            .where(
                FactAssertion.fact_key == FACT_PORT_FDB_MAC,
                FactAssertion.reason == O4_DEFECT_MOVE_REASON,
            )
        )
        or 0
    )


def list_mac_move_supersessions(
    db: Session, *, include_defect: bool = False
) -> list[dict[str, Any]]:
    """Superseded port.fdb_mac with reason=mac_move (active) or defect marker.

    Default excludes O4 defect rows (DEBT-O4-SPURIOUS-MOVES). The count of
    excluded rows is available via count_o4_defect_moves().
    """
    if include_defect:
        rows = db.scalars(
            select(FactAssertion).where(
                FactAssertion.fact_key == FACT_PORT_FDB_MAC,
                FactAssertion.state == "superseded",
                FactAssertion.reason.in_(("mac_move", O4_DEFECT_MOVE_REASON)),
                FactAssertion.superseded_by_id.isnot(None),
            )
        ).all()
    else:
        rows = db.scalars(
            select(FactAssertion).where(
                FactAssertion.fact_key == FACT_PORT_FDB_MAC,
                FactAssertion.state == "superseded",
                FactAssertion.reason == "mac_move",
                FactAssertion.superseded_by_id.isnot(None),
            )
        ).all()
    return [_row_view(r) for r in rows]


def get_port_fdb_mac_by_id(db: Session, assertion_id: int) -> dict[str, Any] | None:
    row = db.get(FactAssertion, assertion_id)
    if row is None or row.fact_key != FACT_PORT_FDB_MAC:
        return None
    return _row_view(row)
