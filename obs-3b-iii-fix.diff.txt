+from sqlalchemy.pool import StaticPool
+
+from app.db import Base
+from app.models import Asset, Interface, Switch, SwitchPort
+from app.services.trust import _build_trust_plan, _portal_extent_from_asset
+
+
+def _session(tmp_path):
+    engine = create_engine(
+        f"sqlite:///{tmp_path / 'portal_extent.db'}",
+        connect_args={"check_same_thread": False},
+        poolclass=StaticPool,
+    )
+    Base.metadata.create_all(engine)
+    return sessionmaker(bind=engine)()
+
+
+def test_multi_mac_first_and_last_from_presence_sources(tmp_path):
+    """First on NIC-A, last on NIC-B via presence_sources (not Observation history)."""
+    db = _session(tmp_path)
+    now = datetime.utcnow().replace(microsecond=0)
+    t0 = now - timedelta(days=3)
+    t2 = now - timedelta(hours=1)
+    asset = Asset(
+        uid="multi",
+        status="noto",
+        trust_level="confirmed_present",
+        presence_state="present",
+        # portal_first_seen deliberately older/narrower than presence span
+        portal_first_seen=t0 + timedelta(hours=1),
+        portal_last_seen=t0 + timedelta(hours=2),
+        presence_sources={
+            "icmp": t0.isoformat() + "Z",
+            "nmap": t2.isoformat() + "Z",
+        },
+        meta={"operational_state": "active"},
+    )
+    db.add(asset)
+    db.flush()
+    db.add_all(
+        [
+            Interface(asset_id=asset.id, mac="AA:BB:CC:DD:EE:01"),
+            Interface(asset_id=asset.id, mac="AA:BB:CC:DD:EE:02"),
+        ]
+    )
+    db.commit()
+
+    first, last, has = _portal_extent_from_asset(asset)
+    assert has is True
+    assert first == t0
+    assert last == t2
+

+def test_fdb_newer_than_portal_drives_derived_last(tmp_path):
+    db = _session(tmp_path)
+    now = datetime.utcnow().replace(microsecond=0)
+    t_portal = now - timedelta(hours=6)
+    t_fdb = now - timedelta(minutes=10)
+    asset = Asset(
+        uid="fdb",
+        status="noto",
+        name="FDBBox",
+        trust_level="confirmed_present",
+        presence_state="present",
+        portal_first_seen=t_portal,
+        portal_last_seen=t_portal,
+        presence_sources={"icmp": t_portal.isoformat() + "Z"},
+        meta={"operational_state": "active"},
+    )
+    db.add(asset)
+    db.flush()
+    db.add(Interface(asset_id=asset.id, mac="DE:AD:BE:EF:00:01"))
+    sw = Switch(code="sw1", name="sw", ip="192.168.1.8")
+    db.add(sw)
+    db.flush()
+    db.add(
+        SwitchPort(
+            switch_id=sw.id,
+            number=1,
+            asset_id=asset.id,
+            last_fdb_at=t_fdb,
+            source="fdb",
+        )
+    )
+    db.commit()
+
+    first, last, has = _portal_extent_from_asset(asset)
+    assert has is True
+    assert first == t_portal
+    assert last == t_portal
+
+    plan = _build_trust_plan(db, now=now, use_prefetch=True)
+    action = next(a for a in plan["actions"] if a["asset_id"] == asset.id)
+    assert action["portal_first"] == t_portal
+    assert action["portal_last"] == t_portal
+    assert action["derived_last"] == t_fdb
+    assert action["has_portal_rows"] is True
