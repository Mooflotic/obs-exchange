+                bucket.sort(key=lambda r: (r.authority, r.id), reverse=True)
+            out[(sid, fk)] = bucket[0]
+    return out

+def _maybe_stale(db: Session, row: FactAssertion, spec) -> None:
+    """R-E: mark stale when ttl_window exceeded for this fact/source."""
+    if spec.ttl_window is None:

+def _check_rh_conflict(
+    db: Session,
+    entity: Any,
+    fact_key: str,
+    value_norm: str,
+    authority: int,
+    *,
+    human_confirm: bool,
+) -> ApplyResult | None:
+    global conflict_review_count
+    if human_confirm:
+        return None
+    for other in _cross_fact_incumbents(db, entity, fact_key):
+        if not semantic_contradiction(
+            incumbent_fact_key=other.fact_key,
+            incumbent_value=other.value_norm,
+            new_fact_key=fact_key,
+            new_value=value_norm,
+        ):
+            continue
+        if authority < other.authority or other.source == "manual":
+            conflict_review_count += 1
+            return ApplyResult(
+                action="conflict_review",
+                incumbent=other,
+                conflict={
+                    "incumbent_fact_key": other.fact_key,
+                    "incumbent_value": other.value_norm,
+                    "incumbent_source": other.source,
+                    "incumbent_authority": other.authority,
+                    "new_fact_key": fact_key,
+                    "new_value": value_norm,
+                },
+            )
+    return None

+def _within_readmission_cooldown(
+    db: Session,
+    subject_type: str,
+    subject_id: int,
+    fact_key: str,
+    excl_key: str,
+    value_norm: str,
+    observed_at: datetime,
+    new_authority: int,
+) -> FactAssertion | None:
+    """R-F: block weak re-admission shortly after stronger supersession."""
+    recent = db.scalars(
+        select(FactAssertion)
+        .where(
+            FactAssertion.subject_type == subject_type,
+            FactAssertion.subject_id == subject_id,
+            FactAssertion.fact_key == fact_key,
+            FactAssertion.excl_key == excl_key,
+            FactAssertion.value_norm == value_norm,
+            FactAssertion.state == "superseded",
+        )
+        .order_by(FactAssertion.valid_to.desc(), FactAssertion.id.desc())
+        .limit(1)
+    ).first()
+    if not recent or not recent.valid_to:
+        return None
+    superseder = None
+    if recent.superseded_by_id:
+        superseder = db.get(FactAssertion, recent.superseded_by_id)
+    if superseder and superseder.authority >= new_authority:
+        if observed_at - recent.valid_to < READMISSION_COOLDOWN:
+            return superseder
+    return None

+def apply_observation(
+    db: Session,
+    *,
+    fact_key: str,
+    entity: Asset | Interface | Any,
+    value: Any,
+    source: str,
+    observed_at: datetime | None = None,
+    confidence: float | None = None,
+    value_json: dict | None = None,
+    human_confirm: bool = False,
+) -> ApplyResult:
+    """Atomic, idempotent fact transition (R-A..R-H)."""
+    global weak_evidence_count
+
+    spec = get_fact_spec(fact_key)
+    src = str(source or "").lower()
+    if src not in spec.sources and src != "manual":
+        raise ValueError(f"source {source!r} not allowed for {fact_key}")
+
+    subject_type, subject_id = subject_of(fact_key, entity)
+    excl_key = spec.excl_key(subject_type, entity, value)
+    value_norm = _norm_value(value)
+    authority = authority_for(src)
+    conf = confidence if confidence is not None else default_confidence(src)
+    when = observed_at or utcnow()
+
+    incumbent = _incumbent_current(db, subject_type, subject_id, fact_key, excl_key)
+
+    rh = _check_rh_conflict(
+        db, entity, fact_key, value_norm, authority, human_confirm=human_confirm
+    )
+    if rh is not None:
+        return rh
+
+    if incumbent is None:
+        row = _new_assertion(
+            subject_type=subject_type,
+            subject_id=subject_id,
+            fact_key=fact_key,
+            excl_key=excl_key,
+            value_norm=value_norm,
+            value_json=value_json,
+            source=src,
+            authority=authority,
+            confidence=conf,
+            observed_at=when,
+            state="current",
+        )
+        db.add(row)
+        db.flush()
+        return ApplyResult(action="created", assertion=row)
+
+    # R-A refresh
+    if incumbent.value_norm == value_norm and incumbent.source == src:
+        incumbent.last_seen_at = when
+        incumbent.observed_at = when
+        if confidence is not None:
+            incumbent.confidence = conf
+        db.flush()
+        return ApplyResult(action="refresh", assertion=incumbent, incumbent=incumbent)
+
+    # R-H same-slot contradiction against manual name claims
+    if (
+        incumbent.fact_key == fact_key
+        and semantic_contradiction(
+            incumbent_fact_key=incumbent.fact_key,
+            incumbent_value=incumbent.value_norm,
+            new_fact_key=fact_key,
+            new_value=value_norm,
+        )
+        and authority < incumbent.authority
+        and not human_confirm
+    ):
+        conflict_review_count += 1
+        return ApplyResult(
+            action="conflict_review",
+            incumbent=incumbent,
+            conflict={
+                "incumbent_value": incumbent.value_norm,
+                "incumbent_source": incumbent.source,
+                "new_value": value_norm,
+                "new_source": src,
+            },
+        )
+
+    # R-D manual pinned
+    if incumbent.source == "manual" and src != "manual" and not human_confirm:
+        weak = _new_assertion(
+            subject_type=subject_type,
+            subject_id=subject_id,
+            fact_key=fact_key,
+            excl_key=excl_key,
+            value_norm=value_norm,
+            value_json=value_json,
+            source=src,
+            authority=authority,
+            confidence=conf,
+            observed_at=when,
+            state="historical",
+            reason="weak_evidence",
+        )
+        db.add(weak)
+        weak_evidence_count += 1
+        db.flush()
+        return ApplyResult(action="weak_evidence", assertion=weak, incumbent=incumbent)
+
+    # R-F readmission cooldown (anti-flap)
+    cooled = _within_readmission_cooldown(
+        db, subject_type, subject_id, fact_key, excl_key, value_norm, when, authority
+    )
+    if cooled is not None and authority < cooled.authority:
+        weak = _new_assertion(
+            subject_type=subject_type,
+            subject_id=subject_id,
+            fact_key=fact_key,
+            excl_key=excl_key,
+            value_norm=value_norm,
+            value_json=value_json,
+            source=src,
+            authority=authority,
+            confidence=conf,
+            observed_at=when,
+            state="historical",
+            reason="cooldown",
+        )
+        db.add(weak)
+        db.flush()
+        return ApplyResult(action="cooldown", assertion=weak, incumbent=cooled)
+
+    # R-C weak evidence
+    if authority < incumbent.authority and not human_confirm:
+        weak = _new_assertion(
+            subject_type=subject_type,
+            subject_id=subject_id,
+            fact_key=fact_key,
+            excl_key=excl_key,
+            value_norm=value_norm,
+            value_json=value_json,
+            source=src,
+            authority=authority,
+            confidence=conf,
+            observed_at=when,
+            state="historical",
+            reason="weak_evidence",
+        )
+        db.add(weak)
+        weak_evidence_count += 1
+        db.flush()
+        return ApplyResult(action="weak_evidence", assertion=weak, incumbent=incumbent)
+
+    # R-B supersession — demote incumbent before inserting new current (partial unique index).
+    incumbent.valid_to = when
+    incumbent.state = "superseded"
+    db.flush()
+
+    new_row = _new_assertion(
+        subject_type=subject_type,
+        subject_id=subject_id,
+        fact_key=fact_key,
+        excl_key=excl_key,
+        value_norm=value_norm,
+        value_json=value_json,
+        source=src,
+        authority=authority,
+        confidence=conf,
+        observed_at=when,
+        state="current",
+    )
+    db.add(new_row)
+    db.flush()
+
+    incumbent.superseded_by_id = new_row.id
+    db.flush()
+    return ApplyResult(action="superseded", assertion=new_row, incumbent=incumbent)
+

+def canonical_name_for_entity(db: Session, entity: Asset | Interface) -> str | None:

+def expire_stale_facts(db: Session, *, now: datetime | None = None) -> int:
+    """R-E batch: mark expired currents as stale (ttl_window None → skip)."""
+    when = now or utcnow()
+    changed = 0
+    rows = db.scalars(select(FactAssertion).where(FactAssertion.state == "current")).all()
+    for row in rows:
+        spec = get_fact_spec(row.fact_key)
+        if spec.ttl_window is None:
+            continue
+        if row.last_seen_at and when - row.last_seen_at > spec.ttl_window:
+            row.state = "stale"
+            row.valid_to = when
+            changed += 1
+    if changed:
+        db.flush()
+    return changed
