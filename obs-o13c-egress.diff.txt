+def _norm_mac(mac: str | None) -> str:
+    return (mac or "").replace("-", ":").strip().upper()

+def _parse_ts(value: Any) -> datetime | None:
+    if value is None:
+        return None
+    if isinstance(value, datetime):
+        return value.replace(tzinfo=None) if value.tzinfo else value
+    if isinstance(value, (int, float)):
+        return datetime.utcfromtimestamp(float(value))
+    text = str(value).strip()
+    if not text:
+        return None
+    try:
+        return datetime.utcfromtimestamp(float(text))
+    except ValueError:
+        pass
+    try:
+        if text.endswith("Z"):
+            text = text[:-1] + "+00:00"
+        dt = datetime.fromisoformat(text)
+        if dt.tzinfo is not None:
+            from datetime import timezone
+
+            return dt.astimezone(timezone.utc).replace(tzinfo=None)
+        return dt
+    except ValueError:
+        return None

+def _dedup_key(kind: str, value: str, mac: str, ip: str, association: str) -> str:
+    raw = f"{kind}|{value}|{mac or ip}|{association}"
+    return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:64]

+def evidence_table_stats(db: Session) -> dict[str, Any]:
+    total = db.scalar(select(func.count()).select_from(ZeekBehaviorEvidence)) or 0
+    today_start = utcnow().replace(hour=0, minute=0, second=0, microsecond=0)
+    today = (
+        db.scalar(
+            select(func.count())
+            .select_from(ZeekBehaviorEvidence)
+            .where(ZeekBehaviorEvidence.first_seen >= today_start)
+        )
+        or 0
+    )
+    # Approximate size via payload lengths (no DB file maintenance).
+    rows = db.scalars(select(ZeekBehaviorEvidence)).all()
+    approx = sum(
+        len((r.dedup_key or ""))
+        + len(r.kind or "")
+        + len(r.value or "")
+        + len(r.src_ip or "")
+        + len(r.src_mac or "")
+        + 64
+        for r in rows
+    )
+    return {
+        "rows_total": int(total),
+        "rows_today": int(today),
+        "approx_bytes": int(approx),
+        "ceilings": {
+            "rows_total": MAX_EVIDENCE_ROWS,
+            "rows_today": MAX_GROWTH_PER_DAY,
+            "bytes": MAX_TABLE_BYTES,
+        },
+    }

+def evaluate_evidence_breaker(db: Session) -> tuple[bool, str]:
+    st = evidence_table_stats(db)
+    if st["rows_total"] >= MAX_EVIDENCE_ROWS:
+        return True, f"zeek_behavior_evidence rows_total={st['rows_total']} >= {MAX_EVIDENCE_ROWS}"
+    if st["rows_today"] >= MAX_GROWTH_PER_DAY:
+        return True, f"zeek_behavior_evidence rows_today={st['rows_today']} >= {MAX_GROWTH_PER_DAY}"
+    if st["approx_bytes"] >= MAX_TABLE_BYTES:
+        return True, f"zeek_behavior_evidence approx_bytes={st['approx_bytes']} >= {MAX_TABLE_BYTES}"
+    return False, ""

+def freshness_for_observed_at(observed_at: datetime | None, *, now: datetime | None = None) -> dict[str, Any]:
+    """Reuse FDB_STALE_HOURS window — same vocabulary as O2/O5/O10 (no second vocab)."""
+    now = now or utcnow()
+    if observed_at is None:
+        return {
+            "state": "not_measured",
+            "display": "non misurato",
+            "source": SENSOR_ID,
+            "age_hours": None,
+            "window_hours": FDB_STALE_HOURS,
+        }
+    age_h = max(0.0, (now - observed_at).total_seconds() / 3600.0)
+    if age_h > FDB_STALE_HOURS:
+        return {
+            "state": "stale",
+            "display": "vecchio",
+            "source": SENSOR_ID,
+            "age_hours": round(age_h, 3),
+            "window_hours": FDB_STALE_HOURS,
+            "last_at": observed_at.isoformat() + "Z",
+            "note": f"età {age_h:.1f} h > finestra {FDB_STALE_HOURS} h",
+        }
+    return {
+        "state": "fresh",
+        "display": "fresco",
+        "source": SENSOR_ID,
+        "age_hours": round(age_h, 3),
+        "window_hours": FDB_STALE_HOURS,
+        "last_at": observed_at.isoformat() + "Z",
+    }

+def upsert_evidence_item(db: Session, item: dict[str, Any]) -> str:
+    """Upsert one fingerprint observation. Returns action created|refreshed|skipped_*."""
+    kind = str(item.get("kind") or "").strip()
+    value = str(item.get("value") or "").strip()
+    if kind not in KINDS_FP or not value or value in ("-", "null"):
+        return "skipped_invalid"
+
+    open_, reason = evaluate_evidence_breaker(db)
+    if open_:
+        log.error("zeek_behavior breaker open: %s", reason)
+        return "skipped_breaker"
+
+    observed_at = _parse_ts(item.get("observed_at")) or utcnow()
+    src_ip = str(item.get("src_ip") or "").strip()
+    src_mac = _norm_mac(item.get("src_mac"))
+
+    if src_mac:
+        assoc = associate_mac_direct(db, src_mac)
+        if src_ip:
+            # Keep IP for display; association remains mac_direct.
+            pass
+    elif src_ip:
+        assoc = associate_ip_at(db, src_ip, observed_at)
+        src_mac = assoc.mac or src_mac
+    else:
+        return "skipped_no_identity"
+
+    subject_type, subject_id, asset_id = subject_for_association(assoc)
+    association = assoc.status
+    if association == "unresolved" and not src_mac and not assoc.asset_id:
+        # Still store unresolved IP-only evidence (I2).
+        pass
+
+    key = _dedup_key(kind, value, src_mac, src_ip, association)
+    existing = db.scalar(
+        select(ZeekBehaviorEvidence).where(ZeekBehaviorEvidence.dedup_key == key)
+    )
+    payload = {
+        "assoc_reason": assoc.reason,
+        "fact_id": assoc.fact_id,
+        "fact_state": assoc.fact_state,
+        "ja4s": item.get("ja4s"),
+        "server_name": item.get("server_name"),
+        "uid": item.get("uid"),
+        "o11_marker": "obs-o11-evidence",
+    }
+    if existing:
+        existing.last_seen = max(existing.last_seen or observed_at, observed_at)
+        existing.observation_count = int(existing.observation_count or 0) + 1
+        if subject_type and not existing.subject_type:
+            existing.subject_type = subject_type
+            existing.subject_id = subject_id
+        if asset_id and not existing.asset_id:
+            existing.asset_id = asset_id
+        if src_mac and not existing.src_mac:
+            existing.src_mac = src_mac
+        ev = dict(existing.evidence or {})
+        ev.update({k: v for k, v in payload.items() if v is not None})
+        existing.evidence = ev
+        db.flush()
+        return "refreshed"
+
+    row = ZeekBehaviorEvidence(
+        dedup_key=key,
+        kind=kind,
+        value=value[:255],
+        src_ip=src_ip,
+        src_mac=src_mac,
+        association=association,
+        subject_type=subject_type or "",
+        subject_id=subject_id,
+        asset_id=asset_id,
+        sensor_id=SENSOR_ID,
+        first_seen=observed_at,
+        last_seen=observed_at,
+        observation_count=1,
+        evidence=payload,
+    )
+    db.add(row)
+    db.flush()
+    return "created"

+def ingest_zeek_behavior_batch(db: Session, items: list[dict[str, Any]]) -> dict[str, Any]:
+    ensure_baseline_meta(db)
+    stats = {
+        "received": len(items),
+        "created": 0,
+        "refreshed": 0,
+        "skipped_invalid": 0,
+        "skipped_breaker": 0,
+        "skipped_no_identity": 0,
+    }
+    for raw in items:
+        if not isinstance(raw, dict):
+            stats["skipped_invalid"] += 1
+            continue
+        action = upsert_evidence_item(db, raw)
+        if action in stats:
+            stats[action] += 1
+        elif action.startswith("skipped_"):
+            stats[action] = stats.get(action, 0) + 1
+    open_, reason = evaluate_evidence_breaker(db)
+    st = evidence_table_stats(db)
+    return {
+        **stats,
+        "breaker_open": open_,
+        "breaker_reason": reason,
+        "table": st,
+        "baseline": baseline_status(db),
+        "o11_marker": "obs-o11-ingest",
+    }

+def _evidence_for_asset(db: Session, asset: Asset) -> list[ZeekBehaviorEvidence]:
+    ids = _member_asset_ids(db, asset)
+    chassis_id = int(asset.chassis_id) if asset.chassis_id else None
+    clauses = [ZeekBehaviorEvidence.asset_id.in_(ids)]
+    if chassis_id is not None:
+        clauses.append(
+            (ZeekBehaviorEvidence.subject_type == "chassis")
+            & (ZeekBehaviorEvidence.subject_id == chassis_id)
+        )
+    # Also by MAC of all member interfaces
+    macs = [
+        _norm_mac(i.mac)
+        for i in db.scalars(select(Interface).where(Interface.asset_id.in_(ids))).all()
+        if i.mac
+    ]
+    if macs:
+        clauses.append(ZeekBehaviorEvidence.src_mac.in_(macs))
+
+    from sqlalchemy import or_
+
+    rows = list(
+        db.scalars(
+            select(ZeekBehaviorEvidence)
+            .where(
+                ZeekBehaviorEvidence.kind != BASELINE_META_KIND,
+                or_(*clauses) if clauses else ZeekBehaviorEvidence.id < 0,
+            )
+            .order_by(ZeekBehaviorEvidence.last_seen.desc())
+        ).all()
+    )
+    # Dedupe by id
+    seen: set[int] = set()
+    out: list[ZeekBehaviorEvidence] = []
+    for r in rows:
+        if r.id in seen:
+            continue
+        seen.add(r.id)
+        out.append(r)
+    return out

+def structural_character_changes(
+    db: Session, *, now: datetime | None = None
+) -> list[dict[str, Any]]:
+    """New JA4/JA4D/dhcp FP not in prior set for subject. Gated by baseline_ready."""
+    now = now or utcnow()
+    bl = baseline_status(db)
+    if not bl["baseline_ready"]:
+        return []
+
+    rows = list(
+        db.scalars(
+            select(ZeekBehaviorEvidence).where(
+                ZeekBehaviorEvidence.kind.in_(["ja4", "ja4d", "dhcp_param_list", "dhcp_vendor"])
+            )
+        ).all()
+    )
+    by_subj: dict[tuple[str, int], list[ZeekBehaviorEvidence]] = {}
+    for r in rows:
+        if not r.subject_type or r.subject_id is None:
+            continue
+        by_subj.setdefault((r.subject_type, int(r.subject_id)), []).append(r)
+
+    cards: list[dict[str, Any]] = []
+    for (stype, sid), group in by_subj.items():
+        # Prior set = values whose first_seen is strictly before the newest first_seen of another value
+        by_kind_value: dict[tuple[str, str], ZeekBehaviorEvidence] = {}
+        for r in group:
+            key = (r.kind, r.value)
+            prev = by_kind_value.get(key)
+            if prev is None or (r.first_seen and prev.first_seen and r.first_seen < prev.first_seen):
+                by_kind_value[key] = r
+
+        # Character change: a fingerprint whose first_seen is after at least one other
+        # distinct fingerprint of same kind already existed for the subject.
+        for (kind, value), r in by_kind_value.items():
+            peers = [
+                p
+                for (k, v), p in by_kind_value.items()
+                if k == kind and v != value and p.first_seen and r.first_seen and p.first_seen < r.first_seen
+            ]
+            if not peers:
+                continue
+            # Also require the new FP last_seen to be fresh enough to surface
+            fr = freshness_for_observed_at(r.last_seen, now=now)
+            if fr["state"] == "stale":
+                continue
+            asset_id = r.asset_id
+            label = f"{stype}:{sid}"
+            if stype == "asset" and asset_id:
+                a = db.get(Asset, asset_id)
+                label = (a.name if a and a.name else None) or f"asset {asset_id}"
+            elif stype == "chassis":
+                ch = db.get(Chassis, sid)
+                label = (ch.label if ch and ch.label else None) or f"chassis {sid}"
+            cards.append(
+                {
+                    "uid": f"beh:cc:{stype}:{sid}:{kind}:{hashlib.sha256(value.encode()).hexdigest()[:12]}",
+                    "signal": "B-C",
+                    "kind": "behavior_character_change",
+                    "title": "dispositivo cambiato carattere",
+                    "subject_type": stype,
+                    "subject_id": sid,
+                    "asset_id": asset_id,
+                    "label": label,
+                    "new_fingerprint": {"kind": kind, "value": value},
+                    "prior_fingerprints": [
+                        {"kind": kind, "value": p.value} for p in peers[:8]
+                    ],
+                    "association": r.association,
+                    "freshness": fr,
+                    "defense_priority_id": "conflitto_identita",
+                    "o11_marker": "obs-o11-character",
+                    "actions": [
+                        {
+                            "id": "open_dossier",
+                            "label": "Apri Dossier",
+                            "href": f"/dossier/{asset_id}" if asset_id else "/dossier",
+                            "reversible": True,
+                        },
+                        {
+                            "id": "acknowledge",
+                            "label": "Riconosci / archivia segnale",
+                            "requires_confirm": True,
+                            "reversible": True,
+                        },
+                    ],
+                }
+            )
+    return cards

+def unknown_with_evidence_cards(db: Session, *, now: datetime | None = None) -> list[dict[str, Any]]:

+def _age_hours(when: datetime | None, *, now: datetime) -> float | None:
+    if when is None:
