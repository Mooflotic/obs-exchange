+export function priorityForFdbCard(card) {

+  const role = String(card?.ruolo_porta || "");
+  const uncertainRole = role === "non_determinato";
+  let p;

+    p = defensePriority(
+      "mac_move_accesso",

+        ? "S-C · ruolo_porta non_determinato · priorità dichiarata ma ridotta in UI"
+        : "S-C MAC-move fra porte di accesso",
+    );
+  } else if (signal === "S-A" || signal === "S-D") {
+    p = defensePriority(
+      "sorgente_non_disponibile",
+      `${signal} · ${card?.state || "sorgente_non_disponibile"}`,
+    );
+  } else {
+    p = defensePriority("igiene_nome", `segnale FDB ${signal || "sconosciuto"}`);
