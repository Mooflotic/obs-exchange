+/**
+ * Deriva i tre simboli da proprietà oggettive. Mai da output AI.
+ * @param {MatrixEvidence} ev
+ * @returns {Record<MatrixColumn, MatrixSymbol>}
+ */
+export function symbolsForEvidence(ev) {
+  const rank = proposalSourceRank(ev.source || "");
+  const weak =
+    ev.weak_source === true ||
+    (ev.source != null && rank > 0 && rank < 80);
+  const stale =
+    ev.freshness === "stale" ||
+    ev.freshness === "not_measured" ||
+    ev.freshness === "not_covered";
+  const missing = ev.present === false || ev.freshness === "not_covered";
+  const divergence = ev.divergence === true;
+  const needsVerify =
+    ev.needs_verify === true || weak || stale || missing || divergence;
+  const settled = ev.already_settled === true;
+
+  /** @type {MatrixSymbol} */
+  let approfondisci = "—";
+  if (ev.band === "inference") {
+    // Inferenza: può motivare verifica, mai «saldare» da sola
+    approfondisci = needsVerify || ev.needs_verify !== false ? "+" : "?";
+  } else if (settled && !divergence && !stale) {
+    approfondisci = "–";
+  } else if (needsVerify) {
+    approfondisci = "+";
+  } else if (ev.kind && ev.present === true && !weak && !stale) {
+    approfondisci = "–";
+  } else {
+    approfondisci = "?";
+  }
+
+  /** @type {MatrixSymbol} */
+  let applica = "—";
+  if (ev.band === "inference") {
+    applica = "?"; // I1: inferenza non sostiene da sola l'applicazione
+  } else if (ev.opposes_apply === true || divergence) {
+    applica = "–";
+  } else if (ev.supports_apply === true) {
+    applica = "+";
+  } else if (ev.kind === "proposal" && !divergence) {
+    applica = weak || stale ? "?" : "+";
+  } else if (ev.kind === "current_name" && settled) {
+    applica = "–"; // nome già fatto: contraddice nuova applicazione
+  } else if (missing && ev.kind !== "proposal") {
+    applica = "?";
+  } else {
+    applica = "—";
+  }
+
+  /** @type {MatrixSymbol} */
+  let non_applicare = "—";
+  if (ev.band === "inference") {
+    non_applicare = "?";
+  } else if (ev.supports_dismiss === true) {
+    non_applicare = "+";
+  } else if (ev.supports_apply === true && !divergence) {
+    non_applicare = "–";
+  } else if (ev.kind === "proposal" && (weak || ev.supports_dismiss)) {
+    non_applicare = weak && !ev.supports_apply ? "+" : "?";
+  } else if (divergence || ev.opposes_apply) {
+    non_applicare = "+";
+  } else if (ev.kind === "proposal") {
+    non_applicare = "?";
+  } else {
+    non_applicare = "—";
+  }
+
+  return { approfondisci, applica, non_applicare };
+}

+/**
+ * Destinazione APPROFONDISCI dal tipo di evidenza (non nuove colonne).
+ * @param {MatrixEvidence[]} rows
+ * @returns {'dossier'|'plant'|'topology'}
+ */
+export function deepenDestinationForRows(rows) {
+  for (const r of rows) {
+    if (r.deepen_dest === "plant" || r.deepen_dest === "topology" || r.deepen_dest === "dossier") {
+      return r.deepen_dest;
+    }
+  }
+  const kinds = new Set(rows.map((r) => r.kind || ""));
+  if ([...kinds].some((k) => /fdb|port|switch|mac_move|placement/i.test(k))) {
+    return "plant";
+  }
+  if ([...kinds].some((k) => /topo|path|wifi/i.test(k))) {
+    return "topology";
+  }
+  return "dossier";
+}

+/**
+ * @param {MatrixEvidence[]} rows
+ * @returns {{ id: string, label: string }[]} ids duplicati (stesso id)
+ */
+export function findDuplicateEvidenceIds(rows) {
+  const seen = new Map();
+  const dup = [];
+  for (const r of rows) {
+    const n = (seen.get(r.id) || 0) + 1;
+    seen.set(r.id, n);
+    if (n === 2) dup.push({ id: r.id, label: r.label });
+  }
+  return dup;
+}

+/**
+ * Raggruppa per banda nell'ordine I1; bande vuote restano dichiarate.
+ * @param {MatrixEvidence[]} rows
+ */
+export function groupEvidenceByBand(rows) {
+  /** @type {Record<MatrixBand, MatrixEvidence[]>} */
+  const groups = {
+    observed: [],
+    missing_stale: [],
+    deterministic: [],
+    inference: [],
+  };
+  for (const r of rows) {
+    const b = MATRIX_BANDS.includes(r.band) ? r.band : "observed";
+    groups[b].push(r);
+  }
+  return MATRIX_BANDS.map((band) => ({
+    band,
+    label: MATRIX_BAND_LABELS[band],
+    rows: groups[band],
+    empty: groups[band].length === 0,
+  }));
+}

+/**
+ * Freschezza da stringa già in vocabolario progetto → id FRESHNESS.
+ * @param {string|null|undefined} raw
+ * @returns {FreshnessId}
+ */
+export function normalizeFreshness(raw) {
+  const s = String(raw || "")
+    .trim()
+    .toLowerCase();
+  if (!s) return "not_measured";
+  if (s.includes("non coperto") || s.includes("not_covered") || s === "cieca")
+    return "not_covered";
+  if (
+    s.includes("non misur") ||
+    s.includes("not_measured") ||
+    s.includes("assente") ||
+    s === "disabilitata"
+  )
+    return "not_measured";
+  if (
+    s.includes("vecchio") ||
+    s.includes("stale") ||
+    s.includes("coperta_vecchia") ||
+    s.includes("non_fresca")
+  )
+    return "stale";
+  if (s.includes("fresco") || s.includes("fresh") || s.includes("coperta_fresca"))
+    return "fresh";
+  return "not_measured";
+}

+ * @param {object} ctx

+export function matrixActionsFor(family, ctx) {

+    let applyHandler = "confirm";
+    if (card.actions?.includes("adotta_proposta") || card.actions?.includes("sostituisci_proposta")) {
+      applyLabel = "Adotta";
+      applyHandler = "adopt";
+    } else if (card.actions?.includes("conferma_manuale")) {
+      applyLabel = "Conferma";
+      applyHandler = "confirm";
+    } else if (card.actions?.includes("correggi")) {
+      applyLabel = card.name_kind === "absent" ? "Assegna" : "Correggi";
+      applyHandler = "rename";
+    }
