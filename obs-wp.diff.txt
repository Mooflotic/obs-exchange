+def _parse(raw):
+    text = str(raw or "").strip()
+    if not text:
+        return None
+    try:
+        dt = datetime.fromisoformat(text.replace("Z", "+00:00"))
+    except ValueError:
+        return None
+    return dt.replace(tzinfo=None) if dt.tzinfo else dt

+def has_fresh_fritz_active(asset, now, stale_hours=24):
+    fritz = dict(dict((asset.meta or {}).get("discovery") or {}).get("fritz") or {})
+    if fritz.get("active") is not True:
+        return False
+    seen = _parse(fritz.get("last_observed_at"))
+    if seen is None:
+        return False
+    return seen >= now - timedelta(hours=max(1, stale_hours))
+
+
+def new_classify(asset, now, stale_hours=24):
+    protected = bool(
+        (asset.name or "").strip()
+        or asset.status == "noto"
+        or asset.is_critical
+        or (asset.meta or {}).get("manual_overrides")
+        or (asset.meta or {}).get("infrastructure_identity")
+    )
+    recent = bool(
+        asset.portal_last_seen
+        and asset.portal_last_seen >= now - timedelta(hours=max(1, stale_hours))
+    )
+    if recent:
+        return "confirmed_present"
+    if protected:
+        return "known"
+    if has_fresh_fritz_active(asset, now, stale_hours):
+        return "known"
+    if asset.portal_last_seen:
+        return "stale_unlocated"
+    return "fritz_historical"
+
+
+def expected_op(level):
+    if level == "fritz_historical":
+        return "fritz_historical"
+    if level == "stale_unlocated":
+        return "stale_unlocated"
+    return "active"
