+def prune_legacy_observations(
+    db: Session, *, retention_days: int, now: Optional[datetime] = None
+) -> int:
+    """Delete legacy ``observations`` rows older than retention_days.
+
+    Dual-write path (materialize → record_observation) still appends for presence
+    evidence (~24h horizon). TTL must stay ≥ presence stale window; default 3d.
+    Does not VACUUM — file size shrinks only with an offline vacuum.
+    """
+    now = now or datetime.utcnow()
+    cutoff = now - timedelta(days=max(0, retention_days))
+    deleted = db.execute(
+        delete(Observation).where(Observation.seen_at < cutoff)
+    ).rowcount
+    return int(deleted or 0)
