+def _iso(dt: datetime | None) -> str | None:
+    if dt is None:
+        return None
+    if dt.tzinfo is None:
+        return dt.isoformat() + "Z"
+    return dt.isoformat()
+
+
+def _age_hours(when: datetime | None, *, now: datetime) -> float | None:
+    if when is None:
+        return None
+    w = when if when.tzinfo else when
+    n = now if now.tzinfo else now
+    # naive compare
+    if getattr(w, "tzinfo", None) and getattr(n, "tzinfo", None) is None:
+        n = n.replace(tzinfo=w.tzinfo)
+    try:
+        return max(0.0, (n - w).total_seconds() / 3600.0)
+    except TypeError:
+        return None

+def _load_asset(db: Session, asset_id: int) -> Asset | None:
+    return db.scalar(
+        select(Asset)
+        .where(Asset.id == int(asset_id))
+        .options(
+            joinedload(Asset.interfaces).joinedload(Interface.addresses),
+            joinedload(Asset.name_proposals),
+        )
+    )

+def _primary_mac(asset: Asset) -> str | None:
+    for iface in asset.interfaces or []:
+        m = normalize_mac(iface.mac or "")
+        if m:
+            return m
+    return None

+def _current_ip(asset: Asset) -> str | None:
+    for iface in asset.interfaces or []:
+        for addr in iface.addresses or []:
+            if addr.is_current and addr.ip:
+                return addr.ip
+    return None

+def _name_block(db: Session, asset: Asset) -> dict[str, Any]:
+    """Cos'è — name with provenance (I2: nome non noto)."""
+    # Prefer chassis-scoped fact if present
+    chassis_id = getattr(asset, "chassis_id", None)
+    if chassis_id is not None:
+        chassis_id = int(chassis_id)
+
+    fact = None
+    subject = None
+    if chassis_id:
+        fact = current(db, "chassis", int(chassis_id), "asset.name")
+        subject = ("chassis", int(chassis_id))
+    if fact is None:
+        fact = current(db, "asset", int(asset.id), "asset.name")
+        subject = ("asset", int(asset.id))
+
+    held = (asset.name or "").strip()
+    if fact and (fact.value_norm or "").strip():
+        return {
+            "name": fact.value_norm.strip(),
+            "name_kind": "manual" if fact.source == "manual" else "fact",
+            "provenance": {
+                "source": fact.source,
+                "authority": fact.authority,
+                "subject_type": subject[0] if subject else None,
+                "subject_id": subject[1] if subject else None,
+                "observed_at": _iso(fact.observed_at),
+                "label": "manuale" if fact.source == "manual" else f"fatto ({fact.source})",
+            },
+            "held_legacy_name": held if held and held != fact.value_norm.strip() else None,

+def _history_block(db: Session, mac: str | None, placement: dict[str, Any]) -> dict[str, Any]:
+    if not mac:
+        return {
+            "available": False,
+            "note": "storia non disponibile — MAC assente (I2)",
+            "events": [],
+            "excluded_defect_count": 0,
+        }
+    hist = list_mac_fdb_history(db, mac)
+    events_out = []
+    for ev in hist["events"]:
+        vj = ev.get("value_json") or {}
+        events_out.append(
+            {
+                "id": ev["id"],
+                "state": ev["state"],
+                "reason": ev["reason"] or None,
+                "port_id": ev["subject_id"],
+                "switch": vj.get("switch"),
+                "port": vj.get("port"),
+                "observed_at": _iso(ev.get("observed_at")),
+                "superseded_by_id": ev.get("superseded_by_id"),
+            }
+        )
+    bl = placement.get("baseline") or {}
+    quality = bl.get("quality") or "absent"
+    if quality == "reconstructed":
+        first_seen_kind = "lower_bound_reconstructed"
+        first_seen_label = (
+            f"presente alla ricostruzione della baseline del {bl.get('baseline_at')}, "
+            "storia precedente non disponibile"
+        )
+    elif quality == "verified":
+        first_seen_kind = "baseline_verified"
+        first_seen_label = (
+            f"presente alla baseline verificata del {bl.get('baseline_at')}"
+        )
+    elif events_out:
+        first_seen_kind = "observed_first"
+        first_seen_label = f"prima osservazione FDB in catena: {events_out[0].get('observed_at')}"
+    else:
+        first_seen_kind = "absent"
+        first_seen_label = "storia FDB assente (I2)"
+
+    return {
+        "available": bool(events_out) or hist["excluded_defect_count"] > 0,
+        "events": events_out,
+        "event_count": len(events_out),
+        "excluded_defect_count": hist["excluded_defect_count"],
+        "excluded_note": (
+            f"{hist['excluded_defect_count']} eventi esclusi per difetto noto "
+            f"({hist['excluded_defect_reason']}) — vedi {hist['debt']}"
+            if hist["excluded_defect_count"]
+            else None
+        ),
+        "debt": hist["debt"],
+        "first_seen": {
+            "kind": first_seen_kind,
+            "label": first_seen_label,
+            "baseline_at": bl.get("baseline_at"),
+            "baseline_quality": quality,
+        },
+    }

+def _expected_block(asset: Asset, name: dict[str, Any], solo_l2: bool) -> dict[str, Any]:
+    status = (asset.status or "").strip() or "nuovo"
+    if status == "noto":
+        state = "atteso"
+    elif status == "ignorato":
+        state = "ignorato"
+    else:
+        state = "non_riconosciuto"
+    return {
+        "state": state,
+        "status_raw": status,
+        "solo_l2": solo_l2,
+        "central": solo_l2 or state == "non_riconosciuto" or name.get("name_kind") == "absent",
+        "ignore_reason": (asset.notes or "").strip() if status == "ignorato" else None,
+        "label": {
+            "atteso": "Atteso (riconosciuto)",
+            "non_riconosciuto": "Non riconosciuto",
+            "ignorato": "Ignorato",
+        }.get(state, state),
+    }

+                "source": "topology_fdb",
+                "confidence": 0.65,
