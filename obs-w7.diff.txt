+def _asset(db, **kw):
+    a = Asset(uid=str(uuid.uuid4()), name=kw.pop("name", ""), category=kw.pop("category", "unknown"), meta=kw.pop("meta", {}))
+    for k, v in kw.items():
+        setattr(a, k, v)
+    db.add(a)
+    db.flush()
+    return a

+def _iface(db, asset, mac, ip=None, is_current=True):
+    i = Interface(asset_id=asset.id, mac=mac)
+    db.add(i)
+    db.flush()
+    if ip is not None:
+        db.add(IpAddress(interface_id=i.id, ip=ip, is_current=is_current, source="dhcp"))
+        db.flush()
+    return i
