+"""K7 circularity guards for identity evidence.
+
+Rejects:
+
+- ``chassis_id`` used as its own evidence (tautology).
+- Evidence whose provenance already cites the link being justified.
+- Evidence produced by a post-merge state that is being re-read as
+  premise for the same merge.
+
+These functions are intentionally simple — they detect textual/structural
+loops on ``subject_hint_type/id``, ``value_norm`` and ``provenance``, and
+must be called by the linker BEFORE building a proposal.
+"""
+
+from __future__ import annotations
+
+from typing import Any, Iterable
+
+
+def is_self_referential_chassis_evidence(evidence: Any) -> bool:
+    """True when a chassis-scoped row uses its own chassis id as value.
+
+    Detects patterns like ``fact_type='chassis_id_declared'`` with
+    ``value_norm`` equal to ``str(subject_hint_id)`` or ``chassis:<id>``.
+    """
+    subj_type = getattr(evidence, "subject_hint_type", "") or ""
+    subj_id = getattr(evidence, "subject_hint_id", None)
+    value = str(getattr(evidence, "value_norm", "") or "").strip()
+    fact = str(getattr(evidence, "fact_type", "") or "")
+
+    if subj_type != "chassis" or subj_id is None:
+        return False
+    if fact not in {"chassis_id", "chassis_id_declared", "group_id"}:
+        return False
+    candidates = {str(subj_id), f"chassis:{subj_id}", f"chassis-{subj_id}"}
+    return value in candidates
+
+
+def _provenance_mentions_pair(
+    provenance: dict | None, *, left_ref: str, right_ref: str
+) -> bool:
+    if not isinstance(provenance, dict):
+        return False
+    justifies = provenance.get("justifies") or provenance.get("link") or {}
+    if isinstance(justifies, dict):
+        left = str(justifies.get("left_ref") or justifies.get("left") or "")
+        right = str(justifies.get("right_ref") or justifies.get("right") or "")
+        pair = {left, right}
+        if pair == {str(left_ref), str(right_ref)}:
+            return True
+    for tag in provenance.get("derived_from") or []:
+        if isinstance(tag, str) and tag in {left_ref, right_ref, f"link:{left_ref}~{right_ref}"}:
+            return True
+    return False
+
+
+def is_circular_for_link(
+    evidence: Any, *, left_ref: str, right_ref: str
+) -> bool:
+    """True when the row's provenance already cites this exact link.
+
+    The linker must reject such evidence rows to satisfy K7.
+    """
+    provenance = getattr(evidence, "provenance", None)
+    return _provenance_mentions_pair(
+        provenance, left_ref=left_ref, right_ref=right_ref
+    )
+
+
+def filter_non_circular(
+    evidence_rows: Iterable[Any],
+    *,
+    left_ref: str,
+    right_ref: str,
+) -> list[Any]:
+    """Drop self-referential and link-circular rows."""
+    keep: list[Any] = []
+    for row in evidence_rows:
+        if is_self_referential_chassis_evidence(row):
+            continue
+        if is_circular_for_link(row, left_ref=left_ref, right_ref=right_ref):
+            continue
+        keep.append(row)
+    return keep
+
+
+def has_circularity(
+    evidence_rows: Iterable[Any],
+    *,
+    left_ref: str,
+    right_ref: str,
+) -> bool:
+    for row in evidence_rows:
+        if is_self_referential_chassis_evidence(row):
+            return True
+        if is_circular_for_link(row, left_ref=left_ref, right_ref=right_ref):
+            return True
+    return False
+
+
+__all__ = [
+    "filter_non_circular",
+    "has_circularity",
+    "is_circular_for_link",
+    "is_self_referential_chassis_evidence",
+]

+"""Evidence levels (E0..E5) and decision map (D0..D3) — OBS-IDENTITY-EVIDENCE W1.5.
+
+Design axioms (Fase 3):
+
+- Level is a **function of the tuple** ``(fact_type, acquisition_method,
+  source, subtype, quals)`` — never ``fact_type`` alone.
+- D4 (silent apply) MUST NOT EXIST in the code path.
+- Links are NOT transitive: ``A~B`` and ``B~C`` does NOT imply ``A~C``.
+- The default relation between two references is ``unresolved``.
+- MAC block adjacency is NEVER a predicate (see :mod:`mac_guards`).
+- LLDP chassis-id reaches E5-bridge ONLY when its subtype is
+  ``macAddress`` (802.1AB chassis-id-subtype 4). Any other subtype
+  degrades one level.
+- Bridge MAC observed only via FDB — never E5-bridge.
+- E3-topological requires ALL qualifiers
+  ``{access_port_measured, mac_count_measured (below threshold),
+  simultaneity_window}`` — otherwise it degrades to E2-temporal.
+- Locally-administered MACs (U/L bit = 1) are never usable as identity
+  evidence (see :mod:`mac_guards`).
+
+Evidence levels (ordered):
+
+    E0-unmeasured  — poll not performed, timeout, sensor missing
+    E0-absent      — measurement performed; the fact was absent
+    E1-corroborant — weak signals (OUI/vendor/temporal hint)
+    E2-temporal    — temporal co-presence or degraded topology
+    E3-topological — verified port co-observation with qualifiers
+    E4-mgmt        — same-agent management-plane consistency
+    E5-bridge      — bridge-layer identity (LLDP chassis-id=macAddress,
+                     declared bridge base MAC, LACP system-ID marker)
+    E5-physical    — physical identifier (ENTITY serial, UUID, chassis

+
+    E5-physical, non-circular, no-contradiction -> D3

+    E1                                          -> never raises (D0)

+def classify_evidence_level(
+    *,
+    fact_type: str,
+    acquisition_method: str,
+    source: str,
+    subtype: str = "",
+    quals: EvidenceQualifiers | None = None,
+) -> str:
+    """Return the E-level for one evidence row.
+
+    NOTE: this is a **pure** function. Callers must pass the collector
+    qualifiers explicitly; no heuristic is applied on missing quals.
+    """
+    q = quals or EvidenceQualifiers()
+
+    if not q.measured:
+        return E0_UNMEASURED
+    if q.fact_absent:
+        return E0_ABSENT
+
+    # A locally-administered MAC value cannot be used as bridge / physical
+    # identity — degrade to corroborant regardless of fact type.
+    if q.value_is_locally_administered_mac and fact_type in (
+        _BRIDGE_FACTS | _PHYSICAL_FACTS
+    ):
+        return E1_CORROBORANT
+
+    ft = (fact_type or "").strip()
+    am = (acquisition_method or "").strip()
+    st = (subtype or "").strip()
+
+    if ft in _PHYSICAL_FACTS:
+        if am in _PHYSICAL_METHODS:
+            return E5_PHYSICAL
+        return E1_CORROBORANT
+
+    if ft in _BRIDGE_FACTS:
+        # Bridge MAC learned only through FDB is never E5.
+        if q.from_fdb_observation_only:
+            return _degrade_bridge_to_temporal(ft)
+        if ft == "lldp_chassis_id" and st not in _LLDP_SUBTYPES_STRONG:
+            return _degrade_bridge_to_temporal(ft)
+        if am not in _BRIDGE_METHODS:
+            return _degrade_bridge_to_temporal(ft)
+        return E5_BRIDGE
+
+    if ft in _MGMT_FACTS:
+        return E4_MGMT
+
+    if ft in _TOPOLOGY_FACTS:
+        # E3 requires the full topology qualifier bundle.
+        if (
+            q.access_port_measured
+            and q.mac_count_measured
+            and q.mac_count > 0
+            and q.mac_count <= q.mac_count_threshold
+            and q.simultaneity_window
+        ):
+            return E3_TOPOLOGICAL
+        return E2_TEMPORAL
+
+    if ft in _TEMPORAL_FACTS:
+        return E2_TEMPORAL
+
+    if ft in _CORROBORANT_FACTS:
+        return E1_CORROBORANT
+
+    # Unknown fact_type must not silently raise the decision surface.
+    return E1_CORROBORANT
+

+DECAY_TTL_HOURS: Mapping[str, int] = {

+@dataclass(frozen=True)
+class DecisionInputs:
+    """Inputs to :func:`max_decision` — everything the linker knows about
+    a candidate link *besides* the evidence set itself."""
+
+    circular: bool = False
+    contradicts: bool = False
+    is_physical_bridge_context: bool = False
+    """When True, E5-bridge propagates fuse-physical semantics and must be
+    demoted (never fuse physical members based on bridge MAC alone)."""
+
+
+def max_decision(
+    levels: Iterable[str],
+    *,
+    circular: bool = False,
+    contradicts: bool = False,
+    is_physical_bridge_context: bool = False,
+) -> str:
+    """Highest permissible decision given the evidence levels + guards.
+
+    Never returns anything above D3 — the D4 (silent apply) path does not
+    exist by design.
+    """
+    if circular or contradicts:
+        return D0_NOTHING
+
+    present = {lvl for lvl in levels if lvl in _LEVEL_ORDER}
+    if not present:
+        return D0_NOTHING
+
+    if E5_PHYSICAL in present:
+        return D3_CONSOLIDABLE
+
+    if E5_BRIDGE in present:
+        # Bridge identity groups virtual-chassis — it must NEVER fuse two
+        # physical members. If the caller declares physical-member context,
+        # demote to proposal-only.
+        return D2_PROPOSAL if is_physical_bridge_context else D3_CONSOLIDABLE
+
+    if E4_MGMT in present and E3_TOPOLOGICAL in present:
+        return D2_PROPOSAL
+
+    if E3_TOPOLOGICAL in present or E2_TEMPORAL in present:
+        return D1_INTERNAL
+
+    # E1 alone never raises above internal-noise.
+    return D0_NOTHING
+
+
+def can_propose(
+    levels: Iterable[str],
+    *,
+    circular: bool = False,
+    contradicts: bool = False,
+    is_physical_bridge_context: bool = False,
+) -> bool:
+    """True when :func:`max_decision` reaches D2 or D3."""
+    d = max_decision(
+        levels,
+        circular=circular,
+        contradicts=contradicts,
+        is_physical_bridge_context=is_physical_bridge_context,
+    )
+    return d in (D2_PROPOSAL, D3_CONSOLIDABLE)
+
+
+def is_consolidable(decision_level: str) -> bool:
+    """D3 alone is human-consolidable; D2 stops at proposal."""
+    return decision_level == D3_CONSOLIDABLE
+

+# --- Retained utilities used by store/linker --------------------------
+
+# NOTE: no per-class weight table any more; scoring is level-driven and
+# deliberately capped so that no single E-level except E5 can propose.
+LEVEL_WEIGHTS: Mapping[str, float] = {
+    E5_PHYSICAL: 1.0,
+    E5_BRIDGE: 0.9,
+    E4_MGMT: 0.6,
+    E3_TOPOLOGICAL: 0.4,
+    E2_TEMPORAL: 0.2,
+    E1_CORROBORANT: 0.1,
+    E0_ABSENT: 0.0,
+    E0_UNMEASURED: 0.0,
+}
+
+
+def score_levels(levels: Iterable[str], *, confidences: Iterable[float] | None = None) -> float:
+    """Weighted sum capped at 1.0 — informational, never a name-ranking score."""
+    lvls = list(levels)
+    confs = list(confidences) if confidences is not None else [1.0] * len(lvls)
+    if len(confs) != len(lvls):
+        confs = (confs + [1.0] * len(lvls))[: len(lvls)]
+    seen: set[str] = set()
+    total = 0.0
+    for lvl, cf in zip(lvls, confs):
+        if lvl in seen:
+            continue
+        seen.add(lvl)
+        w = LEVEL_WEIGHTS.get(lvl, 0.0)
+        total += w * max(0.0, min(1.0, float(cf)))
+    return min(1.0, total)
+

+These are the ONLY functions allowed to move
+:attr:`IdentityLinkProposal.link_state` off ``proposed``. Every call
+appends a JSON entry to :attr:`IdentityLinkProposal.audit`.

+
+- Only D3 proposals can be confirmed (``is_consolidable`` gate).

+  ``confirmed`` proposal returns the row unchanged, but does NOT append
+  a duplicate audit entry for the same actor.
+- ``retract_link`` reverses a confirmation (split) — the row's
+  ``link_state`` becomes ``retracted``, ``member_id`` on the associated
+  evidence remains intact for reattribution.
+- No function mutates ``Asset.chassis_id`` or ``Asset.name``.

+

+
+from datetime import datetime
+from typing import Any
+
+from sqlalchemy.orm import Session

+    D3_CONSOLIDABLE,

+def _append_audit(
+    proposal: IdentityLinkProposal,
+    *,
+    action: str,
+    actor: str,
+    reason: str = "",
+    extra: dict[str, Any] | None = None,
+) -> None:
+    entries = list(proposal.audit or [])
+    entry = {
+        "action": action,
+        "actor": actor,
+        "ts": utcnow().isoformat(),
+    }
+    if reason:
+        entry["reason"] = reason
+    if extra:
+        entry.update(extra)
+    entries.append(entry)
+    proposal.audit = entries

+    proposal: IdentityLinkProposal, *, action: str, actor: str

+) -> IdentityLinkProposal:

+    Returns the up-to-date proposal row. Idempotent: repeated calls by

+    """
+    proposal = db.get(IdentityLinkProposal, proposal_id)
+    if proposal is None:
+        raise DecisionError(f"proposal {proposal_id} not found")

+        # new reason string that is not already in the trail.
+        if not _already_recorded(proposal, action="confirm", actor=actor):

+                action="confirm",
+                actor=actor,
+                reason=reason,
+                extra={"noop": True},

+            db.flush()
+        return proposal
+
+    if proposal.link_state in {"rejected", "retracted"}:
+        raise DecisionError(

+

+        raise DecisionError(

+        raise DecisionError(
+            f"relation must be proven_same|proven_different, got '{relation}'"
+        )
+

+    proposal.link_state = "confirmed"
+    proposal.relation = relation
+    proposal.confirmed_by = actor
+    proposal.confirmed_at = now
+    proposal.updated_at = now

+"""Link scoring and proposal emission.
+
+Rules enforced here:
+
+- Proposals ONLY: ``link_state`` starts at ``proposed`` and NEVER moves
+  to ``confirmed`` from this module. Confirmation is the sole business
+  of :mod:`decisions`.
+- ``relation`` defaults to ``unresolved``. A brand-new proposal is not
+  ``proven_same``.
+- Links are NOT transitive: the linker processes each pair
+  independently; no union-find, no chained equivalence.
+- ``chassis_id`` is NEVER mutated. There is no code path that touches
+  ``Asset.name`` or ``Asset.chassis_id`` here.
+- Evidence rows failing K7 (self-referential chassis id, provenance
+  citing the link) are dropped before scoring.
+- E5-bridge on a physical-member context degrades to D2 so we never
+  propose to *fuse* physical members based on a bridge MAC.
+"""

+from dataclasses import dataclass, field
+from typing import Iterable

+from sqlalchemy import and_, select
+from sqlalchemy.orm import Session

+    right_ref: str

+    is_physical_bridge_context: bool = False

+    if ":" not in ref:

+    """Assign a row to ``"left" | "right" | "other"`` based on subject/member."""
+    left_type, left_id = _parse_ref(left_ref)
+    right_type, right_id = _parse_ref(right_ref)

+        if side == "other":

+    for key, sides in groups.items():
+        left_vals = set(sides.get("left", {}).keys())
+        right_vals = set(sides.get("right", {}).keys())
+        if left_vals and right_vals and left_vals.isdisjoint(right_vals):
+            for side_map in sides.values():

+            .limit(1)
+        ).first()
+
+        evidence_ids = [r.id for r in usable]
+

+            existing.score = score
+            existing.evidence_ids = evidence_ids
+            existing.decision_level = decision
+            # relation stays unresolved until a human confirms.
+            db.flush()

+            right_ref=right,

+            link_state="proposed",

+                    "action": "propose",
+                    "actor": actor,
+                    "decision_level": decision,
+                    "score": score,
+                    "levels": levels,
+                }
+            ],
+        )
+        db.add(proposal)
+        db.flush()
+        proposals.append(proposal)
+    return proposals

+"""MAC guards for identity evidence — U/L exclusion.
+
+Explicitly does NOT export any MAC-block-adjacency helper. The identity
+pipeline must never use "MAC N ± k" style predicates. A CI test in
+``tests/test_identity_evidence.py`` scans this module (and the whole
+package) to make sure such a helper does not sneak back in.
+"""
+
+from __future__ import annotations
+
+import re
+
+
+_MAC_HEX_RE = re.compile(r"[0-9A-Fa-f]")
+
+
+def _normalize(mac: str) -> str:
+    hexed = "".join(_MAC_HEX_RE.findall(mac or ""))
+    if len(hexed) != 12:
+        return ""
+    return ":".join(hexed[i : i + 2].upper() for i in range(0, 12, 2))
+
+
+def is_locally_administered_mac(mac: str) -> bool:
+    """True when the U/L bit (LSB of first octet, mask 0x02) is set.
+
+    Locally-administered MACs (LAA) — random Wi-Fi, VM clones, VRRP/HSRP
+    virtuals — must never be treated as physical-identity evidence.
+    """
+    norm = _normalize(mac)
+    if not norm:
+        return False
+    try:
+        first = int(norm.split(":")[0], 16)
+    except ValueError:
+        return False
+    return bool(first & 0x02)
+
+
+def guard_mac_value(mac: str) -> tuple[bool, str]:
+    """Return ``(acceptable, reason)`` for a MAC used as identity value.
+
+    - ``(False, "laa")`` when U/L bit is set.
+    - ``(False, "malformed")`` when the value is not a valid 48-bit MAC.
+    - ``(True, "ok")`` otherwise.
+    """
+    norm = _normalize(mac)
+    if not norm:
+        return False, "malformed"
+    if is_locally_administered_mac(norm):
+        return False, "laa"
+    return True, "ok"
+
+
+__all__ = ["guard_mac_value", "is_locally_administered_mac"]

+"""MAC presence classification — orthogonal to identity linking.
+
+Classes:
+
+- ``direct`` — ARP/ND REACHABLE or ICMP toward a bound IP.
+- ``fdb_fresh`` — MAC in ``switch_ports.observed_macs`` with recent
+  ``last_fdb_at`` and ``fdb_poll.ok=True``.
+- ``historical_only`` — past FDB/Fritz/presence timestamps without a live
+  signal.
+- ``absent_measured`` — the FDB/ARP/ICMP probe ran and returned nothing.
+- ``absent_unmeasured`` — no probe was performed / all probes failed
+  (SNMP timeout, sensor down). This is NOT proof of absence and must
+  never raise "gone" cards.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass
+from datetime import datetime, timedelta
+from typing import Literal
+
+
+PresenceClass = Literal[
+    "direct",
+    "fdb_fresh",
+    "historical_only",
+    "absent_measured",
+    "absent_unmeasured",
+]
+
+DEFAULT_FDB_FRESH_MINUTES = 30
+
+
+@dataclass(frozen=True)
+class PresenceInputs:
+    """Inputs for :func:`classify_mac_presence`.
+
+    The two ``arp_probe_ok`` / ``fdb_poll_ok`` flags encode measurement
+    liveness — they are required to distinguish ``absent_measured`` from
+    ``absent_unmeasured``.
+    """
+
+    # Live signals.
+    arp_reachable: bool = False
+    icmp_reachable: bool = False
+    fdb_hit: bool = False
+    last_fdb_at: datetime | None = None
+    # Measurement liveness (did the probe actually run?).
+    arp_probe_ok: bool = False
+    fdb_poll_ok: bool = False
+    # Historical corroborants (do NOT lift ``absent`` to a positive class).
+    historical_fdb_at: datetime | None = None
+    historical_presence_at: datetime | None = None
+    fritz_last_at: datetime | None = None
+    # Tuning knobs.
+    now: datetime | None = None
+    fdb_fresh_minutes: int = DEFAULT_FDB_FRESH_MINUTES
+    historical_only: bool = False
+    """When True, callers know the DB has past FDB/Fritz rows but there is
+    no live signal. Preserved for parity with the review's T18 fixture."""
+
+
+def classify_mac_presence(inputs: PresenceInputs) -> PresenceClass:
+    """Return the presence class for a MAC given probe outputs."""
+    if inputs.arp_reachable or inputs.icmp_reachable:
+        return "direct"
+
+    now = inputs.now or datetime.utcnow()
+    fresh_cutoff = now - timedelta(minutes=inputs.fdb_fresh_minutes)
+
+    if (
+        inputs.fdb_hit
+        and inputs.fdb_poll_ok
+        and inputs.last_fdb_at
+        and inputs.last_fdb_at >= fresh_cutoff
+    ):
+        return "fdb_fresh"
+
+    has_history = inputs.historical_only or any(
+        ts is not None
+        for ts in (
+            inputs.historical_fdb_at,
+            inputs.historical_presence_at,
+            inputs.fritz_last_at,
+        )
+    )
+    stale_fdb = (
+        inputs.fdb_hit
+        and inputs.last_fdb_at is not None
+        and inputs.last_fdb_at < fresh_cutoff
+    )
+    if has_history or stale_fdb:
+        return "historical_only"
+
+    # No live signal AND no history. Distinguish measured absence from
+    # unmeasured absence — this is the whole point of the two flags.
+    any_probe_ran = inputs.arp_probe_ok or inputs.fdb_poll_ok
+    if any_probe_ran:
+        return "absent_measured"
+    return "absent_unmeasured"
+
+
+__all__ = [
+    "DEFAULT_FDB_FRESH_MINUTES",
+    "PresenceClass",
+    "PresenceInputs",
+    "classify_mac_presence",
+]

+class IdentityLinkProposal(Base):
+    """Candidate identity links — audited human-confirm only; never auto-fuses.
+
+    Two orthogonal FSMs:
+
+    - link_state: ``proposed | confirmed | rejected | retracted`` — decisions
+      recorded with actor + timestamp + reason in :attr:`audit`.
+    - relation:   ``proven_same | proven_different | unresolved`` (default
+      unresolved); a proposal MUST NOT default to ``proven_same``.
+
+    Design constraints enforced here / in the linker / in decisions:
+
+    - No D4: the schema has no code path that mutates ``Asset.chassis_id`` or
+      ``Asset.name`` on behalf of a proposal.
+    - Links are NOT transitive.
+    - ``retract_link`` / split preserve rows; ``member_id`` on evidence rows
+      allows reattribution.
+    """
+
+    __tablename__ = "identity_link_proposals"
+    __table_args__ = (
+        Index("ix_identity_link_proposals_link_state", "link_state"),
+        Index("ix_identity_link_proposals_relation", "relation"),
+        Index("ix_identity_link_proposals_refs", "left_ref", "right_ref"),
+    )
+
+    id: Mapped[int] = mapped_column(Integer, primary_key=True)
+    left_ref: Mapped[str] = mapped_column(String(64), index=True)
+    right_ref: Mapped[str] = mapped_column(String(64), index=True)
+    # unresolved | proven_same | proven_different
+    relation: Mapped[str] = mapped_column(String(24), default="unresolved")
+    # proposed | confirmed | rejected | retracted
+    link_state: Mapped[str] = mapped_column(String(16), default="proposed")
+    # D2 (proposal only) | D3 (proposal + consolidable on human confirm)
+    decision_level: Mapped[str] = mapped_column(String(4), default="D2")
+    score: Mapped[float] = mapped_column(Float, default=0.0)
+    evidence_ids: Mapped[list] = mapped_column(JSON, default=list)
+    proposed_by: Mapped[str] = mapped_column(String(64), default="linker")
+    proposed_at: Mapped[datetime] = mapped_column(DateTime, default=utcnow)
+    confirmed_by: Mapped[Optional[str]] = mapped_column(String(64), nullable=True)

+#!/usr/bin/env python3
+"""Read-only MAC presence report — DB + optional ip neigh file; no SNMP blast."""
+
+from __future__ import annotations
+
+import argparse
+import json
+import os
+import sqlite3
+import sys
+from datetime import datetime, timedelta
+from pathlib import Path
+
+# Allow import from api when run from observatory/
+_API = Path(__file__).resolve().parents[1] / "api"
+if str(_API) not in sys.path:
+    sys.path.insert(0, str(_API))

+from app.services.macutil import normalize_mac

+def _parse_neigh(path: Path) -> dict[str, bool]:
+    """Parse `ip neigh` dump: MAC → REACHABLE."""
+    reachable: dict[str, bool] = {}
+    if not path.is_file():
+        return reachable
+    for line in path.read_text(encoding="utf-8", errors="replace").splitlines():
+        parts = line.split()
+        if len(parts) < 5:
+            continue
+        state = parts[-1].upper()
+        for token in parts:
+            if ":" in token and len(token.replace(":", "")) == 12:
+                mac = normalize_mac(token)
+                if state == "REACHABLE":
+                    reachable[mac] = True
+                elif mac not in reachable:
+                    reachable[mac] = False
+    return reachable
+
+
+def _mac_in_fdb(conn: sqlite3.Connection, mac: str) -> tuple[bool, datetime | None, bool]:
+    """Return (hit, last_fdb_at, poll_ok-ish) from switch_ports observed_macs."""
+    norm = normalize_mac(mac)
+    last_at: datetime | None = None
+    hit = False
+    for row in conn.execute(
+        "SELECT observed_macs, last_fdb_at FROM switch_ports WHERE observed_macs IS NOT NULL"
+    ):
+        macs_raw = row[0]
+        try:
+            macs = json.loads(macs_raw) if isinstance(macs_raw, str) else (macs_raw or [])
+        except json.JSONDecodeError:
+            macs = []
+        for m in macs:
+            if normalize_mac(str(m)) == norm:
+                hit = True
+                if row[1]:
+                    ts = datetime.fromisoformat(str(row[1]).replace("Z", "+00:00").replace("+00:00", ""))
+                    if last_at is None or ts > last_at:
+                        last_at = ts
+    # poll_ok: we don't SNMP-walk; treat recent last_fdb_at as implicit ok
+    poll_ok = hit and last_at is not None and last_at >= datetime.utcnow() - timedelta(minutes=30)
+    return hit, last_at, poll_ok
+
+
+def _asset_context(conn: sqlite3.Connection, mac: str) -> dict:
+    norm = normalize_mac(mac)
+    row = conn.execute(
+        """
+        SELECT a.id, a.name, a.chassis_id, a.trust_level, a.presence_sources,
+               a.fritz_last_seen, i.id AS iface_id
+        FROM interfaces i
+        JOIN assets a ON a.id = i.asset_id
+        WHERE i.mac = ?
+        LIMIT 1
+        """,
+        (norm,),
+    ).fetchone()
+    if row is None:
+        return {"asset_id": None, "mac": norm}
+    presence_sources = {}
+    try:
+        presence_sources = json.loads(row[4]) if row[4] else {}
+    except (json.JSONDecodeError, TypeError):
+        pass
+    fdb_hist = None
+    fdb_src = presence_sources.get("fdb") if isinstance(presence_sources, dict) else None
+    if isinstance(fdb_src, dict) and fdb_src.get("last"):
+        try:
+            fdb_hist = datetime.fromisoformat(str(fdb_src["last"]).replace("Z", ""))
+        except ValueError:
+            pass
+    fritz_last = None
+    if row[5]:
+        try:
+            fritz_last = datetime.fromisoformat(str(row[5]).replace("Z", ""))
+        except ValueError:
+            pass
+    return {
+        "asset_id": row[0],
+        "name": row[1],
+        "chassis_id": row[2],
+        "trust_level": row[3],
+        "mac": norm,
+        "presence_sources": presence_sources,
+        "fdb_hist": fdb_hist,
+        "fritz_last": fritz_last,
+    }
+
+
+def probe_macs(
+    db_path: Path,
+    macs: list[str],
+    *,
+    neigh_path: Path | None = None,
+) -> list[dict]:
+    conn = sqlite3.connect(str(db_path))
+    conn.row_factory = sqlite3.Row
+    neigh = _parse_neigh(neigh_path) if neigh_path else {}
+    now = datetime.utcnow()
+    reports: list[dict] = []
+
+    for mac in macs:
+        norm = normalize_mac(mac)
+        ctx = _asset_context(conn, norm)
+        fdb_hit, last_fdb_at, fdb_poll_ok = _mac_in_fdb(conn, norm)
+        arp_probe_ok = norm in neigh
+        presence = classify_mac_presence(
+            PresenceInputs(
+                arp_reachable=neigh.get(norm, False),
+                icmp_reachable=False,
+                fdb_hit=fdb_hit,
+                fdb_poll_ok=fdb_poll_ok,
+                arp_probe_ok=arp_probe_ok,
+                last_fdb_at=last_fdb_at,
+                historical_fdb_at=ctx.get("fdb_hist"),
+                historical_presence_at=ctx.get("fdb_hist"),
+                fritz_last_at=ctx.get("fritz_last"),
+                now=now,
+            )
+        )
+        reports.append(
+            {
+                "mac": norm,
+                "presence_class": presence,
+                "asset_id": ctx.get("asset_id"),
+                "chassis_id": ctx.get("chassis_id"),
+                "trust_level": ctx.get("trust_level"),
+                "fdb_hit": fdb_hit,
+                "fdb_last_at": last_fdb_at.isoformat() if last_fdb_at else None,
+                "neigh_reachable": neigh.get(norm),
+            }
+        )
+    conn.close()
+    return reports
+

+def main() -> int:
+    parser = argparse.ArgumentParser(description="Read-only MAC presence probe (no SNMP)")
+    parser.add_argument(
+        "macs",
+        nargs="+",
+        help="MAC addresses to probe",
+    )
+    parser.add_argument(
+        "--db",
+        default=os.environ.get("SQLITE_PATH", "/data/db/observatory.db"),
+        help="SQLite database path",
+    )
+    parser.add_argument(
+        "--neigh",
+        type=Path,
+        default=None,
+        help="Optional ip neigh dump file",
+    )
+    parser.add_argument(
+        "--json",
+        action="store_true",
+        help="Emit JSON",
+    )
+    args = parser.parse_args()
+    reports = probe_macs(Path(args.db), args.macs, neigh_path=args.neigh)
+    if args.json:
+        print(json.dumps(reports, indent=2, default=str))
+    else:
+        for r in reports:
+            print(
+                f"{r['mac']}: {r['presence_class']} "
+                f"(asset={r['asset_id']} chassis={r['chassis_id']} fdb={r['fdb_hit']})"
+            )
+    return 0
