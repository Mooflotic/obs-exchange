+#!/usr/bin/env python3
+"""O13D C6 — gate hard-coded color literals in web/src.
+
+- Token file ``assets/matrix.css``: hex/rgba OK (CSS variables source of truth).
+- ``var(--token, #fallback)`` fallbacks are ignored (not new palette).
+- ``*.test.js`` ignored (assert token values).
+- Other literals FAIL unless the file is in ALLOWLIST with a reason.
+- O13D fixed worst non-semantic greens (Incidents glow, Plant .dot.up / #83c99c);
+  remaining canvas/AI/topology literals deferred (no new palette tokens invented).

+Exit 0 = pass; exit 1 = fail.

+
+import argparse
+import re
+import sys
+import tempfile
+from pathlib import Path
+
+ROOT = Path(__file__).resolve().parents[1]
+WEB_SRC = ROOT / "web" / "src"
+
+TOKEN_FILES = {
+    "assets/matrix.css",

+# relative path -> reason
+ALLOWLIST: dict[str, str] = {

+    "views/Suggestions.vue": "AI purple — deferred",

+    "views/Topology.vue": "topology canvas-heavy — deferred O13D",

+HEX_RE = re.compile(r"#(?:[0-9a-fA-F]{3,4}|[0-9a-fA-F]{6}|[0-9a-fA-F]{8})\b")
+RGB_RE = re.compile(r"\brgba?\(\s*[^)]+\)")
+VAR_RE = re.compile(r"var\([^)]*\)")
+
+SCAN_SUFFIXES = {".vue", ".css", ".js", ".ts", ".scss"}
+
+
+def _strip_var_fallbacks(text: str) -> str:
+    return VAR_RE.sub("var()", text)
+
+
+def iter_files():
+    for p in WEB_SRC.rglob("*"):
+        if not p.is_file():
+            continue
+        if p.suffix not in SCAN_SUFFIXES:
+            continue
+        if ".test." in p.name:
+            continue
+        yield p
+
+
+def scan_file(path: Path) -> list[tuple[int, str, str]]:
+    rel = path.relative_to(WEB_SRC).as_posix()
+    if rel in TOKEN_FILES or rel in ALLOWLIST:
+        return []
+    try:
+        text = path.read_text(encoding="utf-8")
+    except UnicodeDecodeError:
+        return []
+    stripped = _strip_var_fallbacks(text)
+    hits: list[tuple[int, str, str]] = []
+    # Map hits back via original lines approximately
+    orig_lines = text.splitlines()
+    strip_lines = stripped.splitlines()
+    for i, (oline, sline) in enumerate(zip(orig_lines, strip_lines), 1):
+        st = oline.strip()
+        if st.startswith("//") or st.startswith("*") or st.startswith("/*"):
+            continue
+        for m in HEX_RE.finditer(sline):
+            hits.append((i, m.group(0), oline.strip()[:120]))
+        for m in RGB_RE.finditer(sline):
+            hits.append((i, m.group(0), oline.strip()[:120]))
+    return hits
+
+
+def self_test_inject() -> int:
+    """Prove gate fails on injected literal and passes after remove."""
+    tmp = WEB_SRC / "_o13d_color_gate_inject_tmp.vue"
+    try:
+        tmp.write_text(
+            "<style>\n.x { color: #ff00aa; /* O13D_COLOR_GATE_INJECT */ }\n</style>\n",
+            encoding="utf-8",
+        )
+        hits = scan_file(tmp)
+        if not hits:
+            print("SELFTEST FAIL: inject not detected")
+            return 1
+        print(f"SELFTEST inject detected: {hits[0]}")
+    finally:
+        if tmp.exists():
+            tmp.unlink()
+    # After remove — ensure no hits from tmp
+    if tmp.exists():
+        print("SELFTEST FAIL: tmp not removed")
+        return 1
+    print("SELFTEST PASS: inject fails, remove passes")
+    return 0
+
+
+def main() -> int:
+    ap = argparse.ArgumentParser()
+    ap.add_argument("--self-test", action="store_true", help="inject/remove proof")
+    args = ap.parse_args()
+
+    if args.self_test:
+        return self_test_inject()
+
+    all_hits: list[tuple[str, int, str, str]] = []
+    for p in iter_files():
+        for line_no, lit, ctx in scan_file(p):
+            all_hits.append((p.relative_to(WEB_SRC).as_posix(), line_no, lit, ctx))
+
+    if all_hits:
+        print(f"FAIL: {len(all_hits)} hard-coded color literal(s) outside token/allowlist")
+        for rel, ln, lit, ctx in all_hits[:60]:
+            print(f"  {rel}:{ln}: {lit}  |  {ctx}")
+        if len(all_hits) > 60:
+            print(f"  ... +{len(all_hits) - 60} more")
+        return 1
+
+    print("PASS: no hard-coded color literals outside token/allowlist files")
+    print(f"  token_files={sorted(TOKEN_FILES)}")
+    print(f"  allowlist_count={len(ALLOWLIST)}")
+    return 0
+
+
+if __name__ == "__main__":
+    sys.exit(main())
