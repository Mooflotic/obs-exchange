+  }
+  if ((asset?.name || "").trim()) {
+    return { source: null, confidence: null, kind: "unknown" };
+  }
+  return { source: null, confidence: null, kind: "unknown" };
+}
+
+function topPendingOnAsset(asset) {
+  const proposals = (asset?.proposals || asset?.name_proposals || []).filter(
+    (p) => (p.status || "pending") === "pending" && p.value,
+  );
+  if (!proposals.length) return null;
+  return [...proposals].sort((x, y) => {
+    const rankDiff = proposalSourceRank(y.source) - proposalSourceRank(x.source);
+    if (rankDiff !== 0) return rankDiff;
+    return Number(y.confidence || 0) - Number(x.confidence || 0);
+  })[0];
+}
+
+function normalizeConfidence(raw) {
+  let c = Number(raw || 0);
+  if (!Number.isFinite(c)) return 0;
+  if (c > 1) c /= 100;
+  return c;
+}

+/**
+ * Build one name card per multi-member chassis (≥2 visible assets).
+ *
+ * @param {Array<object>} assets
+ * @param {object} chassisPayload
+ * @returns {Array<object>}
+ */
+export function buildChassisNameCards(assets, chassisPayload = {}) {
+  const byId = new Map((assets || []).map((a) => [Number(a.id), a]));
+  const groups = chassisPayload?.chassis || chassisPayload?.groups || [];
+  const cards = [];
+
+  for (const ch of groups) {
+    const cid = Number(ch.id);
+    if (!Number.isFinite(cid)) continue;
+    const memberRefs = ch.members || [];
+    const members = [];
+    for (const m of memberRefs) {
+      const aid = Number(m?.asset_id ?? m?.id);
+      const asset = byId.get(aid);
+      if (!asset) continue;
+      if ((asset.status || "") === "ignorato") continue;
+      members.push(asset);
+    }
+    if (members.length < 2) continue;
+
+    const currency = ch.name_currency || {};
+    const factState = String(currency.state || "absent");
+    const factValue =
+      factState === "fact" ? String(currency.value || "").trim() || null : null;
+    const factSource = currency.source
+      ? String(currency.source).toLowerCase().trim()
+      : null;
+    const factConf =
+      currency.confidence != null ? normalizeConfidence(currency.confidence) : null;
+    const factAuth =
+      currency.authority != null ? Number(currency.authority) : null;
+    const factIsManual =
+      factSource === "manual" ||
+      (factAuth != null && factAuth >= MANUAL_RANK);
+
+    // Presentation: resolver fact → fact; member-held manual → legacy_manual (I1);
+    // else inference; else absent «nome non noto» (I2). Never empty string as fact.
+    let displayName = null;
+    let nameKind = "absent"; // fact | legacy_manual | inference | absent
+    let nameSource = null;
+    let nameConfidence = null;
+    // Manual held on member meta (pre-currency legacy) still counts as manual
+    // for I3 conflict — not only resolver fact_is_manual.
+    let heldManual = !!factIsManual;
+
+    if (factValue) {
+      // name_kind "fact" ONLY when name_currency.state === "fact".
+      displayName = factValue;
+      nameKind = "fact";
+      nameSource = factSource;
+      nameConfidence = factConf;
+    } else {
+      // Prefer canonical member, then any named member.
+      const ordered = [...members].sort((a, b) => {
+        const ra = a.chassis_role === "canonical" ? 0 : 1;
+        const rb = b.chassis_role === "canonical" ? 0 : 1;
+        if (ra !== rb) return ra - rb;
+        return Number(a.id) - Number(b.id);
+      });
+      const named = ordered.find((a) => String(a.name || "").trim());
+      if (named) {
+        const prov = memberNameProvenance(named);
+        displayName = String(named.name).trim();
+        if (prov.kind === "manual") {
+          // Confirmed on interface meta, not yet chassis fact in the store (I1).
+          nameKind = "legacy_manual";
+          nameSource = "manual";
+          heldManual = true;
+        } else {
+          nameKind = "inference";
+          nameSource = prov.source || named.guess_source || null;
+        }
+        nameConfidence = prov.confidence;
+      }
+    }
+
+    // Top pending proposal across ALL members (absorbed — no per-member card).
+    let topProp = null;
+    let topAsset = null;
+    for (const a of members) {
+      const p = topPendingOnAsset(a);
+      if (!p) continue;
+      if (
+        !topProp ||
+        proposalSourceRank(p.source) > proposalSourceRank(topProp.source) ||
+        (proposalSourceRank(p.source) === proposalSourceRank(topProp.source) &&
+          Number(p.confidence || 0) > Number(topProp.confidence || 0))
+      ) {
+        topProp = p;
+        topAsset = a;
+      }
+    }
+
+    const proposedName = topProp
+      ? String(topProp.value || "").trim() || null
+      : null;
+    const proposedSource = topProp
+      ? String(topProp.source || "").toLowerCase()
+      : null;
+    const proposedConf = topProp
+      ? normalizeConfidence(topProp.confidence)
+      : null;
+    const proposedRank = proposedSource
+      ? proposalSourceRank(proposedSource)
+      : 0;
+
+    // Conflict: manual name exists (resolver OR member-held) and a weaker
+    // (or any non-manual) proposal would otherwise overwrite — never silent (I3).
+    let conflict = false;
+    const manualDisplay =
+      heldManual && displayName ? displayName : factIsManual && factValue ? factValue : null;
+    if (
+      heldManual &&
+      manualDisplay &&
+      proposedName &&
+      normalizeName(proposedName) !== normalizeName(manualDisplay)
+    ) {
+      const propIsManual = proposedSource === "manual";
+      if (!propIsManual || proposedRank < MANUAL_RANK) {
+        conflict = true;
+      }
+    }
+
+    const memberRows = members.map((a) => ({
+      asset_id: a.id,
+      role: a.chassis_role || null,
+      name: String(a.name || "").trim() || null,
+      ips: a.ips || [],
+      macs: a.macs || [],
+      presence_state: a.presence_state ?? null,
+      guess: a.guess || null,
+      guess_source: a.guess_source || null,
+    }));
+
+    const memberGuessNotes = memberRows
+      .filter((m) => m.guess && String(m.guess_source || "").toLowerCase())
+      .map((m) => `#${m.asset_id} guess «${m.guess}» (${m.guess_source})`);
+
+
+    // Actions always executable (F-9 for high priority).
+    const actions = ["correggi"]; // always: chassis rename
+    if (proposedName && !conflict) {
+      const isNoise =
+        displayName != null && isNoiseProposal(displayName, proposedName);
+      if (!isNoise) actions.push("adotta_proposta");
+    }
+    if (conflict && proposedName) {
+      actions.push("tieni_manuale", "sostituisci_proposta");
+    }
+    if (nameKind === "inference" && displayName && !proposedName) {
+      actions.push("conferma_manuale"); // adopt displayed inference as manual fact
+    }
+    if (nameKind === "legacy_manual" && displayName && !conflict) {
+      // Consolidate interface-held manual into chassis fact (not a silent rename).
+      actions.push("conferma_manuale");
+    }
+    if (nameKind === "absent") {
+      // correggi already present = assegna nome
+    }
+
+    let priority = "media";
+    let verdict = "chassis_nome";
+    if (conflict) {
+      priority = "alta";
+      verdict = "chassis_conflitto";
+    } else if (nameKind === "absent") {
+      priority = "alta";
+      verdict = "chassis_nome_assente";
+    } else if (nameKind === "inference") {
+      priority = "alta";
+      verdict = "chassis_nome_inferenza";
+    } else if (nameKind === "legacy_manual") {
+      // Confirmed but not consolidated — media; always has correggi+conferma (F-9).
+      priority = "media";
+      verdict = "chassis_nome_legacy";
+    } else if (proposedName && actions.includes("adotta_proposta")) {
+      const moreSpecific =
+        scoreSpecificity(proposedName) > scoreSpecificity(displayName || "");
+      if (
+        moreSpecific &&
+        proposedConf >= ADOPT_CONFIDENCE_MIN &&
+        proposedSource !== "ai"
+      ) {
+        priority = "alta";
+        verdict = "chassis_upgrade";
+      }
+    }
+
+    // High priority MUST have an executable action (F-9).
+    if (priority === "alta" && !actions.length) {
+      priority = "media";
+    }
+
+    const primaryId =
+      members.find((m) => m.chassis_role === "canonical")?.id ?? members[0].id;
+
+    cards.push({
+      kind: "chassis_nome",
+      key: `chassis-name-${cid}`,
+      chassis_id: cid,
+      primary_asset_id: primaryId,
+      members: memberRows,
+      member_asset_ids: members.map((m) => m.id),
+      display_name: displayName, // null → «nome non noto»
+      name_kind: nameKind, // fact | legacy_manual | inference | absent
+      name_source: nameSource,
+      name_confidence: nameConfidence,
+      fact_is_manual: !!heldManual,
+      name_currency: currency,
+      proposed_name: proposedName,
+      proposed_source: proposedSource,
+      proposed_confidence: proposedConf,
+      proposal_id: topProp?.id ?? null,
+      proposal_asset_id: topAsset?.id ?? null,
+      proposal_evidence: topProp?.evidence || topProp?.status_reason || null,
+      conflict,
+      member_guess_notes: memberGuessNotes,
+      actions,
+      priority,
+      verdict,
+      recommended_action: conflict
+        ? "verifica"
+        : nameKind === "absent"
+          ? "assegna"
+          : nameKind === "inference" || nameKind === "legacy_manual"
+            ? "conferma"
+            : actions.includes("adotta_proposta")
+              ? "adotta"
+              : "correggi",
+    });
+  }
+
+  cards.sort((a, b) => {
+    const rank = { alta: 0, media: 1, bassa: 2 };
+    const d = (rank[a.priority] ?? 9) - (rank[b.priority] ?? 9);
+    if (d !== 0) return d;
+    return a.chassis_id - b.chassis_id;
+  });
+  return cards;
+}
+
+/**
+ * Asset ids absorbed into a real chassis name card (≥2 visible non-ignorato).
+ * Empty set when no such card — do NOT fall back to raw chassisMemberIds (D2/T2).
+ *
+ * @param {Array<object>} assets
+ * @param {object} chassisPayload
+ * @returns {Set<number>}
+ */
+export function absorbedChassisAssetIds(assets, chassisPayload = {}) {
+  const cards = buildChassisNameCards(assets, chassisPayload);
+  const out = new Set();
+  for (const c of cards) {
+    for (const id of c.member_asset_ids || []) out.add(Number(id));
+  }
+  return out;
+}
