+/** Parse a timestamp treating naive (no tz) values as UTC — backend stores utcnow(). */
+function _utcMs(iso) {
+  if (!iso) return null;
+  let s = String(iso);
+  if (!/[zZ]|[+-]\d\d:?\d\d$/.test(s)) s += "Z";
+  const ms = Date.parse(s);
+  return Number.isFinite(ms) ? ms : null;

+/**
+ * Per-port FDB coverage state (O2 / B1). Never confuses poll-attempt time with data age.
+ *
+ * @returns {{
+ *   state: 'misurata_fresca'|'misurata_vecchia'|'non_coperta',
+ *   reason: string,
+ *   last_fdb_at: string|null,
+ *   age_hours: number|null,
+ *   switch_id: number|null,
+ *   port_id: number|null,
+ *   port_number: number|null,
+ * }}
+ */
+export function fdbPortCoverageState(port, sw = {}, nowMs = Date.now(), staleHours = FDB_STALE_HOURS) {
+  const cap = sw.capabilities || {};
+  const poll = sw.fdb_poll || {};
+  const last = port?.last_fdb_at || null;
+  const ms = _utcMs(last);
+  const switchId = sw.id ?? sw.switch_id ?? null;
+  const portId = port?.id ?? null;
+  const portNumber = port?.number ?? port?.physical_port_number ?? port?.port ?? null;
+
+  if (cap.fdb_supported === false) {
+    return {
+      state: "non_coperta",
+      reason: "FDB non supportato dall'apparato",
+      last_fdb_at: last,
+      age_hours: ms == null ? null : Math.max(0, (nowMs - ms) / 3_600_000),
+      switch_id: switchId,
+      port_id: portId,
+      port_number: portNumber,
+    };
+  }
+
+  if (ms != null) {
+    const ageH = (nowMs - ms) / 3_600_000;
+    if (ageH <= staleHours) {
+      return {
+        state: "misurata_fresca",
+        reason: "",
+        last_fdb_at: last,
+        age_hours: ageH,
+        switch_id: switchId,
+        port_id: portId,
+        port_number: portNumber,
+      };
+    }
+    return {
+      state: "misurata_vecchia",
+      reason: `ultima mappatura oltre ${staleHours} h`,
+      last_fdb_at: last,
+      age_hours: ageH,
+      switch_id: switchId,
+      port_id: portId,
+      port_number: portNumber,
+    };
+  }
+
+  let reason = "mai misurata";
+  if (poll.ok === false) {
+    reason = "switch non interrogabile";
+  } else if (poll.supported === false) {
+    reason = "FDB non supportato dall'apparato";
+  }
+
+  return {

+    last_fdb_at: null,
+    age_hours: null,

+/**
+ * Full coverage report for Plant (B1). Enumerates ports by state; declares window.
+ *
+ * @returns {{
+ *   window_hours: number,
+ *   window_label: string,
+ *   switches: Array<object>,

+ * }}
+ * `by_state` lists port **ids** (never bare totals without ids).
+ */
+export function fdbCoverageReport(switches = [], nowMs = Date.now(), staleHours = FDB_STALE_HOURS) {
+  const by_state = {
+    misurata_fresca: [],
+    misurata_vecchia: [],
+    non_coperta: [],
+  };
+  const outSwitches = [];
+  for (const sw of switches || []) {

+      ports.push(st);
+      if (st.port_id != null) by_state[st.state].push(Number(st.port_id));
+    }
+    const poll = sw.fdb_poll || {};
+    const cap = sw.capabilities || {};
+    outSwitches.push({
+      switch_id: sw.id,
+      name: sw.switch_display_name || sw.name,
+      fdb_supported: cap.fdb_supported !== false,
+      poll_ok: poll.ok === true,
+      poll_error: poll.ok === false ? String(poll.error || "") : "",
+      // Never treat failed poll.at as freshness of port data.
+      poll_attempt_at: poll.at || null,
+      ports,
+    });
+  }
+  for (const k of Object.keys(by_state)) {
+    by_state[k].sort((a, b) => a - b);
+  }
+  return {
+    window_hours: staleHours,
+    window_label: fdbFreshnessWindowLabel(staleHours),
+    switches: outSwitches,

+/**
+ * FDB coverage freshness across switches (F-13 / UX2.2.4).
+ *
+ * Presentation-only: it does NOT derive currency/state, it reports the newest
+ * measured `SwitchPort.last_fdb_at` and whether it is older than FDB_STALE_HOURS.
+ * The failed `fdb_poll.at` (a poll ATTEMPT time) is deliberately NOT used as
+ * freshness — that misrepresented ~57h-old port data as current.
+ *
+ * Returns null when there is nothing FDB-related to declare, else
+ * { available, stale, newestAt, hoursAgo }.
+ */
+export function fdbCoverageStatus(switches = [], nowMs = Date.now()) {
+  let newestMs = null;
+  let newestIso = "";
+  let anyFdbCapable = false;
+  let anyPollFailed = false;
+  for (const sw of switches || []) {
+    const cap = sw.capabilities || {};
+    if (cap.fdb_supported !== false) anyFdbCapable = true;
+    if (sw.fdb_poll && sw.fdb_poll.ok === false) anyPollFailed = true;
+    for (const port of sw.ports || []) {
+      const ms = _utcMs(port.last_fdb_at);
+      if (ms == null) continue;
+      if (newestMs == null || ms > newestMs) {
+        newestMs = ms;
+        newestIso = port.last_fdb_at;
+      }
+    }
+  }
+  if (newestMs == null) {
+    if (anyFdbCapable && anyPollFailed) {
+      return { available: true, stale: true, newestAt: "", hoursAgo: null };
+    }
+    return null;
+  }
+  const ageMs = nowMs - newestMs;
+  return {
+    available: true,
+    stale: ageMs > FDB_STALE_HOURS * 3_600_000,
+    newestAt: newestIso,
+    hoursAgo: Math.max(0, Math.floor(ageMs / 3_600_000)),
+  };
+}
