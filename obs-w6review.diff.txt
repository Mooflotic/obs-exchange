+"""W6-REVIEW read-only measurement (runs inside api container).
+
+Read-only: rolls back at the end, never commits. Answers:
+- pending NameProposal count by source (ipp/snmp presence)
+- deployed best_guess winners by source; assets won by a pending ipp/snmp proposal
+- FLIPS if ipp/snmp rank is lowered below dhcp (W6R.1 proposed fix ipp=snmp=81)
+- assets with empty best_guess (W6R.2.2), with raw name + presented name (I2)
+- adopted names (manual_overrides includes "name") for adopted_names_changed baseline
+"""
+
+from __future__ import annotations
+
+import sys
+from collections import Counter
+
+sys.path.insert(0, "/app")
+
+from sqlalchemy import create_engine, select
+from sqlalchemy.orm import sessionmaker
+
+from app.config import get_settings
+from app.facts.registry import AUTHORITY_RANK
+from app.models import Asset, NameProposal
+from app.services.identity import classify_oui
+from app.services.inventory import is_synthetic_fritz_name
+from app.services.name_proposal_chassis import presentation_name_for_asset
+from app.services.suggest import best_guess  # deployed (ipp=88 / snmp=87)
+
+# W6R.1 proposed fix: device self-declared management tier below dhcp(85),
+# above fritz(80). Only measured constraint is ipp>fritz (T-c). snmp unconstrained.
+LOWERED = dict(AUTHORITY_RANK)
+LOWERED["ipp"] = 81
+LOWERED["snmp"] = 81
+
+
+def bg_with_rank(asset: Asset, rankmap: dict[str, int]) -> tuple[str, str]:
+    mac = asset.interfaces[0].mac if asset.interfaces else ""
+    ranked: list[tuple[int, float, str, str]] = []
+    for p in asset.name_proposals:
+        if (p.status or "pending") != "pending":
+            continue
+        if not p.value:
+            continue
+        if p.source == "fritz" and is_synthetic_fritz_name(p.value, mac):
+            continue
+        ranked.append(
+            (rankmap.get((p.source or "").lower(), 20), float(p.confidence or 0.5), p.value, p.source)
+        )
+    if asset.vendor and not any(r[3] == "oui" for r in ranked):
+        name, _cat, conf = classify_oui(mac, asset.vendor)
+        if name:
+            ranked.append((rankmap.get("oui", 70), conf, name, "oui"))
+    ranked.sort(key=lambda x: (x[0], x[1]), reverse=True)
+    return (ranked[0][2], ranked[0][3]) if ranked else ("", "")
+
+
+def main() -> int:
+    settings = get_settings()
+    db = sessionmaker(bind=create_engine(f"sqlite:///{settings.sqlite_path}"))()
+    try:
+        assets = db.scalars(select(Asset)).all()
+
+        by_src: Counter = Counter()
+        for p in db.scalars(select(NameProposal).where(NameProposal.status == "pending")).all():
+            by_src[(p.source or "").lower()] += 1
+        print("PENDING_BY_SOURCE:", dict(sorted(by_src.items())))
+
+        winners: Counter = Counter()
+        ipp_snmp_winners = []
+        absent = []
+        flips = []
+        for a in assets:
+            now = tuple(best_guess(a)[:2])
+            low = bg_with_rank(a, LOWERED)
+            winners[now[1]] += 1
+            if now[1] in ("ipp", "snmp"):
+                ipp_snmp_winners.append((a.id, now))
+            if now[0] == "":
+                pres = presentation_name_for_asset(db, a)
+                absent.append((a.id, (a.name or ""), pres))
+            if now != low:
+                flips.append((a.id, now, low))
+
+        print("DEPLOYED_WINNERS_BY_SOURCE:", dict(sorted(winners.items())))
+        print("IPP_SNMP_WINNERS:", sorted(ipp_snmp_winners))
+        print(f"FLIPS deployed(ipp=88,snmp=87) vs lowered(ipp=snmp=81): {len(flips)}")
+        for f in sorted(flips):
+            print("  flip", f)
+
+        print(f"ABSENT_best_guess count: {len(absent)}")
+        for a_id, nm, pres in sorted(absent):
+            print(f"  absent id={a_id} raw_name={nm!r} presented={pres!r}")
+
+        adopted = []
+        for a in assets:
+            ov = set((a.meta or {}).get("manual_overrides") or [])
+            if "name" in ov or "nome" in ov:
+                adopted.append((a.id, a.name))
+        print(f"ADOPTED_NAMES count: {len(adopted)}")
+        for a_id, nm in sorted(adopted):
+            print(f"  adopted id={a_id} name={nm!r}")
+        return 0
+    finally:
+        db.rollback()
+        db.close()
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())

+"""Dump adopted names (manual_overrides includes "name") from a given sqlite path.
+
+Read-only. Usage (inside api container): python3 - <path> < adopted_dump.py
+Path is read from argv[1]; defaults to settings.sqlite_path.
+"""
+
+from __future__ import annotations
+
+import sys
+
+sys.path.insert(0, "/app")
+
+from sqlalchemy import create_engine, select
+from sqlalchemy.orm import sessionmaker
+
+from app.config import get_settings
+from app.models import Asset
+
+
+def main() -> int:
+    path = sys.argv[1] if len(sys.argv) > 1 else get_settings().sqlite_path
+    db = sessionmaker(bind=create_engine(f"sqlite:///{path}"))()
+    try:
+        rows = []
+        for a in db.scalars(select(Asset)).all():
+            ov = set((a.meta or {}).get("manual_overrides") or [])
+            if "name" in ov or "nome" in ov:
+                rows.append((a.id, a.name or ""))
+        print(f"# {path} adopted={len(rows)}")
+        for a_id, nm in sorted(rows):
+            print(f"{a_id}\t{nm}")
+        return 0
+    finally:
+        db.rollback()
+        db.close()
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())
