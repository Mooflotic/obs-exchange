+def _parse_ts(value: Any) -> datetime | None:
+    if value is None:
+        return None
+    if isinstance(value, datetime):
+        return value.replace(tzinfo=None) if value.tzinfo else value
+    if isinstance(value, (int, float)):
+        return datetime.utcfromtimestamp(float(value))
+    text = str(value).strip()
+    if not text:
+        return None
+    try:
+        return datetime.utcfromtimestamp(float(text))
+    except ValueError:
+        pass
+    try:
+        if text.endswith("Z"):
+            text = text[:-1] + "+00:00"
+        dt = datetime.fromisoformat(text)
+        if dt.tzinfo is not None:
+            from datetime import timezone
+
+            return dt.astimezone(timezone.utc).replace(tzinfo=None)
+        return dt
+    except ValueError:
+        return None

+def classify_scope(dst_ip: str) -> str:
+    """Return 'int' for RFC1918 / link-local / ULA, else 'ext'."""
+    text = (dst_ip or "").strip()
+    if not text:

+        return "ext"

+        if any(addr in net for net in _RFC1918):
+            return "int"
+        if addr in _IPV4_LINK_LOCAL:
+            return "int"
+        return "ext"
+    # IPv6: ULA + link-local only (not the broader is_private set).
+    if addr in _IPV6_ULA or addr in _IPV6_LINK_LOCAL:
+        return "int"
+    return "ext"

+def _subject_token(stype: str, sid: int | None, fallback_ip: str = "") -> str:
+    if stype and sid is not None:
+        return f"{stype}:{int(sid)}"
+    if fallback_ip:
+        digest = hashlib.sha256(fallback_ip.encode("utf-8")).hexdigest()[:12]
+        return f"u:{digest}"
+    return "u:unknown"

+def dedup_key_ext(
+    subject_type: str,
+    subject_id: int | None,
+    dst_ip: str,
+    dport: int,
+    proto: str,
+    *,
+    src_ip: str = "",
+) -> str:
+    subj = _subject_token(subject_type, subject_id, src_ip)
+    raw = f"ext|{subj}|{dst_ip}|{int(dport)}|{(proto or 'tcp').lower()}"
+    return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:64]

+def dedup_key_int(
+    subject_type: str,
+    subject_id: int | None,
+    peer_subject_type: str,
+    peer_subject_id: int | None,
+    *,
+    src_ip: str = "",
+    peer_ip: str = "",
+) -> str:
+    # int grain: NO port, NO proto — relation only.
+    a = _subject_token(subject_type, subject_id, src_ip)
+    b = _subject_token(peer_subject_type, peer_subject_id, peer_ip)
+    # Canonical order so A↔B and B↔A collapse when both resolved to subjects.
+    left, right = sorted([a, b])
+    raw = f"int|{left}|{right}"
+    return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:64]

+def ingest_egress_batch(
+    db: Session,
+    items: list[dict[str, Any]],
+    settings: Any | None = None,
+) -> dict[str, Any]:
+    """Ingest hybrid egress. Cap defers creates; breaker skips creates; refresh always OK.
+
+    Cap order when budget bites: ALL ext first (FIFO by first observation in this
+    cycle), then int. Deferred is not loss — the same tuple reappears next cycle
+    (starvation under cap is temporary, not dropped).
+    """
+    enabled = True
+    if settings is not None:
+        enabled = bool(getattr(settings, "egress_ingest_enabled", False))
+    if not enabled:
+        return {
+            "received": len(items),
+            "seen_distinct": 0,
+            "created": 0,
+            "refreshed": 0,
+            "deferred": 0,
+            "skipped_disabled": len(items),
+            "skipped_invalid": 0,
+            "skipped_breaker": 0,
+            "breaker_open": False,
+            "cap_active": False,
+            "baseline": None,
+            "table": table_stats(db) if items else {"rows_total": 0, "rows_today": 0, "approx_bytes": 0},
+            "cycle_seen": 0,
+            "cycle_created": 0,
+            "cycle_deferred": 0,
+            "o13c_marker": O13C_MARKER,
+        }
+
+    ensure_baseline_meta(db)
+
+    candidates: list[dict[str, Any]] = []
+    skipped_invalid = 0
+    for raw in items:
+        if not isinstance(raw, dict):
+            skipped_invalid += 1
+            continue
+        cand = prepare_candidate(db, raw)
+        if cand is None:
+            skipped_invalid += 1
+            continue
+        candidates.append(cand)
+
+    # Dedupe within cycle by dedup_key, keep earliest observed_at (FIFO).
+    by_key: dict[str, dict[str, Any]] = {}
+    for cand in candidates:
+        key = cand["dedup_key"]
+        prev = by_key.get(key)
+        if prev is None or cand["observed_at"] < prev["observed_at"]:
+            by_key[key] = cand
+    distinct = list(by_key.values())
+
+    # Cap order: ALL ext first (FIFO), then int (FIFO).
+    ext = sorted(
+        [c for c in distinct if c["scope"] == "ext"],
+        key=lambda c: c["observed_at"],
+    )
+    internal = sorted(
+        [c for c in distinct if c["scope"] == "int"],
+        key=lambda c: c["observed_at"],
+    )
+    ordered = ext + internal
+
+    creates_today = daily_creates_today(db)
+    created = 0
+    refreshed = 0
+    deferred = 0
+    skipped_breaker = 0
+    cap_active = False
+
+    for cand in ordered:
+        existing = db.scalar(
+            select(EgressObservation).where(EgressObservation.dedup_key == cand["dedup_key"])
+        )
+        if existing:
+            # Refresh under cap is always allowed; breaker emergency may still block
+            # when total/bytes ceilings are hit (checked below for creates; refresh
+            # of existing rows does not increase row count).
+            _refresh_row(existing, cand)
+            refreshed += 1
+            continue

+        # When rows_today already == CAP, remaining candidates are deferred —
+        # not skipped_breaker — so the cycle difference stays visible (N3).
+        if creates_today + created >= DAILY_CREATE_CAP:
+            deferred += 1

+            continue

+        if open_:
+            log.error("egress breaker open: %s", reason)
+            skipped_breaker += 1
+            continue

+        ev = {
+            "src_ip": cand.get("src_ip"),
+            "assoc_reason": cand.get("assoc_reason"),

+            association=cand["association"],
+            peer_association=cand["peer_association"] or "",
+            sensor_id=SENSOR_ID,
+            first_seen=cand["observed_at"],
+            last_seen=cand["observed_at"],
+            observation_count=1,
+            evidence={k: v for k, v in ev.items() if v is not None},
+        )

+        db.flush()
+        created += 1

+    open_, reason = evaluate_egress_breaker(db)

+        cycle_stats={
+            "seen_distinct": len(distinct),

+def acknowledge_egress_signal(

+        confidence=1.0,
+        decided_at=utcnow(),
+        decided_by=user_id,
+    )
+    db.add(s)
+    db.flush()

+"""Zeek conn.log → hybrid egress raw items → POST /api/ingest/egress.
+
+Aggregates unique (src_ip, dst_ip, dport, proto) since last cursor / closed
+minutes. API applies hybrid grain + association. Does NOT call /api/ingest/flows.

+BATCH_SIZE = 200

+def load_state(path: Path) -> dict[str, Any]:
+    if not path.is_file():
+        return {"offsets": {}, "cold_started": {}, "file_ids": {}}
+    try:
+        data = json.loads(path.read_text(encoding="utf-8"))
+    except (OSError, json.JSONDecodeError):
+        return {"offsets": {}, "cold_started": {}, "file_ids": {}}
+    if not isinstance(data, dict):
+        return {"offsets": {}, "cold_started": {}, "file_ids": {}}
+    return {
+        "offsets": data.get("offsets") if isinstance(data.get("offsets"), dict) else {},

+def save_state(path: Path, state: dict[str, Any]) -> None:
+    path.parent.mkdir(parents=True, exist_ok=True)
+    path.write_text(json.dumps(state, indent=0), encoding="utf-8")

+
+def file_identity(path: Path) -> str:
+    st = path.stat()
+    return f"{getattr(st, 'st_dev', 0)}:{getattr(st, 'st_ino', 0)}"
+
+
+def _read_new_lines(
+    path: Path, offset: int, *, file_id: str | None, prev_id: str | None
+) -> tuple[list[str], int, bool]:
+    if not path.is_file():
+        return [], offset, False
+    rotated = False
+    cur_id = file_id or file_identity(path)
+    if prev_id and prev_id != cur_id:
+        offset = 0
+        rotated = True
+    size = path.stat().st_size
+    if offset > size:
+        offset = 0
+        rotated = True
+    with path.open("rb") as fh:
+        fh.seek(offset)
+        raw = fh.read()
+        new_offset = fh.tell()
+    if not raw:
+        return [], new_offset, rotated
+    text = raw.decode("utf-8", errors="replace")
+    lines = [ln for ln in text.splitlines() if ln.strip() and not ln.startswith("#")]
+    return lines, new_offset, rotated

+def run_zeek_egress_cycle(
+    *,
+    log_dir: Path,
+    post_items: Callable[[list[dict[str, Any]]], None],
+    state_path: Path | None = None,
+) -> dict[str, Any]:
+    """Tail conn.log; aggregate unique hybrid raw keys; POST to /api/ingest/egress."""
+    log_dir = Path(log_dir)
+    path = Path(state_path) if state_path else log_dir.parent / STATE_NAME
+    state = load_state(path)
+    offsets: dict[str, int] = dict(state.get("offsets") or {})
+    cold: dict[str, bool] = dict(state.get("cold_started") or {})
+    file_ids: dict[str, str] = dict(state.get("file_ids") or {})
+
+    # Aggregate unique (src, dst, dport, proto) — API does grain+association.
+    agg: dict[tuple[str, str, int, str], dict[str, Any]] = {}
+    stats = {
+        "conn_rows": 0,
+        "unique_keys": 0,
+        "posted": 0,
+        "initial_full_read": 0,
+        "rotations": 0,
+    }
+
+    fpath = log_dir / "conn.log"
+    if fpath.is_file():
+        key = "conn.log"
+        cur_id = file_identity(fpath)
+        if not cold.get(key):
+            offsets[key] = 0
+            cold[key] = True
+            stats["initial_full_read"] += 1
+        lines, new_off, rotated = _read_new_lines(
+            fpath,
+            int(offsets.get(key, 0)),
+            file_id=cur_id,
+            prev_id=file_ids.get(key),
+        )
+        if rotated:
+            stats["rotations"] += 1
+        offsets[key] = new_off
+        file_ids[key] = cur_id
+        for line in lines:
+            parsed = parse_conn_line(line)
+            if not parsed:
+                continue
+            stats["conn_rows"] += 1
+            src = parsed["src_ip"]
+            dst = parsed["dst_ip"]
+            dport = int(parsed.get("dst_port") or 0)
+            proto = str(parsed.get("proto") or "tcp").lower()
+            tkey = (src, dst, dport, proto)
+            prev = agg.get(tkey)
+            ts = parsed["ts"]
+            obs_iso = ts.isoformat() + "Z" if hasattr(ts, "isoformat") else str(ts)
+            if prev is None:
+                agg[tkey] = {
+                    "src_ip": src,
+                    "dst_ip": dst,
+                    "dport": dport,
+                    "proto": proto,
+                    "observed_at": obs_iso,
+                    "orig_bytes": parsed.get("bytes_out") or parsed.get("bytes"),
+                    "_ts": ts,
+                }
+            elif ts < prev.get("_ts", ts):
+                # Keep earliest observation as FIFO anchor for API cycle order.
+                prev["observed_at"] = obs_iso
+                prev["_ts"] = ts
+
+    items = []
+    for v in agg.values():
+        v.pop("_ts", None)
+        items.append(v)
+    stats["unique_keys"] = len(items)
+
+    def _flush(batch: list[dict[str, Any]]) -> None:
+        if not batch:
+            return
+        post_items(batch)
+        stats["posted"] += len(batch)
+
+    batch: list[dict[str, Any]] = []
+    for item in items:
+        # Serialize observed_at if datetime slipped through
+        obs = item.get("observed_at")
+        if hasattr(obs, "isoformat"):
+            item = {**item, "observed_at": obs.isoformat() + "Z"}
+        batch.append(item)
+        if len(batch) >= BATCH_SIZE:
+            _flush(batch)
+            batch = []
+    _flush(batch)
+

+    state["cold_started"] = cold
+    state["file_ids"] = file_ids
+    save_state(path, state)
+    return stats
