+def _parse_ts(value: Any) -> datetime | None:
+    if value is None:
+        return None
+    if isinstance(value, datetime):
+        return value.replace(tzinfo=None) if value.tzinfo else value
+    text = str(value).strip()
+    if not text:
+        return None
+    try:
+        if text.endswith("Z"):
+            text = text[:-1] + "+00:00"
+        dt = datetime.fromisoformat(text)
+        return dt.astimezone().replace(tzinfo=None) if dt.tzinfo else dt
+    except ValueError:
+        return None

+def _interval_sec(settings: Settings, attr: str | None) -> int | None:
+    if attr is None:
+        return None
+    # Collector attrs may live only on collector Settings; API Settings has a subset.
+    # Prefer API settings; fall back to known defaults declared in code (not invented).
+    if hasattr(settings, attr):
+        val = getattr(settings, attr)
+        if val is None:
+            return None
+        raw = max(1, int(val))
+    else:
+        # Defaults from collector/config.py (documented, not invented).
+        defaults = {
+            "topology_poll_interval_sec": 60,
+            "collector_interval_sec": 900,
+            "zeek_provider_interval_sec": 300,
+            "zeek_behavior_interval_sec": 300,
+            "internet_probe_interval_sec": 60,
+            "snmp_nas_poll_interval_sec": 60,
+        }
+        if attr not in defaults:
+            return None
+        raw = defaults[attr]
+    # Floor identici a collector/main.py configure_scheduler (cadenza reale, non inventata).
+    floors = {
+        "topology_poll_interval_sec": 15,
+        "collector_interval_sec": 60,
+        "internet_probe_interval_sec": 30,
+        "zeek_provider_interval_sec": 60,
+        "zeek_behavior_interval_sec": 60,
+        "snmp_nas_poll_interval_sec": 60,
+    }
+    return max(floors.get(attr, 1), raw)

+def _last_sensor_run(db: Session, source: str) -> dict[str, Any]:
+    last_ok = db.scalar(
+        select(SensorRun)

+        "last_success_at": last_ok.started_at if last_ok else None,
+        "last_error_at": last_err.started_at if last_err else None,
+        "last_error": (last_err.error if last_err else None)

+def _last_fdb_snmp(db: Session) -> dict[str, Any]:
+    """Aggregate FDB-capable switches only; GS308EP excluded (I7)."""
+    newest_ok: datetime | None = None
+    newest_err: str | None = None
+    newest_err_at: datetime | None = None
+    supporting = 0
+    for sw in db.scalars(select(Switch)).all():
+        cap = switch_capabilities(getattr(sw, "model", None) or sw.name)
+        if cap.get("fdb_supported") is False:
+            continue
+        supporting += 1
+        meta = dict(sw.meta or {})
+        fp = dict(meta.get("fdb_poll") or {})
+        at = _parse_ts(fp.get("at"))
+        ok = fp.get("ok")
+        err = str(fp.get("error") or "")
+        if ok and at and (newest_ok is None or at > newest_ok):
+            newest_ok = at
+        if ok is False and at and (newest_err_at is None or at > newest_err_at):
+            newest_err_at = at
+            newest_err = err or "poll FDB non ok (errore assente nel meta — difetto di registro)"
+    return {
+        "last_success_at": newest_ok,
+        "last_error_at": newest_err_at,
+        "last_error": newest_err,
+        "last_attempt_ok": None if newest_ok is None and newest_err_at is None else (
+            newest_ok is not None
+            and (newest_err_at is None or newest_ok >= newest_err_at)
+        ),
+        "last_attempt_at": max(
+            [t for t in (newest_ok, newest_err_at) if t is not None],
+            default=None,
+        ),
+        "fdb_supporting_switches": supporting,
+    }

+    last_error: str | None,
+    now: datetime,
+) -> dict[str, Any]:
+    """Pure C2 classifier — used by production and C5 tests."""
+    if not enabled:
+        return {
+            "state": "disabilitata",
+            "reason": f"flag {enable_flag or '?'} = false (scelta operatore, non guasto)",
+            "age_sec": None,
+            "window_sec": interval_sec,
+        }
+    if interval_sec is None:
+        return {
+            "state": "cadenza_non_nota",
+            "reason": "cadenza non dichiarata in configurazione — nessuna finestra inventata",
+            "age_sec": None,
+            "window_sec": None,
+        }
+    if last_success_at is None:
+        return {
+            "state": "cieca",
+            "reason": "nessun successo registrato mentre la sorgente è abilitata",
+            "age_sec": None,
+            "window_sec": interval_sec,
+            "last_error": last_error,
+        }
+    age = (now - last_success_at).total_seconds()
+    if age <= interval_sec:
+        return {
+            "state": "coperta_fresca",
+            "reason": f"ultimo successo entro cadenza {interval_sec}s",
+            "age_sec": round(age, 3),
+            "window_sec": interval_sec,
+        }
+    # Oltre cadenza con data nota → coperta_vecchia (I2). Card allarme: sì (cieca per difesa).
+    return {
+        "state": "coperta_vecchia",
+        "reason": (
+            f"ultimo successo oltre cadenza {interval_sec}s "
+            f"(age={round(age, 1)}s) — data {last_success_at.isoformat()}"
+        ),
+        "age_sec": round(age, 3),
+        "window_sec": interval_sec,
+        "last_error": last_error,
+        "alarm": True,  # oltre cadenza = non coperto per difesa (C4)
+    }
+

+def evaluate_source(
+    db: Session,
+    spec: SourceSpec,
+    settings: Settings,
+    *,
+    now: datetime | None = None,
+    override_enabled: bool | None = None,
+    override_last_success: datetime | None = None,
+    override_last_error: str | None = None,
+    override_interval: int | None = None,
+) -> dict[str, Any]:
+    now = now or utcnow()
+    if spec.kind == "structural_skip":
+        return {
+            "name": spec.name,
+            "label": spec.label,
+            "state": "limite_strutturale",
+            "o12_marker": "obs-o12-structural",
+            "note": "I7 — GS308EP non espone FDB SNMP; non è sorgente cieca",
+            "job_or_endpoint": spec.job_or_endpoint,
+            "feeds": list(spec.feeds),
+            "signals_lost_when_blind": [],
+            "enable_flag": None,
+            "enabled": None,
+            "interval_sec": None,
+            "freshness_vocab": "not_covered",
+            "card_eligible": False,
+        }
+
+    enabled, flag = _flag_enabled(settings, spec.enable_flag)
+    if override_enabled is not None:
+        enabled = override_enabled
+    # flow family: also need zeek_provider for collector side when classifying "should run"
+    if spec.name in {"zeek_conn_flow", "zeek_intel"} and enabled:
+        # API ingest on is not enough if provider off — still "disabilitata" if provider off
+        if not bool(getattr(settings, "zeek_provider_enabled", False)):
+            enabled = False
+            flag = "zeek_provider_enabled"
+    if spec.name == "zeek_dhcp_names":
+        enabled, flag = _flag_enabled(settings, "zeek_provider_enabled")
+        if override_enabled is not None:
+            enabled = override_enabled
+
+    interval = (
+        override_interval
+        if override_interval is not None
+        else _interval_sec(settings, spec.interval_attr)
+    )
+
+    if override_last_success is not None or override_last_error is not None:
+        last = {
+            "last_success_at": override_last_success,
+            "last_error": override_last_error,
+            "last_error_at": None,
+            "last_attempt_ok": None,
+            "last_attempt_at": override_last_success,
+        }
+    elif spec.kind == "sensor_run":
+        last = _last_sensor_run(db, _SENSOR_RUN_SOURCE[spec.name])
+    elif spec.kind == "fdb_snmp":
+        last = _last_fdb_snmp(db)
+    elif spec.kind == "zeek_behavior":
+        last = _last_zeek_behavior(db)
+    elif spec.kind == "portal":
+        last = _last_portal(db)
+    elif spec.kind == "internet":
+        last = _last_internet(db)
+    elif spec.kind == "nas":
+        last = _last_nas(db)
+    elif spec.kind == "flow":
+        if spec.name == "zeek_conn_flow":
+            last = _last_flow(db)
+        else:
+            # intel / dhcp_names: no dedicated success table — declare registry gap
+            last = {
+                "last_success_at": None,
+                "last_error": (
+                    "successo/errore non registrati in tabella dedicata "
+                    "(solo state JSON / log — difetto di registro)"
+                ),
+                "last_error_at": None,
+                "last_attempt_ok": None,
+                "last_attempt_at": None,
+                "registry_note": "cadenza nota; ultimo successo non misurato in DB",
+            }
+    else:
+        last = {
+            "last_success_at": None,
+            "last_error": "kind sconosciuto",
+            "last_error_at": None,
+            "last_attempt_ok": None,
+            "last_attempt_at": None,
+        }
+
+    # Special: flow/intel/dhcp with registry gap but disabled → disabilitata first
+    cls = classify_source(
+        enabled=enabled,
+        enable_flag=flag or spec.enable_flag,
+        interval_sec=interval,
+        last_success_at=last.get("last_success_at"),
+        last_error=last.get("last_error"),
+        now=now,
+    )
+    # Soft: coperta_vecchia is primary; alarm path uses alarm flag / state set
+    display_state = cls["state"]
+    display_extra = None
+
+    last_success = last.get("last_success_at")
+    not_covered = display_state in {"cieca", "coperta_vecchia", "disabilitata"}
+    return {
+        "name": spec.name,
+        "label": spec.label,
+        "state": display_state,
+        "display_extra": display_extra,
+        "reason": cls.get("reason"),
+        "job_or_endpoint": spec.job_or_endpoint,
+        "feeds": list(spec.feeds),
+        "signals_lost_when_blind": list(spec.signals_lost_when_blind)
+        if display_state in {"cieca", "coperta_vecchia"}
+        else (
+            list(spec.signals_lost_when_blind)
+            if display_state == "disabilitata"
+            else []
+        ),
+        "signals_not_observable": list(spec.signals_lost_when_blind) if not_covered else [],
+        "enable_flag": flag or spec.enable_flag,
+        "enabled": enabled,
+        "interval_sec": interval,
+        "interval_attr": spec.interval_attr,
+        "age_sec": cls.get("age_sec"),
+        "window_sec": cls.get("window_sec"),
+        "last_success_at": last_success.isoformat() + "Z" if last_success else None,
+        "last_error": last.get("last_error"),
+        "last_error_at": (
+            last["last_error_at"].isoformat() + "Z"
+            if last.get("last_error_at")
+            else None
+        ),
+        "registry_note": last.get("registry_note"),
+        "freshness_vocab": STATE_TO_FRESHNESS.get(display_state, "not_measured"),
+        "o12_marker": "obs-o12-source",
+        "card_eligible": spec.card_eligible,
+        "defense_priority_id": (
+            "sorgente_non_disponibile"
+            if display_state in {"cieca", "coperta_vecchia"}
+            else "igiene_nome"
+        ),
+    }

+def acknowledge_coverage_signal(
+    db: Session,
+    *,
+    uid: str,
+    action: str,
+    reason: str = "",
+    user_id: int | None = None,
+    kind: str = "coverage_source_blind",
+) -> Suggestion:
+    status = "approved" if action in {"acknowledge", "archive"} else "rejected"
+    existing = db.scalar(select(Suggestion).where(Suggestion.uid == uid))
+    if existing:
+        existing.status = status
+        existing.decided_at = utcnow()
+        existing.decided_by = user_id
+        existing.body = reason or existing.body
+        payload = dict(existing.payload or {})
+        payload["action"] = action
+        existing.payload = payload
+        db.flush()
+        return existing
+    s = Suggestion(
+        uid=uid,
+        kind=kind[:64],
+        source="coverage",
+        status=status,
+        title=(reason or uid)[:255],
+        body=reason or action,
+        payload={"action": action, "o12": True},
+        decided_at=utcnow(),
+        decided_by=user_id,
+    )
+    db.add(s)
+    db.flush()
+    return s
