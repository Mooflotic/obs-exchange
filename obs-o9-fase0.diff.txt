+#!/usr/bin/env python3
+"""O9 X1 — assert breakpoint PNG widths (prove content capture, not hash alone).
+
+Usage:
+  python3 scripts/o9_png_assert.py path.png EXPECTED_WIDTH [EXPECTED_HEIGHT]
+  python3 scripts/o9_png_assert.py --pair a.png WA b.png WB
+
+Exit 0 only if dimensions match. Prints W×H and sha256.
+"""
+from __future__ import annotations
+
+import hashlib
+import struct
+import sys
+from pathlib import Path
+
+
+def png_size(path: Path) -> tuple[int, int]:
+    data = path.read_bytes()
+    if data[:8] != b"\x89PNG\r\n\x1a\n":
+        raise SystemExit(f"FAIL not PNG: {path}")
+    w, h = struct.unpack(">II", data[16:24])
+    return int(w), int(h)
+
+
+def sha256(path: Path) -> str:
+    return hashlib.sha256(path.read_bytes()).hexdigest()
+
+
+def main(argv: list[str]) -> int:
+    if len(argv) >= 2 and argv[0] == "--pair":
+        if len(argv) != 5:
+            print("Uso: --pair a.png WA b.png WB", file=sys.stderr)
+            return 2
+        a, wa, b, wb = Path(argv[1]), int(argv[2]), Path(argv[3]), int(argv[4])
+        aw, ah = png_size(a)
+        bw, bh = png_size(b)
+        print(f"{a.name}: {aw}x{ah} sha256={sha256(a)}")
+        print(f"{b.name}: {bw}x{bh} sha256={sha256(b)}")
+        if aw != wa:
+            print(f"FAIL {a}: width {aw} != {wa}", file=sys.stderr)
+            return 1
+        if bw != wb:
+            print(f"FAIL {b}: width {bw} != {wb}", file=sys.stderr)
+            return 1
+        if aw == bw:
+            print(
+                f"FAIL pair same width {aw}: non prova due breakpoint",
+                file=sys.stderr,
+            )
+            return 1
+        print("PASS pair distinct widths")
+        return 0
+
+    if len(argv) < 2:
+        print(__doc__, file=sys.stderr)
+        return 2
+    path = Path(argv[0])
+    expect_w = int(argv[1])
+    expect_h = int(argv[2]) if len(argv) > 2 else None
+    w, h = png_size(path)
+    print(f"{path.name}: {w}x{h} sha256={sha256(path)}")
+    if w != expect_w:
+        print(f"FAIL width {w} != {expect_w}", file=sys.stderr)
+        return 1
+    if expect_h is not None and h != expect_h:
+        print(f"FAIL height {h} != {expect_h}", file=sys.stderr)
+        return 1
+    print("PASS")
+    return 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(main(sys.argv[1:]))

+#!/usr/bin/env python3
+"""Decode a CDP Page.captureScreenshot JSON dump to PNG and print WxH.
+
+Harness O9 X1 (diagnosi O8):
+  - `browser_take_screenshot` cattura la superficie Electron fissa (~768px) e
+    IGNORA Emulation.setDeviceMetricsOverride → tre file stessa larghezza.
+  - Procedura corretta:
+      1) CDP Emulation.setDeviceMetricsOverride({width, height, deviceScaleFactor:1, mobile})
+      2) CDP Runtime.evaluate → verificare window.innerWidth == width attesa
+      3) CDP Page.captureScreenshot({format:'png', captureBeyondViewport:false|true})
+      4) salvare con questo script / assert con o9_png_assert.py
+
+Usage:
+  python3 scripts/o9_cdp_png_decode.py cdp-response.json out.png
+"""
+from __future__ import annotations
+
+import base64
+import json
+import struct
+import sys
+from pathlib import Path
+
+
+def main() -> int:
+    if len(sys.argv) != 3:
+        print(__doc__, file=sys.stderr)
+        return 2
+    src, dest = Path(sys.argv[1]), Path(sys.argv[2])
+    payload = json.loads(src.read_text())
+    b64 = payload.get("data")
+    if not b64 and isinstance(payload.get("result"), dict):
+        b64 = payload["result"].get("data")
+    if not b64:
+        print("FAIL: no data field in CDP JSON", file=sys.stderr)
+        return 1
+    raw = base64.b64decode(b64)
+    dest.write_bytes(raw)
+    w, h = struct.unpack(">II", raw[16:24])
+    print(f"WROTE {dest} {w}x{h} bytes={len(raw)}")
+    return 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())

+# O9 X1 — harness cattura breakpoint (diagnosi + procedura)
+
+## Diagnosi O8 (non ipotesi)
+
+I tre PNG O8 misurano tutti **768×1756** (anche quelli etichettati 1280/390).
+
+Causa accertata:
+
+1. `Emulation.setDeviceMetricsOverride` **funziona** su `window.innerWidth` / `matchMedia`.
+2. `browser_take_screenshot` (MCP) cattura la **superficie Electron fissa** (~768 CSS px) e **ignora** l’emulazione → stessa larghezza PNG.
+3. Con `deviceScaleFactor: 2` e width 390, `Page.captureScreenshot` produce **780×1688** (= 390×2): i pixel del PNG sono *device* pixels. Per X1 usare sempre **`deviceScaleFactor: 1`** così larghezza PNG = breakpoint CSS.
+
+## Procedura obbligatoria (agente / revisione)
+
+Per ogni breakpoint `W` ∈ {1280, 768, 390}:
+
+1. CDP `Emulation.setDeviceMetricsOverride`  
+   `{ width: W, height: H, deviceScaleFactor: 1, mobile: W < 1280 }`
+2. CDP `Runtime.evaluate` → assert `window.innerWidth === W`
+3. Prova di **contenuto** (non solo hash), es.:
+   - 1280: bottone `☰ Menu` con `display:none`
+   - ≤800: stesso bottone `display:flex` / `visible:true`
+4. CDP `Page.captureScreenshot` `{ format: "png", captureBeyondViewport: false }`  
+   (**non** usare `browser_take_screenshot` per i breakpoint)
+5. Decodare JSON → PNG con `scripts/o9_cdp_png_decode.py`
+6. Assert con `scripts/o9_png_assert.py` / `--pair` (larghezze distinte)
+
+## Full-page
+
