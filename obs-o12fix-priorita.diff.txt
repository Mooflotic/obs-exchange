+/**
+ * @param {string} id
+ * @param {string} [ruleDetail]
+ * @returns {{ level: number, id: string, label: string, rule: string }}
+ */
+export function defensePriority(id, ruleDetail = "") {
+  const base = BY_ID[id] || BY_ID.igiene_nome;
+  return {
+    level: base.level,
+    id: base.id,
+    label: base.label,
+    rule: ruleDetail || base.label,
+  };
+}

+/**
+ * FDB S-A…S-D → livello difensivo (mai soppressi come rumore).
+ * Mapping allineato a `fdb_defense.py` kind (O9 X2):
+ *   S-A / fdb_mac_new  → P2 MAC nuovo
+ *   S-B / fdb_mac_move → P3 MAC-move
+ *   S-C / fdb_l2_only  → P1 solo-L2
+ *   S-D               → P4 sorgente non disponibile
+ * @param {object} card
+ */
+export function priorityForFdbCard(card) {
+  const signal = String(card?.signal || "");
+  const kind = String(card?.kind || "");
+  const role = String(card?.ruolo_porta || "");
+  const uncertainRole = role === "non_determinato";
+  let p;
+  const isSoloL2 = kind === "fdb_l2_only" || signal === "S-C";
+  const isMacNew = kind === "fdb_mac_new" || signal === "S-A";
+  const isMacMove = kind === "fdb_mac_move" || signal === "S-B";
+  const isBlind = signal === "S-D" || kind === "fdb_source_stale";
+
+  if (isSoloL2) {
+    p = defensePriority(
+      "non_riconosciuto_o_solo_l2",
+      uncertainRole
+        ? "S-C solo-L2 · ruolo_porta non_determinato · priorità dichiarata ma ridotta in UI"
+        : "S-C solo-L2 (presente in FDB senza IP corrente)",
+    );
+  } else if (isMacNew) {
+    p = defensePriority(
+      "mac_nuovo_accesso",
+      uncertainRole
+        ? "S-A MAC nuovo · ruolo_porta non_determinato · priorità dichiarata ma ridotta in UI"
+        : "S-A MAC nuovo su porta di accesso",
+    );
+  } else if (isMacMove) {
+    p = defensePriority(
+      "mac_move_accesso",
+      uncertainRole
+        ? "S-B MAC-move · ruolo_porta non_determinato · priorità dichiarata ma ridotta in UI"
+        : "S-B MAC-move fra porte di accesso",
+    );
+  } else if (isBlind) {
+    p = defensePriority(
+      "sorgente_non_disponibile",
+      `S-D · ${card?.state || "sorgente_non_disponibile"}`,
+    );
+  } else {
+    p = defensePriority("igiene_nome", `segnale FDB ${signal || kind || "sconosciuto"}`);
+  }
+  return { ...p, role_uncertain: uncertainRole };
+}

+/**
+ * O12-FIX K3 — legenda e livelli usati devono coincidere (P1–P7, nessun livello fantasma).
+ */
+import test from "node:test";
+import assert from "node:assert/strict";
+import {
+  DEFENSE_PRIORITY_LEGEND,
+  defensePriority,
+  priorityForCoverageCard,
+} from "./oggiPriority.js";
+
+test("legenda dichiara P1–P7 senza buchi", () => {
+  const levels = DEFENSE_PRIORITY_LEGEND.map((x) => x.level);
+  assert.deepEqual(levels, [1, 2, 3, 4, 5, 6, 7]);
+  assert.equal(DEFENSE_PRIORITY_LEGEND.length, 7);
+  assert.ok(DEFENSE_PRIORITY_LEGEND.some((x) => x.id === "igiene_nome" && x.level === 7));
+});
+
+test("coverage cieca → P4; disabilitata → P7 igiene_nome", () => {
+  const blind = priorityForCoverageCard({
+    kind: "coverage_source_blind",
+    defense_priority_id: "sorgente_non_disponibile",
+  });
+  assert.equal(blind.level, 4);
+  assert.ok(DEFENSE_PRIORITY_LEGEND.some((L) => L.level === blind.level));
+
+  const disabled = priorityForCoverageCard({
+    kind: "coverage_source_disabled",
+    defense_priority_id: "igiene_nome",
+  });
+  assert.equal(disabled.level, 7);
+  assert.equal(disabled.id, "igiene_nome");
+  assert.ok(DEFENSE_PRIORITY_LEGEND.some((L) => L.id === disabled.id && L.level === 7));
+});
+
+test("defensePriority non inventa livelli fuori legenda", () => {
+  const p = defensePriority("igiene_nome");
+  assert.equal(p.level, 7);
+  const unknown = defensePriority("id_inesistente_xyz");
+  assert.equal(unknown.id, "igiene_nome");
+  assert.equal(unknown.level, 7);
+});
