+"""OBS-TIMING-034c: fragment merge integrity (correct / stale / missing)."""
+
+from __future__ import annotations
+
+import json
+from pathlib import Path
+
+from app.services.boot_timing import build_api_boot_payload
+
+
+def _env(epoch: int = 1000) -> dict[str, str]:
+    return {
+        "OBS_BOOT_EPOCH_START": str(epoch),
+        "OBS_BOOT_T_IMPORT": "0",
+        "OBS_BOOT_T_BOOTSTRAP_WALL": "10",
+        "OBS_BOOT_VERSION": "0.10.26",
+    }
+
+
+def test_fragment_correct_merges_trust_timers(tmp_path: Path):
+    frag = tmp_path / "obs_boot_timing.json"
+    frag.write_text(
+        json.dumps(
+            {
+                "epoch_start": 1000,
+                "needs_backup": False,
+                "needs_apply": False,
+                "T_trust": 12.5,
+                "T_dry_run": 12.0,
+                "T_prefetch_obs": 11.5,
+                "T_prefetch_fdb": 0.2,
+                "T_plan_loop": 0.3,
+                "T_dry_residuo": 0.0,
+                "T_backup": 0.0,
+                "T_apply": 0.0,
+                "T_residuo_trust": 0.0,
+            }
+        )
+    )
+    payload = build_api_boot_payload(
+        epoch_ready=1025, env=_env(1000), fragment_path=frag
+    )
+    assert payload.get("bootstrap_timing_stale") is None
+    assert payload.get("bootstrap_timing_missing") is None
+    assert payload["needs_apply"] is False
+    assert payload["timers"]["T_prefetch_obs"]["s"] == 11.5
+    assert payload["timers"]["T_prefetch_obs"]["parent"] == "T_dry_run"
+    assert payload["timers"]["T_uvicorn_to_ready"]["s"] == 15.0  # 25 - 0 - 10
+    assert "T_uvicorn_to_health" not in payload["timers"]
+    assert payload["epoch_ready"] == 1025
+
+
+def test_fragment_stale_trust_timers_are_null(tmp_path: Path):
+    frag = tmp_path / "obs_boot_timing.json"
+    frag.write_text(
+        json.dumps(
+            {
+                "epoch_start": 999,  # previous boot
+                "needs_apply": True,
+                "T_prefetch_obs": 48.3,
+                "T_trust": 49.0,
+            }
+        )
+    )
+    payload = build_api_boot_payload(
+        epoch_ready=1100, env=_env(1000), fragment_path=frag
+    )
+    assert payload["bootstrap_timing_stale"] is True
+    assert payload["fragment_epoch_start"] == 999
+    assert payload["host_epoch_start"] == 1000
+    assert payload["needs_apply"] is None
+    assert payload["timers"]["T_prefetch_obs"]["s"] is None
+    assert payload["timers"]["T_trust"]["s"] is None
+    # Host timers still present
+    assert payload["timers"]["T_bootstrap_wall"]["s"] == 10.0
