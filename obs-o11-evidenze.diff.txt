+"""O11 B1 — IP↔MAC association at observation time via the fact store.
+
+Lives under ``api/app/facts/`` so currency is never read outside the resolver
+boundary. Callers outside this package must use these helpers, not raw
+``FactAssertion`` queries for association.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass
+from datetime import datetime
+from typing import Any
+
+from sqlalchemy import and_, or_, select
+from sqlalchemy.orm import Session
+from sqlalchemy import func
+
+from app.models import Asset, FactAssertion, Interface, utcnow
+
+FACT_IFACE_IP = "asset.iface_ip"
+
+
+@dataclass(frozen=True)
+class IpMacAssociation:
+    """Result of resolving an IP at a point in time (I2: uncertain declared)."""
+
+    status: str  # certain | uncertain | unresolved
+    ip: str
+    mac: str = ""
+    interface_id: int | None = None
+    asset_id: int | None = None
+    chassis_id: int | None = None
+    fact_id: int | None = None
+    fact_state: str = ""
+    reason: str = ""
+
+
+def _norm_mac(mac: str | None) -> str:
+    return (mac or "").replace("-", ":").strip().upper()
+
+
+def _covers(fa: FactAssertion, at: datetime) -> bool:
+    """True if the assertion's validity window covers ``at`` (naive UTC)."""
+    vf = fa.valid_from
+    vt = fa.valid_to
+    if vf is not None and at < vf:
+        return False
+    if vt is not None and at >= vt:
+        return False
+    return True
+
+
+def associate_ip_at(
+    db: Session, ip: str, observed_at: datetime | None = None
+) -> IpMacAssociation:
+    """Map IP → MAC/asset using iface_ip facts current (or covering) at ``observed_at``.
+
+    - ``certain``: a ``state=current`` iface_ip fact covers the timestamp.
+    - ``uncertain``: only superseded/historical/stale facts cover it, or the
+      binding existed but was not current at observation time.
+    - ``unresolved``: no iface_ip fact for this IP at all.
+    """
+    ip = (ip or "").strip()
+    at = observed_at or utcnow()
+    if not ip:
+        return IpMacAssociation(status="unresolved", ip="", reason="ip assente")
+
+    rows = list(
+        db.scalars(
+            select(FactAssertion)
+            .where(
+                FactAssertion.fact_key == FACT_IFACE_IP,
+                FactAssertion.value_norm == ip,
+            )
+            .order_by(FactAssertion.id.desc())
+        ).all()
+    )
+    if not rows:
+        return IpMacAssociation(
+            status="unresolved",
+            ip=ip,
+            reason="nessun fatto asset.iface_ip per questo IP",
+        )
+
+    covering = [fa for fa in rows if _covers(fa, at)]
+    current_covering = [fa for fa in covering if (fa.state or "") == "current"]
+
+    def _pack(fa: FactAssertion, status: str, reason: str) -> IpMacAssociation:
+        iface_id = int(fa.subject_id) if fa.subject_type == "interface" else None
+        mac = ""
+        asset_id = None
+        chassis_id = None
+        if iface_id is not None:
+            iface = db.get(Interface, iface_id)
+            if iface:
+                mac = _norm_mac(iface.mac)
+                asset_id = int(iface.asset_id)
+                asset = db.get(Asset, asset_id)
+                if asset and asset.chassis_id:
+                    chassis_id = int(asset.chassis_id)
+        return IpMacAssociation(
+            status=status,
+            ip=ip,
+            mac=mac,
+            interface_id=iface_id,
+            asset_id=asset_id,
+            chassis_id=chassis_id,
+            fact_id=int(fa.id),
+            fact_state=str(fa.state or ""),
+            reason=reason,
+        )
+
+    if current_covering:
+        return _pack(
+            current_covering[0],
+            "certain",
+            "fatto asset.iface_ip current copre observed_at",
+        )
+
+    if covering:
+        return _pack(
+            covering[0],
+            "uncertain",
+            f"fatto asset.iface_ip state={covering[0].state} copre observed_at ma non current",
+        )
+
+    # Fact exists but validity window does not cover observed_at.
+    return _pack(
+        rows[0],
+        "uncertain",
+        "fatto asset.iface_ip esiste ma non copre observed_at (associazione incerta)",
+    )
+
+
+def associate_mac_direct(db: Session, mac: str) -> IpMacAssociation:
+    """MAC taken from Zeek DHCP/JA4D log — no IP join required."""
+    m = _norm_mac(mac)
+    if not m:
+        return IpMacAssociation(status="unresolved", ip="", reason="mac assente")
+    iface = db.scalar(select(Interface).where(Interface.mac == m))
+    if iface is None:
+        iface = db.scalar(
+            select(Interface).where(func.upper(Interface.mac) == m)
+        )
+    if iface is None:
+        return IpMacAssociation(
+            status="mac_direct",
+            ip="",
+            mac=m,
+            reason="MAC da log Zeek; interfaccia non in inventario",
+        )
+    asset = db.get(Asset, int(iface.asset_id))
+    chassis_id = int(asset.chassis_id) if asset and asset.chassis_id else None
+    return IpMacAssociation(
+        status="mac_direct",
+        ip="",
+        mac=m,
+        interface_id=int(iface.id),
+        asset_id=int(iface.asset_id),
+        chassis_id=chassis_id,
+        reason="MAC diretto da dhcp.log/ja4d.log",
+    )
+
+
+def subject_for_association(assoc: IpMacAssociation) -> tuple[str, int | None, int | None]:
+    """Prefer chassis subject when present (B3: one behaviour card per apparatus)."""
+    if assoc.chassis_id:
+        return "chassis", assoc.chassis_id, assoc.asset_id
+    if assoc.asset_id:
+        return "asset", assoc.asset_id, assoc.asset_id
+    return "", None, assoc.asset_id

+"""Zeek ssl.log / dhcp.log / ja4d.log → fingerprint evidence POST.
+
+Cold-start: offsets at EOF (no backlog flood — breaker-safe). Dedup on API.
+"""
+
+from __future__ import annotations
+
+import json
+from pathlib import Path
+from typing import Any, Callable, Optional
+
+STATE_NAME = "zeek_behavior_state.json"
+BATCH_SIZE = 100
+
+
+def load_state(path: Path) -> dict[str, Any]:
+    if not path.is_file():
+        return {"offsets": {}, "cold_started": {}}
+    try:
+        data = json.loads(path.read_text(encoding="utf-8"))
+    except (OSError, json.JSONDecodeError):
+        return {"offsets": {}, "cold_started": {}}
+    if not isinstance(data, dict):
+        return {"offsets": {}, "cold_started": {}}
+    offsets = data.get("offsets") if isinstance(data.get("offsets"), dict) else {}
+    cold = data.get("cold_started") if isinstance(data.get("cold_started"), dict) else {}
+    return {"offsets": offsets, "cold_started": cold}
+
+
+def save_state(path: Path, state: dict[str, Any]) -> None:
+    path.parent.mkdir(parents=True, exist_ok=True)
+    path.write_text(json.dumps(state, indent=0), encoding="utf-8")
+
+
+def _read_new_lines(path: Path, offset: int) -> tuple[list[str], int]:
+    if not path.is_file():
+        return [], offset
+    size = path.stat().st_size
+    if offset > size:
+        offset = 0
+    with path.open("rb") as fh:
+        fh.seek(offset)
+        raw = fh.read()
+        new_offset = fh.tell()
+    if not raw:
+        return [], new_offset
+    text = raw.decode("utf-8", errors="replace")
+    lines = [ln for ln in text.splitlines() if ln.strip() and not ln.startswith("#")]
+    return lines, new_offset
+
+
+def parse_ssl_row(row: dict[str, Any]) -> list[dict[str, Any]]:
+    out: list[dict[str, Any]] = []
+    ip = str(row.get("id.orig_h") or "").strip()
+    ts = row.get("ts")
+    ja4 = row.get("ja4")
+    ja4s = row.get("ja4s")
+    if ja4 and ja4 not in ("-", "", None):
+        out.append(
+            {
+                "kind": "ja4",
+                "value": str(ja4),
+                "src_ip": ip,
+                "observed_at": ts,
+                "ja4s": ja4s,
+                "server_name": row.get("server_name"),
+                "uid": row.get("uid"),
+            }
+        )
+    if ja4s and ja4s not in ("-", "", None):
+        out.append(
+            {
+                "kind": "ja4s",
+                "value": str(ja4s),
+                "src_ip": ip,
+                "observed_at": ts,
+                "uid": row.get("uid"),
+            }
+        )
+    return out
+
+
+def parse_dhcp_row(row: dict[str, Any]) -> list[dict[str, Any]]:
+    out: list[dict[str, Any]] = []
+    mac = str(row.get("mac") or "").strip()
+    ip = str(row.get("assigned_addr") or row.get("client_addr") or "").strip()
+    ts = row.get("ts")
+    host = str(row.get("host_name") or row.get("client_fqdn") or "").strip()
+    if host and host not in ("-", "null"):
+        out.append(
+            {
+                "kind": "dhcp_hostname",
+                "value": host,
+                "src_mac": mac,
+                "src_ip": ip,
+                "observed_at": ts,
+            }
+        )
+    pl = row.get("param_list")
+    if pl:
+        if isinstance(pl, list):
+            val = ",".join(str(x) for x in pl)
+        else:
+            val = str(pl)
+        if val and val not in ("-", "null"):
+            out.append(
+                {
+                    "kind": "dhcp_param_list",
+                    "value": val[:255],
+                    "src_mac": mac,
+                    "src_ip": ip,
+                    "observed_at": ts,
+                }
+            )
+    vc = row.get("vendor_class")
+    if vc and str(vc) not in ("-", "", "null"):
+        out.append(
+            {
+                "kind": "dhcp_vendor",
+                "value": str(vc)[:255],
+                "src_mac": mac,
+                "src_ip": ip,
+                "observed_at": ts,
+            }
+        )
+    return out
+
+
+def parse_ja4d_row(row: dict[str, Any]) -> list[dict[str, Any]]:
+    out: list[dict[str, Any]] = []
+    mac = str(row.get("client_mac") or "").strip()
+    ip = str(row.get("id.orig_h") or "").strip()
+    ts = row.get("ts")
+    ja4d = row.get("ja4d")
+    if ja4d and ja4d not in ("-", "", None):
+        out.append(
+            {
+                "kind": "ja4d",
+                "value": str(ja4d)[:255],
+                "src_mac": mac,
+                "src_ip": ip,
+                "observed_at": ts,
+                "uid": row.get("uid"),
+            }
+        )
+    host = str(row.get("hostname") or "").strip()
+    if host and host not in ("-", "null"):
+        out.append(
+            {
+                "kind": "dhcp_hostname",
+                "value": host,
+                "src_mac": mac,
+                "src_ip": ip,
+                "observed_at": ts,
+            }
+        )
+    vc = row.get("vendor_class_id")
+    if vc and str(vc) not in ("-", "", "null"):
+        out.append(
+            {
+                "kind": "dhcp_vendor",
+                "value": str(vc)[:255],
+                "src_mac": mac,
+                "src_ip": ip,
+                "observed_at": ts,
+            }
+        )
+    return out
+
+
+def run_zeek_behavior_cycle(
+    *,
+    log_dir: Path,
+    post_items: Callable[[list[dict[str, Any]]], None],
+    state_path: Path | None = None,
+) -> dict[str, Any]:
+    """Tail current ssl/dhcp/ja4d. First seen: ingest full current file (not rotated)."""
+    log_dir = Path(log_dir)
+    path = Path(state_path) if state_path else log_dir.parent / STATE_NAME
+    state = load_state(path)
+    offsets: dict[str, int] = dict(state.get("offsets") or {})
+    cold: dict[str, bool] = dict(state.get("cold_started") or {})
+
+    batch: list[dict[str, Any]] = []
+    stats = {
+        "ssl_rows": 0,
+        "dhcp_rows": 0,
+        "ja4d_rows": 0,
+        "items": 0,
+        "posted": 0,
+        "initial_full_read": 0,
+    }
+
+    files: list[tuple[str, str]] = [
+        ("ssl.log", "ssl"),
+        ("dhcp.log", "dhcp"),
+        ("ja4d.log", "ja4d"),
+    ]
+
+    def _flush() -> None:
+        nonlocal batch
+        if not batch:
+            return
+        post_items(batch)
+        stats["posted"] += len(batch)
+        batch = []
+
+    for name, kind in files:
+        fpath = log_dir / name
+        if not fpath.is_file():
+            continue
+        key = name
+        # First encounter: read from byte 0 of the *current* log only.
+        if not cold.get(key):
+            offsets[key] = 0
+            cold[key] = True
+            stats["initial_full_read"] += 1
+        lines, new_off = _read_new_lines(fpath, int(offsets.get(key, 0)))
+        offsets[key] = new_off
+        for line in lines:
+            try:
+                row = json.loads(line)
+            except json.JSONDecodeError:
+                continue
+            if kind == "ssl":
+                stats["ssl_rows"] += 1
+                parsed = parse_ssl_row(row)
+            elif kind == "dhcp":
+                stats["dhcp_rows"] += 1
+                parsed = parse_dhcp_row(row)
+            else:
+                stats["ja4d_rows"] += 1
+                parsed = parse_ja4d_row(row)
+            for item in parsed:
+                batch.append(item)
+                stats["items"] += 1
+                if len(batch) >= BATCH_SIZE:
+                    _flush()
+
+    _flush()
+    state["offsets"] = offsets
+    state["cold_started"] = cold
+    save_state(path, state)
+    return stats

+    if not isinstance(items, list):
