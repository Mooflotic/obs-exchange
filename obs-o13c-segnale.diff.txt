+"""O13C — hybrid egress observations (ext host+port; int peer relation).
+
+Binding (do not reopen):
+- scope=ext: (subject, dst_ip, dport, proto)
+- scope=int: (subject, peer_subject) — NO port, NO proto
+- Private = RFC1918 + link-local + ULA; else external
+- Subject = chassis if present else asset — NEVER NIC
+- Cap ≠ breaker: separate functions and counters
+- Cap bites: ALL ext candidates first (FIFO by first obs in cycle), then int.
+  Starvation of int under cap is not loss — deferred reappears next cycle.

+MAX_GROWTH = 2_000  # breaker: first_seen today (excl. meta)

+def _parse_ts(value: Any) -> datetime | None:
+    if value is None:
+        return None
+    if isinstance(value, datetime):
+        return value.replace(tzinfo=None) if value.tzinfo else value
+    if isinstance(value, (int, float)):
+        return datetime.utcfromtimestamp(float(value))
+    text = str(value).strip()
+    if not text:
+        return None
+    try:
+        return datetime.utcfromtimestamp(float(text))
+    except ValueError:
+        pass
+    try:
+        if text.endswith("Z"):
+            text = text[:-1] + "+00:00"
+        dt = datetime.fromisoformat(text)
+        if dt.tzinfo is not None:
+            from datetime import timezone
+
+            return dt.astimezone(timezone.utc).replace(tzinfo=None)
+        return dt
+    except ValueError:
+        return None
+

+# Explicit private ranges per O13C binding (NOT ipaddress.is_private — that also
+# flags TEST-NET / documentation nets which must stay external for egress grain).
+_RFC1918 = (
+    ipaddress.ip_network("10.0.0.0/8"),
+    ipaddress.ip_network("172.16.0.0/12"),
+    ipaddress.ip_network("192.168.0.0/16"),
+)
+_IPV4_LINK_LOCAL = ipaddress.ip_network("169.254.0.0/16")
+_IPV6_ULA = ipaddress.ip_network("fc00::/7")
+_IPV6_LINK_LOCAL = ipaddress.ip_network("fe80::/10")

+def classify_scope(dst_ip: str) -> str:
+    """Return 'int' for RFC1918 / link-local / ULA, else 'ext'."""
+    text = (dst_ip or "").strip()
+    if not text:
+        return "ext"
+    try:
+        addr = ipaddress.ip_address(text)
+    except ValueError:
+        return "ext"
+    if isinstance(addr, ipaddress.IPv4Address):
+        if any(addr in net for net in _RFC1918):
+            return "int"
+        if addr in _IPV4_LINK_LOCAL:
+            return "int"
+        return "ext"
+    # IPv6: ULA + link-local only (not the broader is_private set).
+    if addr in _IPV6_ULA or addr in _IPV6_LINK_LOCAL:
+        return "int"
+    return "ext"

+def _subject_token(stype: str, sid: int | None, fallback_ip: str = "") -> str:
+    if stype and sid is not None:
+        return f"{stype}:{int(sid)}"
+    if fallback_ip:
+        digest = hashlib.sha256(fallback_ip.encode("utf-8")).hexdigest()[:12]
+        return f"u:{digest}"
+    return "u:unknown"

+def dedup_key_ext(
+    subject_type: str,
+    subject_id: int | None,
+    dst_ip: str,
+    dport: int,
+    proto: str,
+    *,
+    src_ip: str = "",
+) -> str:
+    subj = _subject_token(subject_type, subject_id, src_ip)
+    raw = f"ext|{subj}|{dst_ip}|{int(dport)}|{(proto or 'tcp').lower()}"
+    return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:64]
+
+
+def dedup_key_int(
+    subject_type: str,
+    subject_id: int | None,
+    peer_subject_type: str,
+    peer_subject_id: int | None,
+    *,
+    src_ip: str = "",
+    peer_ip: str = "",
+) -> str:
+    # int grain: NO port, NO proto — relation only.
+    a = _subject_token(subject_type, subject_id, src_ip)
+    b = _subject_token(peer_subject_type, peer_subject_id, peer_ip)
+    # Canonical order so A↔B and B↔A collapse when both resolved to subjects.
+    left, right = sorted([a, b])
+    raw = f"int|{left}|{right}"
+    return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:64]

+def table_stats(db: Session) -> dict[str, Any]:
+    total = (
+        db.scalar(
+            select(func.count())
+            .select_from(EgressObservation)
+            .where(EgressObservation.scope != BASELINE_SCOPE)
+        )
+        or 0
+    )
+    today_start = utcnow().replace(hour=0, minute=0, second=0, microsecond=0)
+    today = (
+        db.scalar(
+            select(func.count())
+            .select_from(EgressObservation)
+            .where(
+                EgressObservation.scope != BASELINE_SCOPE,
+                EgressObservation.first_seen >= today_start,
+            )
+        )
+        or 0
+    )
+    rows = list(
+        db.scalars(
+            select(EgressObservation).where(EgressObservation.scope != BASELINE_SCOPE)
+        ).all()
+    )
+    approx = sum(
+        len(r.dedup_key or "")
+        + len(r.scope or "")
+        + len(r.dst_ip or "")
+        + len(r.peer_ip or "")
+        + len(r.proto or "")
+        + 96
+        for r in rows
+    )
+    return {
+        "rows_total": int(total),
+        "rows_today": int(today),
+        "approx_bytes": int(approx),
+        "ceilings": {
+            "rows_total": MAX_ROWS,
+            "rows_today": MAX_GROWTH,
+            "bytes": MAX_BYTES,
+        },
+    }

+def evaluate_egress_breaker(db: Session) -> tuple[bool, str]:
+    """Emergency stop for ALL new writes (create path). Separate from daily create cap."""
+    st = table_stats(db)
+    if st["rows_total"] >= MAX_ROWS:
+        return True, f"egress_observations rows_total={st['rows_total']} >= {MAX_ROWS}"
+    if st["rows_today"] >= MAX_GROWTH:
+        return True, f"egress_observations rows_today={st['rows_today']} >= {MAX_GROWTH}"
+    if st["approx_bytes"] >= MAX_BYTES:
+        return True, f"egress_observations approx_bytes={st['approx_bytes']} >= {MAX_BYTES}"
+    return False, ""

+def maybe_close_baseline_after_cycle(db: Session, deferred_count: int, *, cycle_stats: dict | None = None) -> dict[str, Any]:
+    """Auto-mark ready after a full cycle with deferred_creates==0 while in_costruzione."""
+    row = ensure_baseline_meta(db)
+    ev = dict(row.evidence or {})
+    if cycle_stats:
+        ev["cycle_seen"] = cycle_stats.get("seen_distinct")
+        ev["cycle_created"] = cycle_stats.get("created")
+        ev["cycle_deferred"] = deferred_count
+        row.evidence = ev
+        row.last_seen = utcnow()
+    if ev.get("baseline_ready"):
+        db.flush()
+        return baseline_status(db)
+    # Full cycle = at least one distinct candidate processed and nothing deferred.
+    activity = int((cycle_stats or {}).get("seen_distinct") or 0)
+    if int(deferred_count or 0) == 0 and activity > 0:
+        now = utcnow()
+        ev["baseline_ready"] = True
+        ev["baseline_status"] = "ready"
+        ev["baseline_ready_at"] = now.isoformat() + "Z"
+        ev["reason"] = "ciclo completo con deferred_creates==0 — baseline ready (auto)"
+        row.evidence = ev
+        row.last_seen = now
+        db.flush()
+    else:
+        db.flush()
+    return baseline_status(db)

+def prepare_candidate(db: Session, raw: dict[str, Any]) -> dict[str, Any] | None:
+    """Map raw {src_ip,dst_ip,dport,proto,observed_at,...} → candidate with association."""
+    src_ip = str(raw.get("src_ip") or "").strip()
+    dst_ip = str(raw.get("dst_ip") or "").strip()
+    if not src_ip or not dst_ip:
+        return None
+    observed_at = _parse_ts(raw.get("observed_at")) or utcnow()
+    try:
+        dport = int(raw.get("dport") if raw.get("dport") is not None else raw.get("dst_port") or 0)
+    except (TypeError, ValueError):
+        dport = 0
+    proto = str(raw.get("proto") or "tcp").strip().lower() or "tcp"
+    scope = classify_scope(dst_ip)
+
+    src_assoc = associate_ip_at(db, src_ip, observed_at)
+    subject_type, subject_id, asset_id = subject_for_association(src_assoc)
+
+    peer_subject_type = ""
+    peer_subject_id: int | None = None
+    peer_unresolved = False
+    peer_assoc_status = ""
+    peer_assoc = None
+
+    if scope == "int":
+        peer_assoc = associate_ip_at(db, dst_ip, observed_at)
+        peer_assoc_status = peer_assoc.status
+        pst, psid, _ = subject_for_association(peer_assoc)
+        if pst and psid is not None:
+            peer_subject_type, peer_subject_id = pst, psid
+        else:
+            # I2: keep peer_ip, mark unresolved — do NOT invent subject.
+            peer_unresolved = True
+            peer_assoc_status = peer_assoc.status if peer_assoc.status != "unresolved" else "unresolved"
+
+    if scope == "ext":
+        key = dedup_key_ext(subject_type, subject_id, dst_ip, dport, proto, src_ip=src_ip)
+    else:
+        key = dedup_key_int(
+            subject_type,
+            subject_id,
+            peer_subject_type,
+            peer_subject_id,
+            src_ip=src_ip,
+            peer_ip=dst_ip,
+        )
+
+    return {
+        "dedup_key": key,
+        "scope": scope,
+        "subject_type": subject_type or "",
+        "subject_id": subject_id,
+        "asset_id": asset_id,
+        "dst_ip": dst_ip if scope == "ext" else "",
+        "dport": dport if scope == "ext" else None,
+        "proto": proto if scope == "ext" else "",
+        "peer_subject_type": peer_subject_type,
+        "peer_subject_id": peer_subject_id,
+        "peer_ip": dst_ip if scope == "int" else "",
+        "peer_unresolved": peer_unresolved,
+        "association": src_assoc.status,
+        "peer_association": peer_assoc_status,
+        "observed_at": observed_at,
+        "src_ip": src_ip,
+        "orig_bytes": raw.get("orig_bytes"),
+        "server_name": raw.get("server_name"),
+        "dns_query": raw.get("dns_query") or raw.get("query"),
+        "assoc_reason": src_assoc.reason,
+        "peer_assoc_reason": peer_assoc.reason if peer_assoc else None,
+    }
+

+def _refresh_row(row: EgressObservation, cand: dict[str, Any]) -> None:
+    observed_at = cand["observed_at"]
+    row.last_seen = max(row.last_seen or observed_at, observed_at)
+    row.observation_count = int(row.observation_count or 0) + 1
+    if cand.get("subject_type") and not row.subject_type:
+        row.subject_type = cand["subject_type"]
+        row.subject_id = cand["subject_id"]
+    if cand.get("asset_id") and not row.asset_id:
+        row.asset_id = cand["asset_id"]
+    if cand.get("scope") == "int" and cand.get("peer_unresolved") is False:
+        if cand.get("peer_subject_type") and (not row.peer_subject_type or row.peer_unresolved):
+            row.peer_subject_type = cand["peer_subject_type"]
+            row.peer_subject_id = cand["peer_subject_id"]
+            row.peer_unresolved = False
+    ev = dict(row.evidence or {})
+    for k in ("server_name", "dns_query", "orig_bytes", "src_ip", "assoc_reason"):
+        if cand.get(k) is not None:
+            ev[k] = cand[k]
+    ev["o13c_marker"] = O13C_MARKER
+    row.evidence = ev

+def ingest_egress_batch(
+    db: Session,
+    items: list[dict[str, Any]],
+    settings: Any | None = None,
+) -> dict[str, Any]:
+    """Ingest hybrid egress. Cap defers creates; breaker skips creates; refresh always OK.
+
+    Cap order when budget bites: ALL ext first (FIFO by first observation in this
+    cycle), then int. Deferred is not loss — the same tuple reappears next cycle
+    (starvation under cap is temporary, not dropped).
+    """
+    enabled = True
+    if settings is not None:
+        enabled = bool(getattr(settings, "egress_ingest_enabled", False))
+    if not enabled:
+        return {
+            "received": len(items),
+            "seen_distinct": 0,
+            "created": 0,
+            "refreshed": 0,
+            "deferred": 0,
+            "skipped_disabled": len(items),
+            "skipped_invalid": 0,
+            "skipped_breaker": 0,
+            "breaker_open": False,
+            "cap_active": False,
+            "baseline": None,
+            "table": table_stats(db) if items else {"rows_total": 0, "rows_today": 0, "approx_bytes": 0},
+            "cycle_seen": 0,
+            "cycle_created": 0,
+            "cycle_deferred": 0,
+            "o13c_marker": O13C_MARKER,
+        }
+
+    ensure_baseline_meta(db)
+
+    candidates: list[dict[str, Any]] = []
+    skipped_invalid = 0
+    for raw in items:
+        if not isinstance(raw, dict):
+            skipped_invalid += 1
+            continue
+        cand = prepare_candidate(db, raw)
+        if cand is None:
+            skipped_invalid += 1
+            continue
+        candidates.append(cand)
+
+    # Dedupe within cycle by dedup_key, keep earliest observed_at (FIFO).
+    by_key: dict[str, dict[str, Any]] = {}
+    for cand in candidates:
+        key = cand["dedup_key"]
+        prev = by_key.get(key)
+        if prev is None or cand["observed_at"] < prev["observed_at"]:
+            by_key[key] = cand
+    distinct = list(by_key.values())
+
+    # Cap order: ALL ext first (FIFO), then int (FIFO).
+    ext = sorted(
+        [c for c in distinct if c["scope"] == "ext"],
+        key=lambda c: c["observed_at"],
+    )
+    internal = sorted(
+        [c for c in distinct if c["scope"] == "int"],
+        key=lambda c: c["observed_at"],
+    )
+    ordered = ext + internal
+
+    creates_today = daily_creates_today(db)
+    created = 0
+    refreshed = 0
+    deferred = 0
+    skipped_breaker = 0
+    cap_active = False
+
+    for cand in ordered:
+        existing = db.scalar(
+            select(EgressObservation).where(EgressObservation.dedup_key == cand["dedup_key"])
+        )
+        if existing:
+            # Refresh under cap is always allowed; breaker emergency may still block
+            # when total/bytes ceilings are hit (checked below for creates; refresh
+            # of existing rows does not increase row count).
+            _refresh_row(existing, cand)
+            refreshed += 1
+            continue
+
+        # Cap governor FIRST: defer create (not drop). Cap ≠ breaker.
+        # When rows_today already == CAP, remaining candidates are deferred —
+        # not skipped_breaker — so the cycle difference stays visible (N3).
+        if creates_today + created >= DAILY_CREATE_CAP:
+            deferred += 1
+            cap_active = True
+            continue
+
+        open_, reason = evaluate_egress_breaker(db)
+        if open_:
+            log.error("egress breaker open: %s", reason)
+            skipped_breaker += 1
+            continue
+
+        ev = {
+            "src_ip": cand.get("src_ip"),
+            "assoc_reason": cand.get("assoc_reason"),
+            "peer_assoc_reason": cand.get("peer_assoc_reason"),
+            "server_name": cand.get("server_name"),
+            "dns_query": cand.get("dns_query"),
+            "orig_bytes": cand.get("orig_bytes"),
+            "o13c_marker": O13C_MARKER,
+        }
+        row = EgressObservation(
+            dedup_key=cand["dedup_key"],
+            scope=cand["scope"],
+            subject_type=cand["subject_type"],
+            subject_id=cand["subject_id"],
+            asset_id=cand["asset_id"],
+            dst_ip=cand["dst_ip"],
+            dport=cand["dport"],
+            proto=cand["proto"],
+            peer_subject_type=cand["peer_subject_type"],
+            peer_subject_id=cand["peer_subject_id"],
+            peer_ip=cand["peer_ip"],
+            peer_unresolved=bool(cand["peer_unresolved"]),
+            association=cand["association"],
+            peer_association=cand["peer_association"] or "",
+            sensor_id=SENSOR_ID,
+            first_seen=cand["observed_at"],
+            last_seen=cand["observed_at"],
+            observation_count=1,
+            evidence={k: v for k, v in ev.items() if v is not None},
+        )
+        db.add(row)
+        db.flush()
+        created += 1
+
+    open_, reason = evaluate_egress_breaker(db)
+    bl = maybe_close_baseline_after_cycle(
+        db,
+        deferred,
+        cycle_stats={
+            "seen_distinct": len(distinct),
+            "created": created,
+            "deferred": deferred,
+        },
+    )
+    st = table_stats(db)
+    return {
+        "received": len(items),
+        "seen_distinct": len(distinct),
+        "created": created,
+        "refreshed": refreshed,
+        "deferred": deferred,
+        "skipped_invalid": skipped_invalid,
+        "skipped_breaker": skipped_breaker,
+        "skipped_disabled": 0,
+        "breaker_open": open_,
+        "breaker_reason": reason,
+        "cap_active": cap_active or deferred > 0,
+        "baseline": bl,
+        "table": st,
+        "cycle_seen": len(distinct),
+        "cycle_created": created,
+        "cycle_deferred": deferred,
+        "o13c_marker": O13C_MARKER,
+    }

+def freshness_for_observed_at(observed_at: datetime | None, *, now: datetime | None = None) -> dict[str, Any]:
+    now = now or utcnow()
+    if observed_at is None:
+        return {
+            "state": "not_measured",
+            "display": "non misurato",
+            "source": SENSOR_ID,
+            "age_hours": None,
+            "window_hours": FDB_STALE_HOURS,
+        }
+    age_h = max(0.0, (now - observed_at).total_seconds() / 3600.0)
+    if age_h > FDB_STALE_HOURS:
+        return {
+            "state": "stale",
+            "display": "vecchio",
+            "source": SENSOR_ID,
+            "age_hours": round(age_h, 3),
+            "window_hours": FDB_STALE_HOURS,
+            "last_at": observed_at.isoformat() + "Z",
+        }
+    return {
+        "state": "fresh",
+        "display": "fresco",
+        "source": SENSOR_ID,
+        "age_hours": round(age_h, 3),
+        "window_hours": FDB_STALE_HOURS,
+        "last_at": observed_at.isoformat() + "Z",
+    }

+def build_egress_for_asset(db: Session, asset: Asset, *, now: datetime | None = None) -> dict[str, Any]:
+    """Dossier helper: external destinations + internal relations for an asset/chassis."""

+/**
+ * O12-FIX K3 — legenda e livelli usati devono coincidere (P1–P7, nessun livello fantasma).
+ */
+import test from "node:test";
+import assert from "node:assert/strict";
+import {
+  DEFENSE_PRIORITY_LEGEND,
+  defensePriority,
+  priorityForCoverageCard,
+  priorityForEgressCard,
+} from "./oggiPriority.js";
+
+test("legenda dichiara P1–P7 senza buchi", () => {
+  const levels = DEFENSE_PRIORITY_LEGEND.map((x) => x.level);
+  assert.deepEqual(levels, [1, 2, 3, 4, 5, 6, 7]);
+  assert.equal(DEFENSE_PRIORITY_LEGEND.length, 7);
+  assert.ok(DEFENSE_PRIORITY_LEGEND.some((x) => x.id === "igiene_nome" && x.level === 7));
+});
+
+test("coverage cieca → P4; disabilitata → P7 igiene_nome", () => {
+  const blind = priorityForCoverageCard({
+    kind: "coverage_source_blind",
+    defense_priority_id: "sorgente_non_disponibile",
+  });
+  assert.equal(blind.level, 4);
+  assert.ok(DEFENSE_PRIORITY_LEGEND.some((L) => L.level === blind.level));
+
+  const disabled = priorityForCoverageCard({
+    kind: "coverage_source_disabled",
+    defense_priority_id: "igiene_nome",
+  });
+  assert.equal(disabled.level, 7);
+  assert.equal(disabled.id, "igiene_nome");
+  assert.ok(DEFENSE_PRIORITY_LEGEND.some((L) => L.id === disabled.id && L.level === 7));
+});
+
+test("defensePriority non inventa livelli fuori legenda", () => {
+  const p = defensePriority("igiene_nome");
+  assert.equal(p.level, 7);
+  const unknown = defensePriority("id_inesistente_xyz");
+  assert.equal(unknown.id, "igiene_nome");
+  assert.equal(unknown.level, 7);
+});
+
+test("O13C egress ext e int → stesso P6 ignoto_con_evidenze_zeek", () => {
+  const ext = priorityForEgressCard({
+    kind: "egress_new_ext",
+    defense_priority_id: "ignoto_con_evidenze_zeek",
+  });
+  const intCard = priorityForEgressCard({

+    defense_priority_id: "ignoto_con_evidenze_zeek",

+def _parse_ts(value: Any) -> datetime | None:
+    if value is None:
+        return None
+    if isinstance(value, datetime):
+        return value.replace(tzinfo=None) if value.tzinfo else value
+    text = str(value).strip()
+    if not text:
+        return None
+    try:
+        if text.endswith("Z"):
+            text = text[:-1] + "+00:00"
+        dt = datetime.fromisoformat(text)
+        return dt.astimezone().replace(tzinfo=None) if dt.tzinfo else dt
+    except ValueError:
+        return None

+def _interval_sec(settings: Settings, attr: str | None) -> int | None:
+    if attr is None:
+        return None
+    # Collector attrs may live only on collector Settings; API Settings has a subset.
+    # Prefer API settings; fall back to known defaults declared in code (not invented).
+    if hasattr(settings, attr):
+        val = getattr(settings, attr)
+        if val is None:
+            return None
+        raw = max(1, int(val))
+    else:
+        # Defaults from collector/config.py (documented, not invented).
+        defaults = {
+            "topology_poll_interval_sec": 60,
+            "collector_interval_sec": 900,
+            "zeek_provider_interval_sec": 300,
+            "zeek_behavior_interval_sec": 300,
+            "egress_ingest_interval_sec": 300,
+            "internet_probe_interval_sec": 60,
+            "snmp_nas_poll_interval_sec": 60,
+        }
+        if attr not in defaults:
+            return None
+        raw = defaults[attr]
+    # Floor identici a collector/main.py configure_scheduler (cadenza reale, non inventata).
+    floors = {
+        "topology_poll_interval_sec": 15,
+        "collector_interval_sec": 60,
+        "internet_probe_interval_sec": 30,
+        "zeek_provider_interval_sec": 60,
+        "zeek_behavior_interval_sec": 60,
+        "egress_ingest_interval_sec": 60,
+        "snmp_nas_poll_interval_sec": 60,
+    }
+    return max(floors.get(attr, 1), raw)

+def _last_sensor_run(db: Session, source: str) -> dict[str, Any]:
+    last_ok = db.scalar(
+        select(SensorRun)
+        .where(SensorRun.source == source, SensorRun.ok.is_(True))
+        .order_by(SensorRun.started_at.desc())
+        .limit(1)
+    )
+    last_err = db.scalar(
+        select(SensorRun)
+        .where(SensorRun.source == source, SensorRun.ok.is_(False))
+        .order_by(SensorRun.started_at.desc())
+        .limit(1)
+    )
+    last_any = db.scalar(
+        select(SensorRun)
+        .where(SensorRun.source == source)
+        .order_by(SensorRun.started_at.desc())
+        .limit(1)
+    )
+    err_text = None
+    if last_err is not None:
+        err_text = (last_err.error or "").strip() or "ultimo errore non registrato"
+    elif last_any is not None and not last_any.ok:
+        err_text = (last_any.error or "").strip() or "ultimo errore non registrato"
+    return {
+        "last_success_at": last_ok.started_at if last_ok else None,
+        "last_error_at": last_err.started_at if last_err else None,
+        "last_error": err_text,
+        "last_attempt_ok": last_any.ok if last_any else None,
+        "last_attempt_at": last_any.started_at if last_any else None,
+    }

+def _last_fdb_snmp(db: Session) -> dict[str, Any]:
+    """Aggregate FDB-capable switches only; GS308EP excluded (I7)."""
+    newest_ok: datetime | None = None
+    newest_err: str | None = None
+    newest_err_at: datetime | None = None
+    supporting = 0
+    for sw in db.scalars(select(Switch)).all():
+        cap = switch_capabilities(getattr(sw, "model", None) or sw.name)
+        if cap.get("fdb_supported") is False:
+            continue
+        supporting += 1
+        meta = dict(sw.meta or {})
+        fp = dict(meta.get("fdb_poll") or {})
+        at = _parse_ts(fp.get("at"))
+        ok = fp.get("ok")
+        err = str(fp.get("error") or "")
+        if ok and at and (newest_ok is None or at > newest_ok):
+            newest_ok = at
+        if ok is False and at and (newest_err_at is None or at > newest_err_at):
+            newest_err_at = at
+            newest_err = err or "poll FDB non ok (errore assente nel meta — difetto di registro)"
+    return {
+        "last_success_at": newest_ok,
+        "last_error_at": newest_err_at,
+        "last_error": newest_err,
+        "last_attempt_ok": None if newest_ok is None and newest_err_at is None else (
+            newest_ok is not None
+            and (newest_err_at is None or newest_ok >= newest_err_at)
+        ),
+        "last_attempt_at": max(
+            [t for t in (newest_ok, newest_err_at) if t is not None],
+            default=None,
+        ),
+        "fdb_supporting_switches": supporting,
+    }

+def _last_internet(db: Session) -> dict[str, Any]:
+    mon = db.scalar(select(Monitor).where(Monitor.kind == "internet").limit(1))
+    if mon is None:
+        return {
+            "last_success_at": None,
+            "last_error_at": None,
+            "last_error": "monitor.kind=internet assente",
+            "last_attempt_ok": None,
+            "last_attempt_at": None,
+            "registry_note": "da monitor.kind=internet",
+        }
+    health = dict((mon.config or {}).get("internet_health") or {})
+    last_probe = health.get("last_probe")
+    probe_at = None
+    if isinstance(last_probe, dict):
+        probe_at = _parse_ts(
+            last_probe.get("at")
+            or last_probe.get("checked_at")
+            or last_probe.get("observed_at")
+        )
+    at = (
+        _parse_ts(health.get("checked_at") or health.get("at") or health.get("last_ok_at"))
+        or probe_at
+        or getattr(mon, "last_checked_at", None)
+    )
+    if isinstance(at, datetime):
+        at = at.replace(tzinfo=None) if at.tzinfo else at
+    state = str(health.get("state") or "").lower()
+    cause = health.get("cause")
+    ok = state in {"online", "ok", "up"} if state else None
+    if ok is True and at is not None:
+        return {
+            "last_success_at": at,
+            "last_error_at": None,
+            "last_error": None,
+            "last_attempt_ok": True,
+            "last_attempt_at": at,
+            "registry_note": "da monitor.last_checked_at / internet_health",
+        }
+    if ok is True and at is None:
+        return {
+            "last_success_at": None,
+            "last_error_at": None,
+            "last_error": "internet online senza timestamp (difetto di registro)",
+            "last_attempt_ok": True,
+            "last_attempt_at": None,
+            "registry_note": "da monitor.kind=internet config.internet_health",
+        }
+    err = f"internet_health.state={state or 'assente'}; cause={cause or 'assente'}"
+    return {
+        "last_success_at": None,
+        "last_error_at": at,
+        "last_error": err,

+    }

+def _last_nas(db: Session) -> dict[str, Any]:
+    """Ultimo poll ASUSTOR da meta.asustor_snmp su asset NAS."""
+    newest_ok: datetime | None = None
+    newest_err_at: datetime | None = None
+    newest_err: str | None = None
+    for asset in db.scalars(select(Asset)).all():
+        meta = dict(asset.meta or {})
+        snmp = dict(meta.get("asustor_snmp") or {})
+        if not snmp:
+            continue
+        at = _parse_ts(
+            snmp.get("observed_at")
+            or snmp.get("at")
+            or snmp.get("polled_at")
+            or snmp.get("updated_at")
+        )
+        err = snmp.get("poll_error") or snmp.get("error")
+        ok = snmp.get("ok")
+        if ok is False and at and (newest_err_at is None or at > newest_err_at):
+            newest_err_at = at
+            newest_err = str(err or "asustor_snmp ok=false")
+        if ok is not False and at and (newest_ok is None or at > newest_ok):
+            newest_ok = at
+    return {
+        "last_success_at": newest_ok,
+        "last_error_at": newest_err_at,
+        "last_error": newest_err
+        if newest_err
+        else (None if newest_ok else "nessun meta.asustor_snmp"),
+        "last_attempt_ok": None
+        if newest_ok is None and newest_err_at is None
+        else (
+            newest_ok is not None
+            and (newest_err_at is None or newest_ok >= newest_err_at)
+        ),
+        "last_attempt_at": max(
+            [t for t in (newest_ok, newest_err_at) if t is not None],
+            default=None,
+        ),
+        "registry_note": "da asset.meta.asustor_snmp (ok/observed_at)",
+    }

+def classify_source(
+    *,
+    enabled: bool,
+    enable_flag: str | None,
+    interval_sec: int | None,
+    last_success_at: datetime | None,
+    last_error: str | None,
+    now: datetime,
+) -> dict[str, Any]:
+    """Pure C2 classifier — used by production and C5 tests."""
+    if not enabled:
+        return {
+            "state": "disabilitata",
+            "reason": f"flag {enable_flag or '?'} = false (scelta operatore, non guasto)",
+            "age_sec": None,
+            "window_sec": interval_sec,
+        }
+    if interval_sec is None:
+        return {
+            "state": "cadenza_non_nota",
+            "reason": "cadenza non dichiarata in configurazione — nessuna finestra inventata",
+            "age_sec": None,
+            "window_sec": None,
+        }
+    if last_success_at is None:
+        return {
+            "state": "cieca",
+            "reason": "nessun successo registrato mentre la sorgente è abilitata",
+            "age_sec": None,
+            "window_sec": interval_sec,
+            "last_error": last_error,
+        }
+    age = (now - last_success_at).total_seconds()
+    if age <= interval_sec:
+        return {
+            "state": "coperta_fresca",
+            "reason": f"ultimo successo entro cadenza {interval_sec}s",
+            "age_sec": round(age, 3),
+            "window_sec": interval_sec,
+        }
+    # Oltre cadenza con data nota → coperta_vecchia (I2). Card allarme: sì (cieca per difesa).
+    return {
+        "state": "coperta_vecchia",
+        "reason": (
+            f"ultimo successo oltre cadenza {interval_sec}s "
+            f"(age={round(age, 1)}s) — data {last_success_at.isoformat()}"
+        ),
+        "age_sec": round(age, 3),
+        "window_sec": interval_sec,
+        "last_error": last_error,
+        "alarm": True,  # oltre cadenza = non coperto per difesa (C4)
+    }

+def evaluate_source(
+    db: Session,
+    spec: SourceSpec,
+    settings: Settings,
+    *,
+    now: datetime | None = None,
+    override_enabled: bool | None = None,
+    override_last_success: datetime | None = None,
+    override_last_error: str | None = None,
+    override_interval: int | None = None,
+) -> dict[str, Any]:
+    now = now or utcnow()
+    if spec.kind == "structural_skip":
+        return {
+            "name": spec.name,
+            "label": spec.label,
+            "state": "limite_strutturale",
+            "o12_marker": "obs-o12-structural",
+            "note": "I7 — GS308EP non espone FDB SNMP; non è sorgente cieca",
+            "job_or_endpoint": spec.job_or_endpoint,
+            "feeds": list(spec.feeds),
+            "signals_lost_when_blind": [],
+            "enable_flag": None,
+            "enabled": None,
+            "interval_sec": None,
+            "freshness_vocab": "not_covered",
+            "card_eligible": False,
+        }
+
+    enabled, flag = _flag_enabled(settings, spec.enable_flag)
+    if override_enabled is not None:
+        enabled = override_enabled
+    # flow family: also need zeek_provider for collector side when classifying "should run"
+    if spec.name in {"zeek_conn_flow", "zeek_intel"} and enabled:
+        # API ingest on is not enough if provider off — still "disabilitata" if provider off
+        if not bool(getattr(settings, "zeek_provider_enabled", False)):
+            enabled = False
+            flag = "zeek_provider_enabled"
+    if spec.name == "zeek_dhcp_names":
+        enabled, flag = _flag_enabled(settings, "zeek_provider_enabled")
+        if override_enabled is not None:
+            enabled = override_enabled
+
+    interval = (
+        override_interval
+        if override_interval is not None
+        else _interval_sec(settings, spec.interval_attr)
+    )
+
+    if override_last_success is not None or override_last_error is not None:
+        last = {
+            "last_success_at": override_last_success,
+            "last_error": override_last_error,
+            "last_error_at": None,
+            "last_attempt_ok": None,
+            "last_attempt_at": override_last_success,
+        }
+    elif spec.kind == "sensor_run":
+        last = _last_sensor_run(db, _SENSOR_RUN_SOURCE[spec.name])
+    elif spec.kind == "fdb_snmp":
+        last = _last_fdb_snmp(db)
+    elif spec.kind == "zeek_behavior":
+        last = _last_zeek_behavior(db)
+    elif spec.kind == "zeek_egress":
+        last = _last_zeek_egress(db)
+    elif spec.kind == "portal":
+        last = _last_portal(db)
+    elif spec.kind == "internet":
+        last = _last_internet(db)
+    elif spec.kind == "nas":
+        last = _last_nas(db)
+    elif spec.kind == "flow":
+        if spec.name == "zeek_conn_flow":
+            last = _last_flow(db)
+        elif spec.name == "zeek_intel":
+            last = _last_zeek_named_sensor(db, "zeek_intel")
+        elif spec.name == "zeek_dhcp_names":
+            last = _last_zeek_named_sensor(db, "zeek_dhcp_names")
+        else:
+            last = {
+                "last_success_at": None,
+                "last_error": "ultimo errore non registrato",
+                "last_error_at": None,
+                "last_attempt_ok": None,
+                "last_attempt_at": None,
+            }
+    else:
+        last = {
+            "last_success_at": None,
+            "last_error": "kind sconosciuto",
+            "last_error_at": None,
+            "last_attempt_ok": None,
+            "last_attempt_at": None,
+        }
+
+    # Special: flow/intel/dhcp with registry gap but disabled → disabilitata first
+    cls = classify_source(
+        enabled=enabled,
+        enable_flag=flag or spec.enable_flag,
+        interval_sec=interval,
+        last_success_at=last.get("last_success_at"),
+        last_error=last.get("last_error"),
+        now=now,
+    )
+    # Soft: coperta_vecchia is primary; alarm path uses alarm flag / state set
+    display_state = cls["state"]
+    display_extra = None
+
+    last_success = last.get("last_success_at")
+    not_covered = display_state in {"cieca", "coperta_vecchia", "disabilitata"}
+    return {
+        "name": spec.name,
+        "label": spec.label,
+        "state": display_state,
+        "display_extra": display_extra,
+        "reason": cls.get("reason"),
+        "job_or_endpoint": spec.job_or_endpoint,
+        "feeds": list(spec.feeds),
+        "signals_lost_when_blind": list(spec.signals_lost_when_blind)
+        if display_state in {"cieca", "coperta_vecchia"}
+        else (
+            list(spec.signals_lost_when_blind)
+            if display_state == "disabilitata"
+            else []
+        ),
+        "signals_not_observable": list(spec.signals_lost_when_blind) if not_covered else [],
+        "enable_flag": flag or spec.enable_flag,
+        "enabled": enabled,
+        "interval_sec": interval,
+        "interval_attr": spec.interval_attr,
+        "age_sec": cls.get("age_sec"),
+        "window_sec": cls.get("window_sec"),
+        "last_success_at": last_success.isoformat() + "Z" if last_success else None,
+        "last_error": last.get("last_error"),
+        "last_error_at": (
+            last["last_error_at"].isoformat() + "Z"
+            if last.get("last_error_at")
+            else None
+        ),
+        "registry_note": last.get("registry_note"),
+        "freshness_vocab": STATE_TO_FRESHNESS.get(display_state, "not_measured"),
+        "o12_marker": "obs-o12-source",
+        "card_eligible": spec.card_eligible,
+        "defense_priority_id": (
+            "sorgente_non_disponibile"
+            if display_state in {"cieca", "coperta_vecchia"}
+            else "igiene_nome"
+        ),
+    }

+def coverage_alarm_cards(

+def acknowledge_coverage_signal(
+    db: Session,
+    *,
+    uid: str,
+    action: str,
+    reason: str = "",
+    user_id: int | None = None,
+    kind: str = "coverage_source_blind",
+) -> Suggestion:
+    status = "approved" if action in {"acknowledge", "archive"} else "rejected"
+    existing = db.scalar(select(Suggestion).where(Suggestion.uid == uid))
+    if existing:
+        existing.status = status
+        existing.decided_at = utcnow()
+        existing.decided_by = user_id
+        existing.body = reason or existing.body
+        payload = dict(existing.payload or {})
+        payload["action"] = action
+        existing.payload = payload
+        db.flush()
+        return existing
+    s = Suggestion(
+        uid=uid,
+        kind=kind[:64],
+        source="coverage",
+        status=status,
+        title=(reason or uid)[:255],
+        body=reason or action,
+        payload={"action": action, "o12": True},
+        decided_at=utcnow(),
+        decided_by=user_id,
+    )
+    db.add(s)
+    db.flush()
+    return s
