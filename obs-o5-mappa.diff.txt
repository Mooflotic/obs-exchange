+export function plantEmptyStateLabel(plantPort) {
+  const st = plantPort?.empty_state;
+  if (st === "misurata_senza_dispositivi") {
+    return {
+      code: st,
+      label: "misurata · senza dispositivi",
+      hint: "FDB fresco, zero MAC — porta senza dispositivo (≠ non coperta).",
+    };
+  }
+  if (st === "non_coperta") {
+    return {
+      code: st,
+      label: "non coperta · FDB non supportato",
+      hint: "Limite strutturale dell'apparato (I7) — nessuna telemetria inventata.",
+    };
+  }
+  if (st === "sorgente_non_fresca") {
+    return {
+      code: st,
+      label: "sorgente non fresca",
+      hint: plantPort?.poll_error
+        ? `sorgente_non_disponibile — ${plantPort.poll_error}`
+        : "sorgente_non_disponibile (≠ dispositivo_assente).",
+    };
+  }
+  return null;
+}

+/** @param {string} quality */
+export function baselineQualityLabel(quality) {
+  if (quality === "verified") return "baseline verificata";
+  if (quality === "reconstructed") {
+    return "baseline ricostruita — storia precedente non disponibile";
+  }
+  return "baseline assente";
+}

+/**
+ * @param {object[]} plantPorts
+ * @param {string} q
+ */
+export function filterPlantPortsByQuery(plantPorts, q) {
+  const needle = String(q || "")
+    .trim()
+    .toLowerCase();
+  if (!needle) return plantPorts || [];
+  return (plantPorts || []).filter((p) => {
+    const hay = [
+      p.switch_code,
+      p.switch_name,
+      String(p.port_number),
+      ...(p.macs || []).flatMap((m) => [m.mac, m.name_label, m.oui, m.name]),
+    ]
+      .filter(Boolean)
+      .join(" ")
+      .toLowerCase();
+    return hay.includes(needle);
+  });
+}
+
+/**
+ * @param {object} port
+ * @param {{ soloL2?: boolean, randomized?: boolean, reconstructed?: boolean, stale?: boolean }} filters
+ */
+export function plantPortMatchesFilters(port, filters = {}) {
+  const macs = port.macs || [];
+  if (filters.soloL2 && !macs.some((m) => m.solo_l2)) return false;
+  if (filters.randomized && !macs.some((m) => m.randomized)) return false;
+  if (filters.reconstructed && port.baseline_quality !== "reconstructed") return false;
+  if (filters.stale && port.freshness !== "sorgente_non_fresca" && port.empty_state !== "sorgente_non_fresca") {
+    return false;
+  }
+  return true;
+}

+function fdbBanner(sw) {
+  const capabilities = sw.capabilities || {};
+  if (capabilities.fdb_supported === false) {
+    return {
+      kind: "muted",
+      text: capabilities.fdb_message || "Rilevamento automatico non disponibile",
+    };
+  }
+  const poll = sw.fdb_poll || {};
+  if (!sw.ip) return { kind: "warn", text: "Nessun IP switch: rilevamento non disponibile" };
+  if (!sw.snmp_configured && !poll.at) {
+    return {
+      kind: "warn",
+      text: "Rilevamento automatico non configurato",
+    };
+  }
+  if (poll.at && poll.ok === false) {
+    return {
+      kind: "warn",
+      text: `Rilevamento non riuscito: ${poll.error || "switch non raggiungibile"}`,
+    };
+  }
+  if (poll.at && poll.ok) {
+    const when = fmtWhen(poll.at);
+    return {
+      kind: "ok",
+      text: `Rilevamento attivo · ${poll.entries || 0} device su ${poll.ports_touched || 0} porte · ${when}`,
+    };
+  }
+  return { kind: "muted", text: "In attesa del primo rilevamento automatico" };

+/**
+ * Badge freschezza a livello switch (F-13).
+ * Un poll FALLITO (`fdb_poll.ok===false`) NON è freschezza: in quel caso la vera
+ * data dei dati di porta è l'ultima mappatura FDB riuscita (`port.last_fdb_at`),
+ * marcata «non aggiornata» se oltre la finestra. Mai spacciare l'ora del tentativo.
+ */
+function switchFreshness(sw) {
+  const poll = sw.fdb_poll || {};
+  if (poll.at && poll.ok) return { text: `dati al ${fmtWhen(poll.at)}`, kind: "ok" };
+  const cov = fdbCoverageStatus([sw]);
+  if (cov && cov.newestAt) {
+    return {
+      text: `mappa porte da FDB del ${fmtWhen(cov.newestAt)}${cov.stale ? " — non aggiornata" : ""}`,
+      kind: cov.stale ? "warn" : "muted",
+    };
+  }
+  const manualAt = sw.field_sources?.updated_at || sw.meta?.manual_edit_at;
+  if (manualAt) return { text: `dati al ${fmtWhen(manualAt)} (manuale)`, kind: "muted" };
+  if (sw.capabilities?.fdb_supported === false) {
+    return { text: "dati da mappa manuale", kind: "muted" };
+  }
+  return { text: "dati non ancora disponibili", kind: "muted" };
+}

+function traffic(value) {
