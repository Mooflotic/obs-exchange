+def _seed(db):
+    sw328 = Switch(code="328c", name="328C", model="Linksys LGS328C", port_count=28, ip="192.168.1.10")
+    sw310 = Switch(code="310c", name="310C", model="Linksys LGS310C", port_count=10, ip="192.168.1.11")

+    p24 = SwitchPort(switch_id=sw328.id, number=24)  # attestation — no managed uplink

+    # management identity of 308

+    db.flush()
+    iface308 = Interface(asset_id=asset308.id, mac=MAC_308)

+def test_o6_t3_manual_assign_reversible_conflict(db):
+    s = _seed(db)
+    apply_fdb_observation(
+        db, switch_code="328c", port_number=24, macs=[MAC_308, MAC_BEHIND]
+    )
+    db.commit()
+    port = s["ports_308"][1]  # number 2
+    r = assign_device_to_308_port(db, port_id=port.id, mac=MAC_BEHIND)
+    db.commit()
+    assert r["ok"] is True
+    cur = current_assignment_for_port(db, port.id)
+    assert cur is not None
+    assert cur["source"] == "manual"
+    assert cur["value_norm"] == MAC_BEHIND
+    view = build_gs308_view(db)
+    p2 = next(p for p in view["ports"] if p["port_id"] == port.id)
+    assert p2["state"] == "assegnata"
+    assert p2["source_label"] == "manual"
+    # reverse
+    r2 = assign_device_to_308_port(db, port_id=port.id, clear=True)
+    db.commit()
+    assert r2["ok"] is True
+    view2 = build_gs308_view(db)
+    p2b = next(p for p in view2["ports"] if p["port_id"] == port.id)
+    assert p2b["state"] == "sconosciuta"
+    # re-assign then diverge on second port → I3
+    assign_device_to_308_port(db, port_id=port.id, mac=MAC_BEHIND)
+    db.commit()
+    other = s["ports_308"][2]
+    assign_device_to_308_port(db, port_id=other.id, mac=MAC_BEHIND)
+    db.commit()
+    view3 = build_gs308_view(db)
+    assert any(c["kind"] == "i3_divergence" for c in view3["conflicts"])
+    # weaker proposal never wrote — manual rows still current
+    assert current_assignment_for_port(db, port.id)["value_norm"] == MAC_BEHIND
