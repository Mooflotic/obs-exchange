+/**

+ *
+ * @returns {{
+ *   state: 'misurata_fresca'|'misurata_vecchia'|'non_coperta',

+ *   last_fdb_at: string|null,

+ *   port_number: number|null,

+  const ms = _utcMs(last);
+  const switchId = sw.id ?? sw.switch_id ?? null;

+      reason: "FDB non supportato dall'apparato",
+      last_fdb_at: last,
+      age_hours: ms == null ? null : Math.max(0, (nowMs - ms) / 3_600_000),
+      switch_id: switchId,
+      port_id: portId,

+    };

+export function portCellPresentation(port, opts = {}) {
+  const role = port?.role || "unknown";
+  const fdb = opts.fdb || port?.fdb_coverage || null;
+  if (role === "span") {
+    return { primary: "Port mirror", secondary: "", free: false, span: true };
+  }
+  if (port?.neighbor_name) {
+    return { primary: port.neighbor_name, secondary: "", free: false };
+  }
+  if (port?.asset_name) {
+    return { primary: port.asset_name, secondary: "", free: false };
+  }
+  if (role === "unknown" && port?.observed_macs?.length) {
+    return { primary: "Device non identificato", secondary: "", free: false };
+  }
+  // O2/B2: uncovered FDB → «non misurato», never «nessun device» / empty as absence.
+  if (fdb?.state === "non_coperta") {
+    return {
+      primary: "non misurato",
+      secondary: fdb.reason || "copertura FDB assente",
+      free: false,
+      unmeasured: true,
+    };
+  }
+  if (role === "free") {
+    if (fdb?.state === "misurata_vecchia") {
+      return {
+        primary: "nessun endpoint (mappa non aggiornata)",
+        secondary: fdb.last_fdb_at ? `FDB ${fdb.last_fdb_at}` : "",
+        free: true,
+        stale_fdb: true,
+      };
+    }
+    return { primary: "nessun endpoint osservato", secondary: "", free: true };
+  }
+  if (fdb?.state === "misurata_vecchia") {
+    return {
+      primary: "mappa non aggiornata",
+      secondary: fdb.last_fdb_at ? `ultima FDB ${fdb.last_fdb_at}` : "",
+      free: false,
+      stale_fdb: true,
+    };
+  }
+  return { primary: "nessun endpoint osservato", secondary: "", free: false };
+}
+
+export function portRoleTooltip(port) {
+  if (port?.role === "span") {
+    return [
+      port?.port_path_label,
+      "SPAN",
+      "destinazione port-mirror — no forwarding, MAC eth1 assente by design",
+      [port?.role_source || "manual", "100%"].join(" · "),
+    ]
+      .filter(Boolean)
+      .join("\n");
+  }
+  const role = rolePresentation(port).label || "Ruolo non identificato";
+  const peer = [port?.neighbor_name, port?.peer_port_label].filter(Boolean).join(" · ");
+  const confidence = Number.isFinite(Number(port?.role_confidence))
+    ? `${Math.round(Number(port.role_confidence) * 100)}%`
+    : "n/d";
+  const evidence = [port?.role_source || "fonte non disponibile", confidence].join(" · ");
+  return [port?.port_path_label, role, peer, evidence].filter(Boolean).join("\n");
+}

+export function explainNoiseRule(current, proposed) {
+  const cur = String(current || "").trim();
+  const prop = String(proposed || "").trim();
+  if (!prop) return "Proposta vuota.";
+  if (cur && normalizeName(cur) === normalizeName(prop)) {
+    return `Stesso nome già adottato (parità normalizzata): «${prop}».`;
+  }
+  if (scoreSpecificity(prop) === 1) {
+    return `Nome sintetico o MAC/hex (rank 1): «${prop}» non è un'etichetta umana.`;
+  }
+  if (scoreSpecificity(prop) <= scoreSpecificity(cur)) {
+    return `La proposta «${prop}» non è più specifica del nome attuale «${cur || "anonimo"}».`;
+  }
+  return `Classificata come rumore dalle regole di specificità: «${prop}».`;

+        name: cur || "(anonimo)",

+/**
+ * Normalize Suggestion(kind=move) into Oggi problem card shape.
+ * Stale FDB evidence → media priority with age declared (O2/B3 / F-9).
+ *
+ * @param {object} sug
+ * @param {{ last_fdb_at?: string|null, switch_name?: string, port_label?: string, nowMs?: number }} [ctx]
+ */
+export function moveSuggestionFields(sug, ctx = {}) {
+  const payload = sug.payload || {};
+  const title = sug.title || "Sposta porta";
+  const lastFdb =
+    ctx.last_fdb_at ||
+    payload.last_fdb_at ||
+    payload.port_last_fdb_at ||
+    null;
+  const nowMs = ctx.nowMs != null ? Number(ctx.nowMs) : Date.now();
+  let ageHours = null;
+  if (lastFdb) {
+    let s = String(lastFdb);
+    if (!/[zZ]|[+-]\d\d:?\d\d$/.test(s)) s += "Z";
+    const ms = Date.parse(s);
+    if (Number.isFinite(ms)) ageHours = Math.max(0, (nowMs - ms) / 3_600_000);
+  }
+  const stale =
+    ageHours == null || ageHours > FDB_STALE_HOURS;
+  const switchName = ctx.switch_name || payload.switch_name || payload.switch || "?";
+  const portLabel =
+    ctx.port_label ||
+    payload.to_port_label ||
+    payload.from_port_label ||
+    title;
+  const ageBit =
+    ageHours == null
+      ? "età FDB non disponibile (sorgente non misurata)"
+      : `età FDB ≈ ${Math.floor(ageHours)} h (finestra ${FDB_STALE_HOURS} h)`;
+  const provenance = `switch ${switchName} · porta ${portLabel} · ${ageBit}`;
+
+  return {
+    kind: "move",
+    key: `move-${sug.id}`,
+    suggestion_id: sug.id,
+    asset_id: sug.asset_id,
+    title,
+    body: sug.body || "",
+    confidence: sug.confidence,
+    // Stale / unknown age: not high priority (F-9: alta only with honest actionable evidence).
+    priority: stale ? "media" : "alta",
+    fdb_stale: stale,
+    fdb_age_hours: ageHours,
+    cause:
+      (sug.body || `Rilevamento FDB diverso dal bind manuale (${title}).`) +
+      ` Provenienza: ${provenance}.`,
+    impact: stale
+      ? "La mappa FDB non è fresca: conferma solo dopo verifica fisica / poll ripristinato."
+      : "Se confermi, l'associazione porta manuale viene aggiornata.",
+    action: "Conferma lo spostamento oppure Rifiuta (resta il bind attuale).",
+    certainty: stale
+      ? `evidenza FDB non corrente · ${ageBit} · confidenza ${Math.round(Number(sug.confidence || 0) * 100)}%`
+      : `evidenza FDB · confidenza ${Math.round(Number(sug.confidence || 0) * 100)}%`,
+    outcome: "Porta aggiornata (source=suggestion) oppure proposta archiviata.",
+    from_port: payload.from_port_label || "",
+    to_port: payload.to_port_label || title.replace(/^Sposta su\s+/i, ""),
+    mac: payload.mac || "",
+  };
+}

+  if (n >= 1_000) return `${(n / 1_000).toFixed(1)} Kb/s`;
+  return `${Math.round(n)} b/s`;
+}

+function fmtWhen(iso) {
+  return formatDateTime(iso);
+}
+
+function sourceBadge(port) {
+  return port.role_source || "";
+}
+
+function deviceName(port, sw) {
+  return portCellPresentation(port, { fdb: fdbPortCoverageState(port, sw) }).primary;
+}
+
+function deviceHidden(port) {
+  return !!port.asset_id && port.asset_visible_by_default === false && !showUnreachable.value;
+}
+
+function displayedDeviceName(port, sw) {
+  return deviceHidden(port) ? "Storico nascosto" : deviceName(port, sw);
+}
+
+function displayedDeviceParts(port, sw) {
+  if (deviceHidden(port)) return { primary: "storico nascosto", secondary: "" };
+  return portCellPresentation(port, { fdb: fdbPortCoverageState(port, sw) });
+}
+
+function portFdbBadge(port, sw) {
+  const st = fdbPortCoverageState(port, sw);
+  if (st.state === "misurata_fresca") {
+    return { text: "FDB fresca", kind: "ok", title: `ultima mappatura ${fmtWhen(st.last_fdb_at)}` };
+  }
+  if (st.state === "misurata_vecchia") {
+    return {
+      text: `FDB ${fmtWhen(st.last_fdb_at)}`,
+      kind: "warn",
+      title: `misurata_vecchia · ≈${Math.floor(st.age_hours || 0)} h · ${coverageWindowLabel.value}`,
+    };
+  }
+  return {
+    text: st.reason || "non coperta",
+    kind: "muted",
+    title: `non_coperta · ${st.reason}`,
+  };
+}
+
+function portTooltip(port) {
+  const evidence = port.asset_visibility_reason
+    ? `Mantenuto: ${port.asset_visibility_reason}`
+    : port.asset_hidden_reason
+      ? `Device storico nascosto: ${port.asset_hidden_reason}`
+      : "Nessuna associazione attiva";
+  return `${portRoleTooltip(port)}\n${evidence}`;
+}
+
+function portGridStyle(sw) {
+  const count = sw.ports?.length || 1;
+  return {
+    "--port-columns": Math.min(count, 14),
+    "--port-columns-medium": Math.min(count, 7),
+  };
+}
+
+function macOui(port) {
+  const mac = port.asset_mac || port.observed_macs?.[0] || "";
+  if (!mac) return "";
+  return port.asset_vendor ? `${mac} · ${port.asset_vendor}` : mac;
+}

+function statusLabel(value) {
+  return {
+    up: "attivo",
+    down: "non attivo",
+    lowerLayerDown: "non attivo",
+    notPresent: "non presente",
+    testing: "test",
+    dormant: "in attesa",
+  }[value] || "sconosciuto";
+}

+    error.value = e.message || String(e);

+  }

+    busyKey.value = "";
+  }

+  if (s === "recent") return "recente";
+  if (s === "fritz_only") return "solo Fritz";

+    return fritzOutage.value?.active
+      ? "copertura sorgente non disponibile"
+      : "non visto di recente";
+  }
+  return s;
