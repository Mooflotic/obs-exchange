"""W2 shadow writers — parallel fact_assertion writes; no consumers read yet.

S1: never degrade the observed ingest path. Kill switch + circuit breaker
stop writes without touching Asset / reconcile fields.

Writes use a SAVEPOINT on the ingest session so (1) freshly flushed interfaces
are visible and (2) assertion rollback never rolls back the observation.
"""

from __future__ import annotations

import logging
import threading
from collections import defaultdict
from datetime import date, datetime
from pathlib import Path
from typing import Any

from sqlalchemy import func, select, text
from sqlalchemy.orm import Session

from app.config import get_settings
from app.facts.resolver import apply_observation
from app.models import FactAssertion, Interface, utcnow

log = logging.getLogger("observatory.facts.shadow")

# Circuit breaker ceilings (W2.3.3) — do not raise to pass an assert.
MAX_FACT_ROWS = 20_000
MAX_GROWTH_PER_DAY = 2_000
MAX_TABLE_BYTES = 50 * 1024 * 1024

FACT_IFACE_IP = "asset.iface_ip"

_lock = threading.Lock()
_counters: dict[tuple[str, str], dict[str, int]] = defaultdict(
    lambda: {
        "refresh": 0,
        "supersession": 0,
        "weak_evidence": 0,
        "conflict_review": 0,
        "created": 0,
        "cooldown": 0,
        "errors": 0,
        "skipped_breaker": 0,
        "skipped_disabled": 0,
    }
)
_breaker_open = False
_breaker_reason = ""
_breaker_opened_at: datetime | None = None
_day_anchor: date | None = None
_day_start_rows = 0
_error_logged = False

# Source aliases from ingest kinds → registry-allowed labels.
_SOURCE_MAP: dict[str, str] = {
    "fritz": "fritz",
    "dhcp": "dhcp",
    "nmap": "nmap",
    "mgmt": "mgmt",
    "manual": "manual",
    "scan": "nmap",
    "arp": "nmap",
    "icmp": "nmap",
    "mdns": "nmap",
    "ssdp": "nmap",
    "portal": "fritz",
}


def _norm_source(source: str) -> str:
    key = str(source or "").strip().lower()
    return _SOURCE_MAP.get(key, "nmap")


def _bump(fact_key: str, source: str, action: str) -> None:
    with _lock:
        bucket = _counters[(fact_key, source)]
        if action == "superseded":
            bucket["supersession"] += 1
        elif action in bucket:
            bucket[action] += 1
        else:
            bucket.setdefault(action, 0)
            bucket[action] += 1


def reset_shadow_stats_for_tests() -> None:
    """Test helper — clears in-memory counters / breaker."""
    global _breaker_open, _breaker_reason, _breaker_opened_at
    global _day_anchor, _day_start_rows, _error_logged
    with _lock:
        _counters.clear()
        _breaker_open = False
        _breaker_reason = ""
        _breaker_opened_at = None
        _day_anchor = None
        _day_start_rows = 0
        _error_logged = False


def force_breaker_open_for_tests(reason: str = "test") -> None:
    global _breaker_open, _breaker_reason, _breaker_opened_at
    with _lock:
        _breaker_open = True
        _breaker_reason = reason
        _breaker_opened_at = utcnow()


def _table_bytes(db: Session) -> int:
    try:
        row = db.execute(
            text(
                "SELECT IFNULL(SUM(pgsize), 0) FROM dbstat WHERE name IN "
                "('fact_assertions', 'uq_fact_assertions_current_slot', "
                "'ix_fact_assertions_subject_state', 'ix_fact_assertions_fact_state')"
            )
        ).scalar()
        if row is not None:
            return int(row)
    except Exception:  # noqa: BLE001 — dbstat optional
        pass
    settings = get_settings()
    path = Path(settings.sqlite_path)
    try:
        return int(path.stat().st_size) if path.exists() else 0
    except OSError:
        return 0


def _row_count(db: Session) -> int:
    return int(db.scalar(select(func.count()).select_from(FactAssertion)) or 0)


def _top_transition_groups(db: Session, limit: int = 20) -> list[dict[str, Any]]:
    rows = db.execute(
        text(
            """
            SELECT fact_key, subject_id, COUNT(*) AS n
            FROM fact_assertions
            GROUP BY fact_key, subject_id
            ORDER BY n DESC
            LIMIT :lim
            """
        ),
        {"lim": limit},
    ).mappings().all()
    return [dict(r) for r in rows]


def evaluate_circuit_breaker(db: Session) -> tuple[bool, str]:
    """Return (open, reason). Opens latch once; stays open until process restart."""
    global _breaker_open, _breaker_reason, _breaker_opened_at
    global _day_anchor, _day_start_rows

    if _breaker_open:
        return True, _breaker_reason

    today = utcnow().date()
    rows = _row_count(db)
    if _day_anchor != today:
        _day_anchor = today
        _day_start_rows = rows
    growth = rows - _day_start_rows
    size = _table_bytes(db)

    reason = ""
    if rows >= MAX_FACT_ROWS:
        reason = f"rows>={MAX_FACT_ROWS} (have {rows})"
    elif growth >= MAX_GROWTH_PER_DAY:
        reason = f"growth>={MAX_GROWTH_PER_DAY}/day (have {growth})"
    elif size >= MAX_TABLE_BYTES:
        reason = f"table_bytes>={MAX_TABLE_BYTES} (have {size})"

    if reason:
        with _lock:
            _breaker_open = True
            _breaker_reason = reason
            _breaker_opened_at = utcnow()
        log.error(
            "fact_shadow_circuit_breaker_open reason=%s top_groups=%s",
            reason,
            _top_transition_groups(db),
        )
        return True, reason
    return False, ""


def shadow_stats() -> dict[str, Any]:
    with _lock:
        by_key_source = {
            f"{fk}|{src}": dict(vals) for (fk, src), vals in _counters.items()
        }
        return {
            "enabled": bool(get_settings().fact_shadow_writers_enabled),
            "breaker_open": _breaker_open,
            "breaker_reason": _breaker_reason,
            "breaker_opened_at": (
                _breaker_opened_at.isoformat() if _breaker_opened_at else None
            ),
            "ceilings": {
                "max_rows": MAX_FACT_ROWS,
                "max_growth_per_day": MAX_GROWTH_PER_DAY,
                "max_table_bytes": MAX_TABLE_BYTES,
            },
            "counters": by_key_source,
        }


def safe_shadow_iface_ip(
    db: Session,
    iface: Interface,
    *,
    ip: str,
    source: str,
    observed_at: datetime | None = None,
    confidence: float | None = None,
) -> None:
    """Apply asset.iface_ip inside a SAVEPOINT; never raise to caller (S1)."""
    global _error_logged

    settings = get_settings()
    src = _norm_source(source)
    if not settings.fact_shadow_writers_enabled:
        _bump(FACT_IFACE_IP, src, "skipped_disabled")
        return

    ip_n = (ip or "").strip()
    if not ip_n or iface is None or not getattr(iface, "id", None):
        return

    try:
        open_, _reason = evaluate_circuit_breaker(db)
        if open_:
            _bump(FACT_IFACE_IP, src, "skipped_breaker")
            return

        with db.begin_nested():
            result = apply_observation(
                db,
                fact_key=FACT_IFACE_IP,
                entity=iface,
                value=ip_n,
                source=src,
                observed_at=observed_at,
                confidence=confidence,
            )

        action = result.action
        if action == "superseded":
            _bump(FACT_IFACE_IP, src, "supersession")
        elif action == "refresh":
            _bump(FACT_IFACE_IP, src, "refresh")
        elif action == "weak_evidence":
            _bump(FACT_IFACE_IP, src, "weak_evidence")
        elif action == "conflict_review":
            _bump(FACT_IFACE_IP, src, "conflict_review")
        elif action == "created":
            _bump(FACT_IFACE_IP, src, "created")
        elif action == "cooldown":
            _bump(FACT_IFACE_IP, src, "cooldown")

        evaluate_circuit_breaker(db)
    except Exception as exc:  # noqa: BLE001 — S1 barrier
        _bump(FACT_IFACE_IP, src, "errors")
        if not _error_logged:
            _error_logged = True
            log.exception(
                "fact_shadow_apply_isolated fact_key=%s interface_id=%s err=%s",
                FACT_IFACE_IP,
                getattr(iface, "id", None),
                exc,
            )
        else:
            log.warning(
                "fact_shadow_apply_isolated fact_key=%s interface_id=%s err=%s",
                FACT_IFACE_IP,
                getattr(iface, "id", None),
                exc,
            )

"""Single resolver for fact currency — R-A..R-H live here only."""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Any

from sqlalchemy import and_, select
from sqlalchemy.orm import Session

from app.facts.registry import (
    authority_for,
    default_confidence,
    get_fact_spec,
    semantic_contradiction,
)
from app.models import Asset, FactAssertion, Interface, utcnow

READMISSION_COOLDOWN = timedelta(hours=4)

# R-C observability (W2 will expose per fact_key/source).
weak_evidence_count = 0
conflict_review_count = 0


@dataclass
class ApplyResult:
    action: str
    assertion: FactAssertion | None = None
    incumbent: FactAssertion | None = None
    conflict: dict[str, Any] | None = None


def _norm_value(value: Any) -> str:
    return " ".join(str(value or "").strip().split())


def subject_of(fact_key: str, entity: Any) -> tuple[str, int]:
    spec = get_fact_spec(fact_key)
    return spec.subject_ref(entity)


def _base_query(
    subject_type: str,
    subject_id: int,
    fact_key: str | None = None,
    *,
    state: str | None = None,
    excl_key: str | None = None,
):
    clauses = [
        FactAssertion.subject_type == subject_type,
        FactAssertion.subject_id == subject_id,
    ]
    if fact_key is not None:
        clauses.append(FactAssertion.fact_key == fact_key)
    if state is not None:
        clauses.append(FactAssertion.state == state)
    if excl_key is not None:
        clauses.append(FactAssertion.excl_key == excl_key)
    return select(FactAssertion).where(and_(*clauses))


def _incumbent_current(
    db: Session,
    subject_type: str,
    subject_id: int,
    fact_key: str,
    excl_key: str,
) -> FactAssertion | None:
    return db.scalars(
        _base_query(subject_type, subject_id, fact_key, state="current", excl_key=excl_key)
        .order_by(FactAssertion.id.desc())
        .limit(1)
    ).first()


def _related_subjects(entity: Any) -> list[tuple[str, int]]:
    asset: Asset | None
    if isinstance(entity, Asset):
        asset = entity
    elif isinstance(entity, Interface):
        asset = entity.asset
    else:
        return []
    subjects = [("asset", int(asset.id))]
    if asset.chassis_id:
        subjects.append(("chassis", int(asset.chassis_id)))
    return subjects


def _cross_fact_incumbents(
    db: Session,
    entity: Any,
    applying_fact_key: str,
) -> list[FactAssertion]:
    """Current rows on related subjects for R-H (name vs os, etc.)."""
    seen: set[tuple[str, int, str, str]] = set()
    rows: list[FactAssertion] = []
    for subject_type, subject_id in _related_subjects(entity):
        for row in db.scalars(
            _base_query(subject_type, subject_id, state="current").order_by(
                FactAssertion.id.desc()
            )
        ).all():
            if row.fact_key == applying_fact_key:
                continue
            key = (row.subject_type, row.subject_id, row.fact_key, row.excl_key)
            if key in seen:
                continue
            seen.add(key)
            rows.append(row)
    return rows


def current(
    db: Session,
    subject_type: str,
    subject_id: int,
    fact_key: str,
    *,
    excl_key: str | None = None,
    apply_ttl: bool = False,
) -> FactAssertion | None:
    """R-G: absent → None (caller must declare I2)."""
    spec = get_fact_spec(fact_key)
    if excl_key is not None:
        row = _incumbent_current(db, subject_type, subject_id, fact_key, excl_key)
        if row and apply_ttl:
            _maybe_stale(db, row, spec)
        return row

    rows = list(
        db.scalars(
            _base_query(subject_type, subject_id, fact_key, state="current").order_by(
                FactAssertion.id.desc()
            )
        ).all()
    )
    if not rows:
        return None
    if spec.cardinality == "single" and len(rows) > 1:
        # single cardinality: prefer highest authority then newest
        rows.sort(key=lambda r: (r.authority, r.id), reverse=True)
    row = rows[0]
    if apply_ttl:
        _maybe_stale(db, row, spec)
    return row


def current_map(
    db: Session,
    subject_type: str,
    subject_ids: list[int],
    fact_keys: list[str],
) -> dict[tuple[int, str], FactAssertion | None]:
    if not subject_ids or not fact_keys:
        return {}
    rows = db.scalars(
        select(FactAssertion).where(
            FactAssertion.subject_type == subject_type,
            FactAssertion.subject_id.in_(subject_ids),
            FactAssertion.fact_key.in_(fact_keys),
            FactAssertion.state == "current",
        )
    ).all()
    grouped: dict[tuple[int, str], list[FactAssertion]] = {}
    for row in rows:
        grouped.setdefault((row.subject_id, row.fact_key), []).append(row)
    out: dict[tuple[int, str], FactAssertion | None] = {}
    for sid in subject_ids:
        for fk in fact_keys:
            bucket = grouped.get((sid, fk))
            if not bucket:
                out[(sid, fk)] = None
                continue
            spec = get_fact_spec(fk)
            if spec.cardinality == "single" and len(bucket) > 1:
                bucket.sort(key=lambda r: (r.authority, r.id), reverse=True)
            out[(sid, fk)] = bucket[0]
    return out


def history(
    db: Session,
    subject_type: str,
    subject_id: int,
    fact_key: str,
    *,
    excl_key: str | None = None,
) -> list[FactAssertion]:
    q = _base_query(subject_type, subject_id, fact_key, excl_key=excl_key).order_by(
        FactAssertion.valid_from.asc(),
        FactAssertion.id.asc(),
    )
    return list(db.scalars(q).all())


def _maybe_stale(db: Session, row: FactAssertion, spec) -> None:
    """R-E: mark stale when ttl_window exceeded for this fact/source."""
    if spec.ttl_window is None:
        return
    ttl = spec.ttl_window
    if row.last_seen_at and utcnow() - row.last_seen_at > ttl:
        row.state = "stale"
        row.valid_to = utcnow()
        db.flush()

def _check_rh_conflict(
    db: Session,
    entity: Any,
    fact_key: str,
    value_norm: str,
    authority: int,
    *,
    human_confirm: bool,
) -> ApplyResult | None:
    global conflict_review_count
    if human_confirm:
        return None
    for other in _cross_fact_incumbents(db, entity, fact_key):
        if not semantic_contradiction(
            incumbent_fact_key=other.fact_key,
            incumbent_value=other.value_norm,
            new_fact_key=fact_key,
            new_value=value_norm,
        ):
            continue
        if authority < other.authority or other.source == "manual":
            conflict_review_count += 1
            return ApplyResult(
                action="conflict_review",
                incumbent=other,
                conflict={
                    "incumbent_fact_key": other.fact_key,
                    "incumbent_value": other.value_norm,
                    "incumbent_source": other.source,
                    "incumbent_authority": other.authority,
                    "new_fact_key": fact_key,
                    "new_value": value_norm,
                },
            )
    return None


def _within_readmission_cooldown(
    db: Session,
    subject_type: str,
    subject_id: int,
    fact_key: str,
    excl_key: str,
    value_norm: str,
    observed_at: datetime,
    new_authority: int,
) -> FactAssertion | None:
    """R-F: block weak re-admission shortly after stronger supersession."""
    recent = db.scalars(
        select(FactAssertion)
        .where(
            FactAssertion.subject_type == subject_type,
            FactAssertion.subject_id == subject_id,
            FactAssertion.fact_key == fact_key,
            FactAssertion.excl_key == excl_key,
            FactAssertion.value_norm == value_norm,
            FactAssertion.state == "superseded",
        )
        .order_by(FactAssertion.valid_to.desc(), FactAssertion.id.desc())
        .limit(1)
    ).first()
    if not recent or not recent.valid_to:
        return None
    superseder = None
    if recent.superseded_by_id:
        superseder = db.get(FactAssertion, recent.superseded_by_id)
    if superseder and superseder.authority >= new_authority:
        if observed_at - recent.valid_to < READMISSION_COOLDOWN:
            return superseder
    return None


def apply_observation(
    db: Session,
    *,
    fact_key: str,
    entity: Asset | Interface | Any,
    value: Any,
    source: str,
    observed_at: datetime | None = None,
    confidence: float | None = None,
    value_json: dict | None = None,
    human_confirm: bool = False,
) -> ApplyResult:
    """Atomic, idempotent fact transition (R-A..R-H)."""
    global weak_evidence_count

    spec = get_fact_spec(fact_key)
    src = str(source or "").lower()
    if src not in spec.sources and src != "manual":
        raise ValueError(f"source {source!r} not allowed for {fact_key}")

    subject_type, subject_id = subject_of(fact_key, entity)
    excl_key = spec.excl_key(subject_type, entity, value)
    value_norm = _norm_value(value)
    authority = authority_for(src)
    conf = confidence if confidence is not None else default_confidence(src)
    when = observed_at or utcnow()

    incumbent = _incumbent_current(db, subject_type, subject_id, fact_key, excl_key)

    rh = _check_rh_conflict(
        db, entity, fact_key, value_norm, authority, human_confirm=human_confirm
    )
    if rh is not None:
        return rh

    if incumbent is None:
        row = _new_assertion(
            subject_type=subject_type,
            subject_id=subject_id,
            fact_key=fact_key,
            excl_key=excl_key,
            value_norm=value_norm,
            value_json=value_json,
            source=src,
            authority=authority,
            confidence=conf,
            observed_at=when,
            state="current",
        )
        db.add(row)
        db.flush()
        return ApplyResult(action="created", assertion=row)

    # R-A refresh
    if incumbent.value_norm == value_norm and incumbent.source == src:
        incumbent.last_seen_at = when
        incumbent.observed_at = when
        if confidence is not None:
            incumbent.confidence = conf
        db.flush()
        return ApplyResult(action="refresh", assertion=incumbent, incumbent=incumbent)

    # R-H same-slot contradiction against manual name claims
    if (
        incumbent.fact_key == fact_key
        and semantic_contradiction(
            incumbent_fact_key=incumbent.fact_key,
            incumbent_value=incumbent.value_norm,
            new_fact_key=fact_key,
            new_value=value_norm,
        )
        and authority < incumbent.authority
        and not human_confirm
    ):
        conflict_review_count += 1
        return ApplyResult(
            action="conflict_review",
            incumbent=incumbent,
            conflict={
                "incumbent_value": incumbent.value_norm,
                "incumbent_source": incumbent.source,
                "new_value": value_norm,
                "new_source": src,
            },
        )

    # R-D manual pinned
    if incumbent.source == "manual" and src != "manual" and not human_confirm:
        weak = _new_assertion(
            subject_type=subject_type,
            subject_id=subject_id,
            fact_key=fact_key,
            excl_key=excl_key,
            value_norm=value_norm,
            value_json=value_json,
            source=src,
            authority=authority,
            confidence=conf,
            observed_at=when,
            state="historical",
            reason="weak_evidence",
        )
        db.add(weak)
        weak_evidence_count += 1
        db.flush()
        return ApplyResult(action="weak_evidence", assertion=weak, incumbent=incumbent)

    # R-F readmission cooldown (anti-flap)
    cooled = _within_readmission_cooldown(
        db, subject_type, subject_id, fact_key, excl_key, value_norm, when, authority
    )
    if cooled is not None and authority < cooled.authority:
        weak = _new_assertion(
            subject_type=subject_type,
            subject_id=subject_id,
            fact_key=fact_key,
            excl_key=excl_key,
            value_norm=value_norm,
            value_json=value_json,
            source=src,
            authority=authority,
            confidence=conf,
            observed_at=when,
            state="historical",
            reason="cooldown",
        )
        db.add(weak)
        db.flush()
        return ApplyResult(action="cooldown", assertion=weak, incumbent=cooled)

    # R-C weak evidence
    if authority < incumbent.authority and not human_confirm:
        weak = _new_assertion(
            subject_type=subject_type,
            subject_id=subject_id,
            fact_key=fact_key,
            excl_key=excl_key,
            value_norm=value_norm,
            value_json=value_json,
            source=src,
            authority=authority,
            confidence=conf,
            observed_at=when,
            state="historical",
            reason="weak_evidence",
        )
        db.add(weak)
        weak_evidence_count += 1
        db.flush()
        return ApplyResult(action="weak_evidence", assertion=weak, incumbent=incumbent)

    # R-B supersession — demote incumbent before inserting new current (partial unique index).
    incumbent.valid_to = when
    incumbent.state = "superseded"
    db.flush()

    new_row = _new_assertion(
        subject_type=subject_type,
        subject_id=subject_id,
        fact_key=fact_key,
        excl_key=excl_key,
        value_norm=value_norm,
        value_json=value_json,
        source=src,
        authority=authority,
        confidence=conf,
        observed_at=when,
        state="current",
    )
    db.add(new_row)
    db.flush()

    incumbent.superseded_by_id = new_row.id
    db.flush()
    return ApplyResult(action="superseded", assertion=new_row, incumbent=incumbent)


def canonical_name_for_entity(db: Session, entity: Asset | Interface) -> str | None:
    """Chassis-level canonical name for a NIC member (W1 read helper)."""
    subject_type, subject_id = subject_of("asset.name", entity)
    row = current(db, subject_type, subject_id, "asset.name")
    return row.value_norm if row else None


def expire_stale_facts(db: Session, *, now: datetime | None = None) -> int:
    """R-E batch: mark expired currents as stale (ttl_window None → skip)."""
    when = now or utcnow()
    changed = 0
    rows = db.scalars(select(FactAssertion).where(FactAssertion.state == "current")).all()
    for row in rows:
        spec = get_fact_spec(row.fact_key)
        if spec.ttl_window is None:
            continue
        if row.last_seen_at and when - row.last_seen_at > spec.ttl_window:
            row.state = "stale"
            row.valid_to = when
            changed += 1
    if changed:
        db.flush()
    return changed
