+#!/usr/bin/env python3
+"""W5.2 gate di equivalenza G5 (read-only): risoluzione AP is_current vs contestuale.
+
+Confronta, per ogni associazione Wi-Fi che ricade sul path ap_ip (dopo id/mac), il
+detentore `is_current` (vecchio) col detentore `resolve_asset_by_ip_at(@observed_at)`
+(nuovo). Enumera OGNI differenza per id. Eseguire a writer fermi (collector stop).
+
+  sudo docker compose stop collector
+  sudo docker compose exec -T api python3 - < scripts/w5_gate.py
+  sudo docker compose start collector
+"""
+
+from __future__ import annotations
+
+import sys
+from datetime import datetime
+from pathlib import Path
+
+API = Path("/app") if Path("/app/app").exists() else Path(__file__).resolve().parents[1] / "api"
+if str(API) not in sys.path:
+    sys.path.insert(0, str(API))
+
+from sqlalchemy import create_engine, select  # noqa: E402
+from sqlalchemy.orm import sessionmaker  # noqa: E402
+
+from app.config import get_settings  # noqa: E402
+from app.models import Asset, Interface, IpAddress  # noqa: E402
+from app.services.identity import find_asset_by_mac, resolve_asset_by_ip_at  # noqa: E402
+from app.services.macutil import normalize_mac  # noqa: E402
+from app.services.topology import _asset_link, _parse_observed_at  # noqa: E402
+
+
+def main():
+    settings = get_settings()
+    db = sessionmaker(bind=create_engine(f"sqlite:///{settings.sqlite_path}"))()
+    now = datetime.utcnow()
+    try:
+        wifi = ip_path = diffs = 0
+        rows = []
+        for asset in db.scalars(select(Asset)).all():
+            link = _asset_link(asset)
+            if link.get("media") != "wifi" and not link.get("ap_mac"):
+                continue
+            wifi += 1
+            if link.get("ap_asset_id"):
+                continue  # path id: identico
+            ap_mac = normalize_mac(link.get("ap_mac") or "")
+            if ap_mac:
+                m = find_asset_by_mac(db, ap_mac)
+                if m and m.id != asset.id:
+                    continue  # path mac: identico
+            ap_ip = str(link.get("ap_ip") or "").strip()
+            if not ap_ip:
+                continue
+            ip_path += 1
+            old = db.scalar(
+                select(Asset)
+                .join(Interface)
+                .join(IpAddress)
+                .where(IpAddress.ip == ap_ip, IpAddress.is_current == True)  # noqa: E712
+            )
+            old_id = old.id if old and old.id != asset.id else None
+            observed_at = _parse_observed_at(link.get("observed_at")) or now
+            new = resolve_asset_by_ip_at(db, ap_ip, observed_at)
+            new_id = new.id if new and new.id != asset.id else None
+            if old_id != new_id:
+                diffs += 1
+                rows.append(
+                    f"  client={asset.id} ap_ip={ap_ip} observed_at={link.get('observed_at')} "
+                    f"old(is_current)={old_id} new(contextual)={new_id}"
+                )
+        print(f"G5 AP-resolution: wifi_links={wifi} ip_path={ip_path} differenze={diffs}")
+        for r in rows:
+            print(r)
+    finally:
+        db.rollback()
+        db.close()
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())
